**[ 풀이 ]**
```cpp
#include <iostream>

using namespace std;

int main(void)
{
    int n, m;
    cin >> n >> m;

    for (int i = 0; i < m; i++)
    {
        for (int j = 0; j < n; j++)
        {
            cout << '*';
        }
        cout << '\n';
    }

    return 0;
}
```
**[ 해설 ]**
```cpp
#include <iostream>     // cout, cin을 사용하기 위한 헤더

using namespace std;    // std::를 생략하기 위해 사용

int main(void)
{
    int n, m;

    // n = 가로 길이(한 줄에 출력할 별 개수)
    // m = 세로 길이(출력할 줄 수)
    cin >> n >> m;

    // 세로 길이(m)만큼 반복
    // 한 번 반복할 때마다 별 한 줄을 출력
    for (int i = 0; i < m; i++)
    {
        // 가로 길이(n)만큼 반복
        // 별(*)을 n개 출력
        for (int j = 0; j < n; j++)
        {
            cout << '*';
        }

        // 한 줄의 별 출력이 끝났으므로 줄바꿈
        cout << '\n';
    }

    // 프로그램 정상 종료
    return 0;
}
```
