**[ 해설 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> solution(int k, vector<int> score)
{
    // 매일 발표되는 명예의 전당 최하위 점수를 저장할 벡터
    vector<int> answer;

    // 지금까지 나온 점수들을 저장할 벡터
    vector<int> scores;

    // 하루씩 점수를 확인
    for (int i = 0; i < score.size(); i++)
    {
        // 오늘 나온 점수를 추가
        scores.push_back(score[i]);

        // 점수를 높은 순(내림차순)으로 정렬
        // 예) [10, 100, 20] -> [100, 20, 10]
        sort(scores.begin(), scores.end(), greater<int>());

        // 아직 명예의 전당이 다 차지 않은 경우
        // (출연한 가수가 k명보다 적은 경우)
        if (scores.size() < k)
        {
            // 현재 점수들 중 가장 낮은 점수를 발표
            // 내림차순이므로 마지막 원소가 최하위 점수
            answer.push_back(scores.back());
        }
        else
        {
            // 명예의 전당이 k명 이상이면
            // 내림차순 정렬에서 (k-1)번째 인덱스가
            // 명예의 전당의 최하위 점수
            //
            // 예) k = 3
            // scores = [150, 100, 20, 10]
            //          0    1    2   3
            // 명예의 전당은 [150, 100, 20]
            // 발표 점수는 20 (= scores[2])
            answer.push_back(scores[k - 1]);
        }
    }

    // 매일의 발표 점수 반환
    return answer;
}
```
**[ 타 답안 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> solution(int k, vector<int> score)
{
    // answer : 매일 발표되는 명예의 전당 최하위 점수 저장
    // tmp : 현재 명예의 전당 점수(상위 k개)를 저장
    vector<int> answer, tmp;

    // score에 있는 점수를 하루씩 확인
    for (auto s : score)
    {
        // 오늘의 점수를 명예의 전당 후보에 추가
        tmp.push_back(s);

        // 점수를 높은 순(내림차순)으로 정렬
        // 예) [10, 100, 20] -> [100, 20, 10]
        sort(tmp.begin(), tmp.end(), greater<int>());

        // 명예의 전당에 k명보다 많은 점수가 있으면
        if (tmp.size() >= k)
        {
            // k번째 이후의 점수는 모두 제거
            // 즉, 상위 k개의 점수만 남김
            tmp.erase(tmp.begin() + k, tmp.end());
        }

        // 현재 명예의 전당에서 가장 낮은 점수
        // (내림차순이므로 마지막 원소)
        answer.push_back(tmp.back());
    }

    // 매일 발표한 점수를 반환
    return answer;
}
```
