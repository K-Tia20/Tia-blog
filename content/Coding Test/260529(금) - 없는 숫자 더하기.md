풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

int solution(vector<int> numbers) {
    int sum = 0;
    
    for (int n : numbers) {
        sum += n;
    }
    
    return 45 - sum;
}
```

```cpp
#include <string>
#include <vector>

using namespace std;

// 0~9까지 숫자 중 numbers에 없는 숫자들의 합을 구하는 함수
int solution(vector<int> numbers) {

    // numbers에 들어있는 모든 숫자의 합을 저장할 변수
    int sum = 0;

    // 향상된 for문 (range-based for)
    // numbers 배열 안의 모든 원소를 하나씩 꺼내서 n에 넣음
    for (int n : numbers) {

        // 현재 숫자 n을 sum에 계속 누적해서 더함
        sum += n;
    }

    // 0부터 9까지 모든 숫자의 합은 항상 45로 고정
    // (0 + 1 + 2 + ... + 9 = 45)

    // 따라서 전체 합(45)에서
    // 배열에 존재하는 숫자들의 합(sum)을 빼면
    // "없는 숫자들의 합"이 계산됨
    return 45 - sum;
}
```

타 답안

```cpp
#include <string>
#include <vector>

using namespace std;

// 0~9 중 numbers에 없는 숫자들의 합을 구하는 함수
int solution(vector<int> numbers) {

    // 0부터 9까지의 전체 합은 항상 45로 고정
    // (0 + 1 + 2 + ... + 9 = 45)
    int answer = 45;

    // numbers 배열의 모든 원소를 순회하면서
    // 해당 값을 answer에서 빼준다
    for (int i = 0 ; i < numbers.size() ; i++) {

        // numbers[i]는 배열의 i번째 숫자
        // 이 숫자가 "존재하는 숫자"이므로
        // 전체 합에서 제거하는 방식
        answer -= numbers[i];
    }

    // 결과적으로 남는 값은
    // 0~9 전체 합(45) - 존재하는 숫자들의 합
    // => 존재하지 않는 숫자들의 합
    return answer;
}
```
