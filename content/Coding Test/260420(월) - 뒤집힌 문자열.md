풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

string solution(string my_string) 
{
    string answer = "";

    for(int i = my_string.size(); i > 0; i--) {
        answer += my_string[i - 1];
    }
    return answer;
}
```

타 답안

```cpp
#include <bits/stdc++.h> // C++ 표준 라이브러리 대부분을 한 번에 포함 (실무에선 비권장)

using namespace std;

string solution(string my_string) {
    // rbegin() : 문자열의 끝(뒤)부터 시작하는 반복자
    // rend()   : 문자열의 시작 "이전"까지를 가리키는 반복자
    // → [끝 → 시작] 방향으로 범위를 잡음

    // string(시작 반복자, 끝 반복자)
    // → 해당 범위를 이용해서 새로운 문자열 생성

    return string(my_string.rbegin(), my_string.rend());
    // 결과적으로 문자열을 뒤집어서 새 string으로 반환
}
```

```cpp
#include <string>
#include <vector>      // (이 문제에서는 사실 필요 없음)
#include <algorithm>   // reverse() 함수 사용을 위해 필요

using namespace std;

string solution(string my_string) {
    // reverse(시작 반복자, 끝 반복자)
    // → 해당 범위를 "제자리에서" 뒤집는다 (in-place)
    reverse(my_string.begin(), my_string.end());

    // 뒤집힌 문자열 반환
    return my_string;
}
```
