풀이 코드

```cpp
#include <iostream>
#include <vector>
using namespace std;

int solution(int n)
{
    int answer = 0;
    vector<int> vec;
    int num =0;
    
    while (n > 0)
    {
        num = n % 10;
        vec.push_back(num);
        n /= 10;
    }
    for (int i = 0; i < vec.size(); i++)
    {
        answer += vec[i];
    }
    return answer;
}
```
