풀이 코드

```cpp
#include <string>   // string 사용 헤더 (현재 코드에서는 실제로 사용 안 함)
#include <vector>   // vector 사용을 위한 헤더

using namespace std; // std::vector 등을 간단하게 사용하기 위해 작성


/*
    solution 함수

    매개변수:
    long long n
    -> 매우 큰 정수를 받음

    반환값:
    vector<int>

    즉:
    숫자의 각 자릿수를 저장한 배열 반환
*/
vector<int> solution(long long n) 
{
    /*
        결과를 저장할 vector 생성

        vector<int>
        -> int형 숫자를 여러 개 저장하는 배열

        처음 상태:
        answer = {}
    */
    vector<int> answer;


    /*
        while 반복문

        형식:

        while(조건)

        의미:
        조건이 true인 동안 계속 반복

        ------------------------------------------------

        현재 조건:

        n > 0

        즉:
        n이 0보다 큰 동안 반복
    */
    while(n > 0)
    {
        /*
            n % 10

            -> 10으로 나눈 나머지

            ------------------------------------------------

            이 계산은
            "맨 뒤 자리 숫자"를 구할 때 사용

            ------------------------------------------------

            예시:

            n = 12345

            12345 % 10

            결과:
            5

            ------------------------------------------------

            또 예시:

            1234 % 10 = 4
            123 % 10 = 3

            즉:
            항상 마지막 자리 숫자가 나옴
        */
        answer.push_back(n % 10);


        /*
            push_back()

            vector 맨 뒤에 값 추가

            ------------------------------------------------

            예시:

            answer = {}

            push_back(5)

            결과:
            {5}

            ------------------------------------------------

            또 추가:

            push_back(4)

            결과:
            {5, 4}
        */


        /*
            n /= 10

            의미:

            n = n / 10

            과 같은 뜻

            ------------------------------------------------

            숫자를 10으로 나누면
            마지막 자리 숫자가 제거됨

            ------------------------------------------------

            예시:

            n = 12345

            n /= 10

            결과:
            1234

            ------------------------------------------------

            또 예시:

            1234 / 10 = 123
            123 / 10 = 12

            ------------------------------------------------

            왜 가능한가?

            int끼리 나누면 소수점 버림 처리되기 때문

            즉:

            1234 / 10

            = 123.4

            ↓

            123
        */
        n /= 10;                
    }


    /*
        최종 결과 반환

        예시:

        n = 12345

        진행 과정:

        12345 % 10 -> 5 저장
        1234 % 10 -> 4 저장
        123 % 10 -> 3 저장
        12 % 10 -> 2 저장
        1 % 10 -> 1 저장

        결과:

        {5, 4, 3, 2, 1}

        ------------------------------------------------

        즉:
        숫자가 뒤집힌 순서로 저장됨
    */
    return answer;
}
```
