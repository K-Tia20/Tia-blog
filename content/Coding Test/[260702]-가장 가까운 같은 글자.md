**[ 풀이 ]**
```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(string s)
{
    vector<int> answer;

    int last[26];

    for (int i = 0; i < 26; i++)
    {
        last[i] = -1;
    }

    for (int i = 0; i < s.size(); i++)
    {
        int idx = s[i] - 'a';

        if (last[idx] == -1)
        {
            answer.push_back(-1);
        }
        else
        {
            answer.push_back(i - last[idx]);
        }

        last[idx] = i;
    }

    return answer;
}
```

**[ 해설 ]**
```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(string s)
{
    // 결과를 저장할 벡터
    vector<int> answer;

    // 각 알파벳이 마지막으로 등장한 위치를 저장하는 배열
    // 인덱스 0 = 'a', 1 = 'b', ..., 25 = 'z'
    int last[26];

    // 처음에는 어떤 문자도 등장하지 않았으므로
    // 모든 값을 -1로 초기화
    for (int i = 0; i < 26; i++)
    {
        last[i] = -1;
    }

    // 문자열을 처음부터 끝까지 한 글자씩 확인
    for (int i = 0; i < s.size(); i++)
    {
        // 현재 문자를 배열의 인덱스로 변환
        // 예)
        // 'a' - 'a' = 0
        // 'b' - 'a' = 1
        // 'c' - 'a' = 2
        int idx = s[i] - 'a';

        // 현재 문자가 처음 나온 경우
        // last[idx]가 -1이면 이전에 등장한 적이 없음
        if (last[idx] == -1)
        {
            // 앞에 같은 문자가 없으므로 -1 저장
            answer.push_back(-1);
        }
        else
        {
            // 이전에 같은 문자가 있었다면
            // 현재 위치 - 마지막으로 나온 위치
            // = 가장 가까운 같은 문자와의 거리
            answer.push_back(i - last[idx]);
        }

        // 현재 문자의 위치를 마지막 등장 위치로 갱신
        // 다음에 같은 문자가 나오면 이 위치를 사용하게 된다.
        last[idx] = i;
    }

    // 최종 결과 반환
    return answer;
}
```
