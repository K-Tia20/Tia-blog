풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

// n이 어떤 수의 제곱수인지 확인하는 함수
// 제곱수라면 (x+1)^2 반환
// 아니라면 -1 반환
long long solution(long long n)
{
    // x를 1부터 시작해서 검사
    // x * x <= n 까지만 반복하면 됨
    // 이유:
    // x*x가 n보다 커지는 순간 더 볼 필요가 없음
    for (long long x = 1; x * x <= n; x++)
    {
        // 현재 x의 제곱이 n과 같은지 확인
        if (x * x == n)
        {
            // n이 x의 제곱수라면
            // (x + 1)의 제곱을 반환
            return (x + 1) * (x + 1);
        }
    }

    // 반복문이 끝날 때까지 못 찾았다면
    // n은 제곱수가 아님
    return -1;
}
```

타 답안

```cpp
#include <cmath>    // sqrt, pow 사용

// n이 제곱수인지 확인하는 함수
long long solution(long long n)
{
    // sqrt(n)
    // n의 제곱근을 구함
    // 예:
    // sqrt(9)  -> 3
    // sqrt(16) -> 4
    // sqrt(10) -> 3.16...
    long long x = sqrt(n);

    // x * x 가 n과 같다면
    // n은 완전제곱수
    if (x * x == n)
    {
        // pow(a, b)
        // a의 b제곱
        // 즉 (x+1)^2 계산
        return pow(x + 1, 2);
    }

    // 완전제곱수가 아니면 -1 반환
    return -1;
}
```
