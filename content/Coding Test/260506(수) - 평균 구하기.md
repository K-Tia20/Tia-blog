#include <string>
#include <vector>

using namespace std;

double solution(vector<int> arr) 
{
    double answer = 0;
    int aaa = 0;
    for(int i=0; i < arr.size(); i++)
    {
        aaa += arr[i];
    }
    answer = (double)aaa/arr.size();
    return answer;
}
