풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(int n)
{
    vector<int> answer; 
    int k;
    
    if (n % 2 == 0)
    {
        k = n - 1;
    }else
    {
        k = n;
    }
    for (int i = 1; i <= k; i += 2)
    {
        answer.push_back(i);
    }
    return answer;
}
```
