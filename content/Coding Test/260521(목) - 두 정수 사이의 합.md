풀이 코드

```
long long solution(int a, int b)
{
    // 최종 합계를 저장할 변수
    // 숫자의 범위가 커질 수 있으므로 long long 사용
    long long answer = 0;

    // a와 b 중 더 작은 값을 start에 저장
    // 예: a=5, b=3 이면 start=3
    int start = min(a, b);

    // a와 b 중 더 큰 값을 end에 저장
    // 예: a=5, b=3 이면 end=5
    int end = max(a, b);

    // start부터 end까지 1씩 증가하면서 반복
    // i는 현재 더하고 있는 숫자
    for (int i = start; i <= end; i++)
    {
        // answer에 현재 숫자 i를 누적해서 더함
        // 예:
        // 첫 반복: answer = 0 + 3
        // 두 번째: answer = 3 + 4
        // 세 번째: answer = 7 + 5
        answer += i;
    }

    // 모든 숫자를 더한 최종 결과 반환
    return answer;
}
```

타 답안

```
long long solution(int A, int B)
{
    // 모든 숫자의 합을 저장할 변수
    // 값이 커질 수 있으므로 long long 사용
    long long answer = 0;

    // 삼항 연산자 사용
    // A가 B보다 작으면 a에 A 저장
    // 아니면 a에 B 저장
    // 즉, 두 수 중 작은 값을 a에 저장
    int a = A < B ? A : B;

    // 삼항 연산자 사용
    // A가 B보다 작으면 b에 B 저장
    // 아니면 b에 A 저장
    // 즉, 두 수 중 큰 값을 b에 저장
    int b = A < B ? B : A;

    // 작은 값 a부터 큰 값 b까지 반복
    // i가 1씩 증가하면서 모든 숫자를 순서대로 확인
    for (int i = a; i <= b; i++)

        // 현재 숫자 i를 answer에 누적해서 더함
        answer += i;

    // 최종 합계 반환
    return answer;
}
```
