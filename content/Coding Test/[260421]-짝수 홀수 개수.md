풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(vector<int> num_list) 
{
    
    int a = 0;
    int b = 0;
    for(int i = 0; i < num_list.size(); i++) 
    {
        if(num_list[i] % 2 == 0) 
        {
            a = a + 1;
        } else {
            b = b + 1;
        }
    }
    vector<int> answer(2);
    answer[0] = a;
    answer[1] = b;

    return answer;
}
```

타 답안

```cpp
#include <string>
#include <vector>

using namespace std;

// 정수 배열을 받아서
// [짝수 개수, 홀수 개수]를 담은 벡터를 반환하는 함수
vector<int> solution(vector<int> num_list) {

    // 크기가 2이고, 모든 값이 0으로 초기화된 벡터 생성
    // answer[0] → 짝수 개수
    // answer[1] → 홀수 개수
    vector<int> answer(2, 0);

    // num_list 안의 값을 하나씩 꺼내서 num에 넣음
    // (range-based for문: 배열을 앞에서부터 끝까지 자동 반복)
    for (int num : num_list) {

        // num % 2 결과
        // → 0이면 짝수
        // → 1이면 홀수

        // 그 값을 인덱스로 사용해서 해당 위치를 1 증가
        // 짝수면 answer[0]++
        // 홀수면 answer[1]++
        answer[num % 2]++;
    }

    // [짝수 개수, 홀수 개수] 반환
    return answer;
}
```

```cpp
#include <string>
#include <vector>

using namespace std;

// 정수 배열을 받아서 [짝수 개수, 홀수 개수]를 반환
vector<int> solution(vector<int> num_list) {
    
    // 크기 2짜리 벡터 생성 + 모든 값 0으로 초기화
    // answer[0] = 짝수 개수
    // answer[1] = 홀수 개수
    vector<int> answer(2, 0);

    // 배열 길이 저장 (반복문에서 계속 size() 안 쓰려고 따로 빼둠)
    int length = num_list.size();

    // 0부터 length-1까지 반복
    for(int i = 0; i < length; i++){
        
        // 현재 값이 짝수라면
        if(num_list[i] % 2 == 0){
            // 짝수 개수 +1
            answer[0] += 1;
        } 
        else{
            // 홀수 개수 +1
            answer[1] += 1;
        }
    }

    // 결과 반환
    return answer;
}
```
