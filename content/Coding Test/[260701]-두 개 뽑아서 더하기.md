**[ 풀이 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> solution(vector<int> numbers)
{
    vector<int> answer;

    for (int i = 0; i < numbers.size() - 1; i++)
    {
        for (int j = i + 1; j < numbers.size(); j++)
        {
            answer.push_back(numbers[i] + numbers[j]);
        }
    }

    sort(answer.begin(), answer.end());

    answer.erase(unique(answer.begin(), answer.end()), answer.end());

    return answer;
}
```

**[ 해설 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> solution(vector<int> numbers)
{
    // 두 수를 더한 값을 저장할 벡터
    vector<int> answer;

    // 첫 번째 숫자 선택
    for (int i = 0; i < numbers.size() - 1; i++)
    {
        // 첫 번째 숫자 뒤에 있는 숫자 선택
        // 같은 인덱스를 두 번 사용하지 않기 위해 i+1부터 시작
        for (int j = i + 1; j < numbers.size(); j++)
        {
            // 두 숫자의 합을 벡터에 저장
            answer.push_back(numbers[i] + numbers[j]);
        }
    }

    // 저장된 숫자를 오름차순으로 정렬
    sort(answer.begin(), answer.end());

    // 같은 숫자가 여러 개 있으면 하나만 남기고 제거
    answer.erase(unique(answer.begin(), answer.end()), answer.end());

    // 결과 반환
    return answer;
}
```
**[ 타 답안 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>
#include <set>

using namespace std;

vector<int> solution(vector<int> numbers) {

    // 최종 결과를 저장할 벡터
    vector<int> answer;

    // 중복을 허용하지 않고 자동으로 오름차순 정렬해주는 set
    // 두 수의 합을 여기에 저장하면 같은 값은 한 번만 저장된다.
    set<int> st;

    // 첫 번째 숫자를 하나씩 선택
    for (int i = 0; i < numbers.size(); ++i)
    {
        // 첫 번째 숫자 뒤에 있는 숫자만 선택
        // j를 i+1부터 시작하는 이유
        // 1. 같은 인덱스를 두 번 선택하지 않기 위해 (예: numbers[0] + numbers[0])
        // 2. 이미 계산한 조합을 또 계산하지 않기 위해
        //    예) 2+3과 3+2는 같은 조합이므로 한 번만 계산
        for (int j = i + 1; j < numbers.size(); ++j)
        {
            // 선택한 두 숫자를 더해서 set에 저장
            // 예)
            // numbers[i] = 2
            // numbers[j] = 3
            // => 5 저장

            // set은 중복을 허용하지 않기 때문에
            // 이미 5가 저장되어 있다면 자동으로 무시된다.
            st.insert(numbers[i] + numbers[j]);
        }
    }

    // set에 저장된 값을 vector(answer)에 복사
    //
    // begin() : 첫 번째 원소
    // end()   : 마지막 다음 위치
    //
    // set은 자동으로 오름차순 정렬되어 있으므로
    // answer에도 정렬된 상태로 복사된다.
    answer.assign(st.begin(), st.end());

    // 최종 결과 반환
    return answer;
}
```
