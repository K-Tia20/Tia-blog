풀이 코드

```cpp
#include <string>   // string 자료형 사용을 위한 헤더
#include <vector>   // 현재 코드에서는 사용하지 않지만 포함되어 있는 헤더

using namespace std; // std::string 등을 간단히 쓰기 위해 사용


/*
    solution 함수

    매개변수:
    string s
    -> 문자열 형태의 숫자를 전달받음

    예시:
    "1234"
    "-1234"
    "+1234"

    반환값:
    int형 숫자
*/
int solution(string s) 
{
    /*
        최종 숫자를 저장할 변수

        처음에는 0으로 시작

        예시:
        answer = 0
    */
    int answer = 0;


    /*
        숫자를 읽기 시작할 위치 저장 변수

        기본값:
        0

        ------------------------------------------------

        예시:

        s = "1234"

        숫자가 처음부터 시작하므로
        snum = 0

        ------------------------------------------------

        하지만:

        s = "-1234"

        에서는
        0번째 문자가 '-' 이므로

        실제 숫자는 1번째부터 시작

        그래서:
        snum = 1
    */
    int snum = 0;


    /*
        부호 저장 변수

        기본값:
        1

        의미:
        양수

        ------------------------------------------------

        만약 음수라면:

        giho = -1

        로 변경
    */
    int giho = 1;


    /*
        문자열 첫 글자가 '-' 인지 확인

        s[0]
        -> 문자열의 첫 번째 문자

        ------------------------------------------------

        예시:

        s = "-1234"

        s[0]
        -> '-'
    */
    if (s[0] == '-')
    {
        /*
            음수이므로 부호를 -1로 변경

            나중에 마지막 결과에 곱해줌
        */
        giho = -1;


        /*
            숫자는 1번째부터 시작하므로

            시작 위치를 1로 변경

            즉:
            '-' 는 건너뜀
        */
        snum = 1;
    }


    /*
        문자열 첫 글자가 '+' 인지 확인

        예시:

        s = "+1234"
    */
    else if (s[0] == '+')
    {
        /*
            '+' 역시 숫자가 아니므로
            1번째부터 읽기 시작

            즉:
            '+' 건너뜀
        */
        snum = 1;
    }


    /*
        문자열 숫자를 하나씩 읽는 반복문

        ------------------------------------------------

        i = snum
        -> 숫자가 시작되는 위치부터 시작

        i < s.size()
        -> 문자열 끝까지 반복

        i++
        -> 반복할 때마다 1 증가

        ------------------------------------------------

        예시:

        s = "1234"

        반복 과정:

        i=0 -> '1'
        i=1 -> '2'
        i=2 -> '3'
        i=3 -> '4'
    */
    for (int i = snum; i < s.size(); i++)
    {
        /*
            기존 숫자 자릿수 한 칸 올리기

            answer = answer * 10

            ------------------------------------------------

            예시:

            answer = 12

            12 * 10

            결과:
            120

            ------------------------------------------------

            왜 하는가?

            새로운 숫자를 뒤에 붙일 자리 만들기 위해
        */
        answer = answer * 10;


        /*
            현재 문자 숫자를 실제 숫자로 변환해서 더함

            ------------------------------------------------

            예시:

            s[i] = '3'

            문자 '3' 자체는 아스키 코드 문자

            숫자 3으로 바꾸려면:

            '3' - '0'

            계산

            ------------------------------------------------

            아스키 코드 예시:

            '0' = 48
            '1' = 49
            '2' = 50
            '3' = 51

            따라서:

            '3' - '0'

            = 51 - 48
            = 3

            ------------------------------------------------

            예시 진행:

            answer = 120

            현재 문자:
            '3'

            answer += 3

            결과:
            123
        */
        answer += s[i] - '0';
    }


    /*
        최종 결과 반환

        ------------------------------------------------

        양수인 경우:

        answer * 1

        그대로 반환

        ------------------------------------------------

        음수인 경우:

        answer * -1

        음수로 반환

        ------------------------------------------------

        예시:

        answer = 1234
        giho = -1

        결과:
        -1234
    */
    return answer * giho;
}
```

타 답안

```cpp
#include <string>

using namespace std;

int solution(string s) 
{
    // 문자열을 정수(int)로 변환
    // stoi는 +, - 부호도 자동 처리해준다
    int answer = stoi(s);

    // 변환한 정수 반환
    return answer;
}
```

```cpp
#include <string>
#include <vector>

using namespace std;

int solution(string s) {
    int answer = 0;
    int m = 1;
    for(int i = 0; i < s.size();i++)
    {
        if(s[i] >= '0' && s[i] <= '9')
        {
        answer *= 10;
        answer += (s[i] - 48);
        }
        else if(s[i] == '-')
            m = -1;
    }
    return answer*m;
}
```
