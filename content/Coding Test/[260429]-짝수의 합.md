풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

int solution(int n) 
{
    int answer = 0;
    int k;
    
    if (n % 2 == 0)
    {
        k = n;
    }else
    {
        k = n - 1;
    }
    for (int i = 0; i <= k; i += 2)
    {
        answer += i;
    }
    return answer;
}
```
