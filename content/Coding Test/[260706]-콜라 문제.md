**[ 풀이 ]**
```cpp
#include <string>
#include <vector>

using namespace std;

int solution(int a, int b, int n)
{
    int answer = 0;

    while (n >= a)
    {
        int exchange = n / a;
        int cola = exchange * b;

        answer += cola;

        n = (n % a) + cola;
    }

    return answer;
}
```

**[ 해설 ]**
```cpp
#include <string>
#include <vector>

using namespace std;

// a : 교환에 필요한 빈 병 개수
// b : 빈 병 a개를 주면 받을 수 있는 콜라 개수
// n : 처음 가지고 있는 빈 병 개수
int solution(int a, int b, int n)
{
    // 최종적으로 받게 되는 콜라의 총 개수
    int answer = 0;

    // 빈 병이 a개 이상 있는 동안 계속 교환 가능
    while (n >= a)
    {
        // 현재 가진 빈 병으로 몇 번 교환할 수 있는지 계산
        // 예) n = 20, a = 3
        //     20 / 3 = 6번 교환 가능
        int exchange = n / a;

        // 이번 교환에서 받는 콜라 개수
        // 교환 1번당 b병을 받으므로
        // 예) exchange = 6, b = 1
        //     cola = 6
        int cola = exchange * b;

        // 지금까지 받은 콜라 개수에 이번에 받은 콜라를 더함
        answer += cola;

        // 다음 반복에서 사용할 빈 병 개수 계산
        //
        // n % a : 교환하고 남은 빈 병
        // cola  : 새로 받은 콜라를 모두 마시면 생기는 빈 병
        //
        // 예)
        // n = 20, a = 3, b = 1
        //
        // 교환
        // 20개 -> 18개 사용 -> 6병 받음
        //
        // 남은 빈 병 = 20 % 3 = 2
        // 새로 생긴 빈 병 = 6
        //
        // 다음에 가지고 있는 빈 병 = 2 + 6 = 8
        n = (n % a) + cola;
    }

    // 지금까지 받은 총 콜라 개수 반환
    return answer;
}
```

```cpp
}
```
