풀이 코드

```cpp
#include <string>
#include <vector>  
//vector = 크기가 자동으로 관리되는 배열(동적 배열, 초기 데이터가 순서 없이 저장)

using namespace std;

vector<int> solution(vector<int> numbers) {

    vector<int> a(numbers.size()); 
    //미리 칸을 만들어주는 것? > a[i]를 쓰려면 1번째 칸이 존재해야해서
    //ex) numbers = [1,2,3]
    //a = [0,0,0] <- 기본값 0

    for (int i = 0; i < numbers.size(); i++) {
        a[i] = numbers[i] * 2;
    }
    
    return a;
}
```

타 답안

```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(vector<int> numbers) {
    vector<int> answer;
    for(int i = 0; i < numbers.size(); i++){
        answer.push_back(numbers[i]*2);
        //.push_back(값); <- vector 맨 뒤에 값을 하나 추가하는 함수
        //numbers[i]
        //numbers[i] * 2
        //answer.push_back(...) → 뒤에 붙이기
        //i == numbers.size()되면 끝
    }
    return answer;
}
```

#include <vector>

using namespace std;

vector<int> solution(vector<int> numbers) {
    for(int i=0; i<numbers.size(); i++)
        numbers[i] = numbers[i] << 1;
        //비트 연산자 (Bitwise Operators)  <<: 왼쪽 시프트 <- 왼쪽으로 1비트 이동=2배
        //int a = 15;    (00001111)
        //int b = a << 2;  // 왼쪽으로 2비트 시프트 (00111100, 즉 60)
    return numbers;
}
