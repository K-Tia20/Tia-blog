```cpp
#include <string>
#include <vector>

using namespace std;

string solution(int num) {
    string answer = "";
    return answer;
}
```

풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

string solution(int num) {
    string answer = "";
    
    if(num % 2 == 0)
    {
        return answer = "Even";
    }else if(num == 0)
    {
        return answer = "Even";
    }else
    {
        return answer = "Odd";
    }
}
```

타 답안

```cpp
#include <string>
#include <vector>

using namespace std;

string solution(int num) {
    string answer = "";

    return num & 1 ? "Odd" : "Even";
}
```
