풀이 코드

```cpp
#include <string>

using namespace std;

string solution(string phone_number) {
    int len = phone_number.length();
    
    string answer = "";
    
    for (int i = 0; i < len - 4; i++) {
        answer += "*";
    }
    
    answer += phone_number.substr(len - 4);
    
    return answer;
}
```

해설 코드

```cpp
#include <string>

using namespace std;

// 전화번호를 입력받아 뒤 4자리를 제외한 나머지를 '*'로 마스킹하는 함수
string solution(string phone_number) {

    // 1. 전화번호 전체 길이를 구함
    //    예: "01012345678" -> 11
    int len = phone_number.length();
    
    // 2. 결과를 저장할 문자열 초기화
    //    여기에는 '*'들과 마지막 4자리가 순서대로 들어감
    string answer = "";
    
    // 3. 앞부분을 '*'로 채우는 과정
    //    전체 길이 - 4 만큼 반복
    //    이유: 뒤 4자리는 유지해야 하므로 제외
    //
    //    예:
    //    len = 11이면 len - 4 = 7
    //    -> "*******" 생성
    for (int i = 0; i < len - 4; i++) {
        
        // 매 반복마다 '*' 하나씩 추가
        answer += "*";
    }
    
    // 4. 전화번호의 마지막 4자리 추출해서 붙이기
    //
    //    substr(startIndex)
    //    -> startIndex부터 끝까지 잘라서 반환
    //
    //    예:
    //    "01012345678".substr(7) -> "5678"
    answer += phone_number.substr(len - 4);
    
    // 5. 최종 결과 반환
    //    예: "*******5678"
    return answer;
}
```

타 답안

```cpp
#include <string>

using namespace std;

// 전화번호의 뒤 4자리를 제외하고 앞부분을 '*'로 마스킹하는 함수
string solution(string phone_number) {

    // 결과를 담을 문자열 (현재는 거의 필요 없지만 구조상 선언)
    string answer = "";

    // 1. 반복문으로 "앞부분만" 순회
    //    phone_number.size() - 4 까지만 돈다
    //
    //    이유:
    //    뒤 4자리는 그대로 두고 앞부분만 '*'로 바꿔야 하기 때문
    //
    //    예:
    //    "01012345678" (길이 11)
    //    -> 0 ~ 6 까지 (총 7개 문자)만 변경
    for (int i = 0; i < phone_number.size() - 4; i++ )
    {
        // 2. 실제 문자열 자체를 직접 수정
        //
        //    phone_number[i] = '*'
        //    → i번째 문자를 '*'로 덮어쓰기 (in-place 수정)
        //
        //    예:
        //    "01012345678"
        //    ↓
        //    "***12345678"
        phone_number[i] = '*';
    }

    // 3. 이미 수정된 phone_number를 그대로 answer에 복사
    //
    //    사실 이 줄 없이 바로 return phone_number 해도 동일함
    answer = phone_number;

    // 4. 결과 반환
    return answer;
}
```
