풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

string solution(string my_string, string letter) 
{
    string answer = "";
    for (int i = 0; i < my_string.size(); i++) 
    {
        if (my_string[i] != letter[0])
        {
            answer += my_string[i];
        }
    }
    return answer;
}
```

타 답안

```cpp
#include <string>
#include <vector>

using namespace std;

string solution(string my_string, string letter) {
    string answer = ""; // (현재 코드에서는 사용 안 함. 없어도 됨)

    // 문자열을 앞에서부터 하나씩 검사
    for(int i = 0; i < my_string.size(); i++)
    {
        // 현재 문자(my_string[i])가 제거할 문자(letter[0])와 같으면
        if(my_string[i] == letter[0])
        {
            // 해당 위치의 문자 삭제
            // erase는 지정 위치의 문자를 지우고 뒤의 문자열을 앞으로 당김
            my_string.erase(my_string.begin() + i);

            // 한 칸 당겨졌기 때문에 인덱스도 한 칸 뒤로 보정
            // 안 하면 다음 문자 하나를 건너뛰게 됨
            i--;
        }
    }

    // 최종적으로 letter가 제거된 문자열 반환
    return my_string;
}
```
