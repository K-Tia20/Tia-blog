풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

int solution(vector<int> absolutes, vector<bool> signs) {
    int answer = 0;

    for (int i = 0; i < absolutes.size(); i++)
    {
        if (signs[i] == true)
        {
            answer += absolutes[i];
        }
        else
        {
            answer -= absolutes[i];
        }
    }

    return answer;
}
```

해석 코드

```cpp
#include <string>
#include <vector>

using namespace std;

// solution 함수
// absolutes : 숫자의 절댓값들이 들어있는 배열
// signs : 각 숫자의 부호(true = 양수, false = 음수)가 들어있는 배열
int solution(vector<int> absolutes, vector<bool> signs) 
{
    // 최종 합계를 저장할 변수
    // 처음에는 아무 값도 더하지 않았으므로 0으로 시작
    int answer = 0;

    // 배열의 처음부터 끝까지 반복하는 반복문
    // i는 배열의 인덱스 번호(순서 번호)
    // i = 0 → 첫 번째
    // i = 1 → 두 번째
    // i = 2 → 세 번째 ...
    //
    // absolutes.size()
    // → 배열의 전체 길이(개수)를 반환
    for (int i = 0; i < absolutes.size(); i++)
    {
        // signs[i] 가 true 인지 확인
        // true 라는 것은 해당 숫자가 양수라는 뜻
        if (signs[i] == true)
        {
            // answer에 현재 숫자를 더함
            //
            // 예:
            // answer = 3
            // absolutes[i] = 5
            //
            // answer += 5
            // → answer = 8
            answer += absolutes[i];
        }
        else
        {
            // signs[i] 가 false 라면 음수라는 뜻
            //
            // 따라서 현재 숫자를 빼줌
            //
            // 예:
            // answer = 10
            // absolutes[i] = 4
            //
            // answer -= 4
            // → answer = 6
            answer -= absolutes[i];
        }
    }

    // 모든 계산이 끝난 최종 결과 반환
    return answer;
}
```

타 답안

```cpp
#include <vector>

// solution 함수
// absolutes : 절댓값들이 들어있는 배열
// signs : 부호 정보가 들어있는 배열
//
// true  = 양수
// false = 음수
int solution(std::vector<int> absolutes, std::vector<bool> signs) 
{
    // 최종 결과를 저장할 변수
    int answer = 0;

    // signs 배열 크기만큼 반복
    //
    // i는 배열의 순서 번호(인덱스)
    // i = 0 → 첫 번째 값
    // i = 1 → 두 번째 값
    for(int i = 0; i < signs.size(); i++)
    {
        // signs[i] ? 1 : -1
        //
        // 삼항 연산자라고 부름
        //
        // 형식:
        // 조건 ? 참일때값 : 거짓일때값
        //
        // 여기서는:
        // signs[i] 가 true 이면 1 반환
        // signs[i] 가 false 이면 -1 반환
        //
        // 즉,
        // 양수면 1을 곱하고
        // 음수면 -1을 곱해서
        // 실제 부호를 만들어주는 코드

        answer += absolutes[i] * (signs[i] ? 1 : -1);

        /*
            예시 1)
            absolutes[i] = 5
            signs[i] = true

            → (true ? 1 : -1)
            → 1

            → 5 * 1
            → 5

            answer += 5


            예시 2)
            absolutes[i] = 5
            signs[i] = false

            → (false ? 1 : -1)
            → -1

            → 5 * -1
            → -5

            answer += -5
        */
    }

    // 최종 합계 반환
    return answer;
}
```
