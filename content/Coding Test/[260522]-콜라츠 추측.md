풀이 코드

```
int solution(int num) 
{
    /*
        long long

        int보다 더 큰 숫자를 저장할 수 있는 자료형

        ------------------------------------------------

        왜 long long을 사용하는가?

        콜라츠 추측 계산 과정에서 숫자가 매우 커질 수 있기 때문

        예를 들어:

        n = 626331

        같은 숫자는 계산 도중 매우 큰 값으로 증가할 수 있음

        ------------------------------------------------

        int 범위:
        약 -21억 ~ 21억

        long long 범위:
        훨씬 더 큼

        ------------------------------------------------

        따라서 안전하게 계산하기 위해
        long long 사용
    */
    long long n = num;

    
    /*
        for 반복문

        형식:

        for(초기값; 조건; 증가)

        ------------------------------------------------

        int count = 0;
        -> count를 0으로 시작

        count <= 500;
        -> count가 500 이하인 동안 반복

        count++
        -> 반복 끝날 때마다 1 증가

        ------------------------------------------------

        count 역할:
        "몇 번 계산했는지 저장"

        ------------------------------------------------

        반복 흐름 예시:

        1번째 반복:
        count = 0

        2번째 반복:
        count = 1

        3번째 반복:
        count = 2

        ...
    */
    for (int count = 0; count <= 500; count++)
    {
        /*
            현재 숫자(n)가 1인지 확인

            ------------------------------------------------

            콜라츠 추측 규칙:
            숫자가 1이 되면 종료

            ------------------------------------------------

            예시:

            n = 1

            조건 참(true)

            return count 실행
        */
        if (n == 1)
        {
            /*
                현재까지 반복한 횟수 반환

                예시:

                count = 8

                -> 8 반환
            */
            return count;
        }

        
        /*
            짝수인지 확인

            % 연산자:
            나머지 구하기

            ------------------------------------------------

            예시:

            10 % 2 = 0
            -> 짝수

            11 % 2 = 1
            -> 홀수

            ------------------------------------------------

            따라서:

            n % 2 == 0

            의 의미는

            "n이 짝수인가?"
        */
        if (n % 2 == 0)
        {
            /*
                짝수일 경우 규칙:

                2로 나눈다

                ------------------------------------------------

                예시:

                n = 10

                n / 2

                결과:
                5

                ------------------------------------------------

                계산 후 다시 n에 저장
            */
            n = n / 2;
        }

        
        /*
            else

            위의 if 조건이 거짓일 경우 실행

            즉:
            홀수인 경우 실행됨
        */
        else
        {
            /*
                홀수일 경우 규칙:

                3을 곱하고 1 더하기

                ------------------------------------------------

                공식:

                n = n * 3 + 1

                ------------------------------------------------

                예시:

                n = 5

                5 * 3 + 1

                결과:
                16

                ------------------------------------------------

                계산 결과를 다시 n에 저장
            */
            n = n * 3 + 1;
        }
    }

    
    /*
        반복문이 끝났다는 의미

        즉:
        500번 반복했는데도
        n이 1이 되지 않음

        ------------------------------------------------

        문제 조건:
        500번 안에 1이 안 되면 -1 반환
    */
    return -1;
}
```

타 답안

```cpp
#include <iostream>
using namespace std;

// 콜라츠 추측 함수
// num이 1이 될 때까지 반복한 횟수를 반환
int collatz(int num)
{
    // 작업을 몇 번 수행했는지 저장하는 변수
    int answer = 0;

    // 시작 숫자 출력
    cout << num << "\n";

    // 최대 500번까지만 반복
    // answer++ :
    // 현재 answer 값을 먼저 비교한 뒤 1 증가시킴
    while (answer++ <= 500)
    {
        // 삼항 연산자 사용
        // num이 짝수이면 2로 나누고
        // 홀수이면 3을 곱한 뒤 1 더함
        num = num % 2 == 0 ? num / 2 : num * 3 + 1;

        // 계산된 현재 숫자 확인용 출력
        // cout << num << "\n";

        // num이 1이 되면 반복 종료
        if (num == 1)
            break;
    }

    // 반복 횟수가 500 초과면 -1 반환
    // 아니면 반복 횟수 반환
    return answer > 500 ? -1 : answer;
}

int main()
{
    // 테스트용 입력값
    int testCase = 6;

    // collatz 함수 실행 결과 저장
    int testAnswer = collatz(testCase);

    // 결과 출력
    cout << testAnswer;
}
```
