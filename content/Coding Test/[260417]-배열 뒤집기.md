풀이 코드

```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(vector<int> num_list) {

    int a = num_list.size();
    vector<int> answer(a);
    
    for(int i=0; i<a; i++){
        answer[i] = num_list[a-1-i];
    }
    return answer;
}
```

타 답안

```cpp
#include <string>
#include <vector>

using namespace std;

vector<int> solution(vector<int> num_list) {
    vector<int> answer;
    for(int i = num_list.size()-1; i >= 0; i--){
        answer.push_back(num_list[i]);
    }
    return answer;
}
```
