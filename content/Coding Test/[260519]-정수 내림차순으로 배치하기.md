풀이 코드

```cpp
#include <string>   // 문자열(string)을 사용하기 위한 헤더
#include <vector>   // vector 자료형을 사용하기 위한 헤더

using namespace std; // std:: 를 생략하고 사용하기 위해 작성

// long long 타입의 n을 받아서
// 각 자리수를 내림차순 정렬한 숫자를 반환하는 함수
long long solution(long long n)
{
    // 숫자들을 저장할 vector 생성
    // 예: 118372 -> 2,7,3,8,1,1 저장 예정
    vector<int> arr;

    // n이 0보다 클 동안 반복
    // 자리수를 하나씩 꺼내기 위한 반복문
    while (n > 0)
    {
        // n % 10
        // 숫자의 마지막 자리 꺼내기
        // 예:
        // 118372 % 10 -> 2
        arr.push_back(n % 10);

        // n /= 10
        // 마지막 자리 제거
        // 예:
        // 118372 -> 11837
        n /= 10;
    }

    // 숫자들을 큰 수부터 정렬하기 위한 반복문

    // i는 현재 비교할 위치
    for (int i = 0; i < arr.size(); i++)
    {
        // j는 i 다음 위치부터 끝까지 검사
        for (int j = i + 1; j < arr.size(); j++)
        {
            // 현재 숫자보다 뒤의 숫자가 더 크다면
            if (arr[i] < arr[j])
            {
                // 두 값을 바꾸기 위해 임시 변수 사용
                int temp = arr[i];

                // 큰 값을 앞으로 이동
                arr[i] = arr[j];

                // 원래 값을 뒤로 이동
                arr[j] = temp;
            }
        }
    }

    // 최종 숫자를 저장할 변수
    long long answer = 0;

    // 정렬된 숫자들을 다시 하나의 숫자로 만들기
    for (int i = 0; i < arr.size(); i++)
    {
        // 기존 숫자에 10을 곱해서 자릿수 한 칸 증가
        // 그리고 현재 숫자를 뒤에 붙임

        // 예:
        // answer = 87
        // arr[i] = 3

        // 87 * 10 + 3
        // = 873
        answer = answer * 10 + arr[i];
    }

    // 완성된 숫자 반환
    return answer;
}
```

타 답안

```cpp
#include <string>      // string 자료형 사용을 위한 헤더
#include <vector>      // vector 사용 헤더 (이 코드에서는 사실 안 써도 됨)
#include <algorithm>   // sort 함수를 사용하기 위한 헤더

using namespace std;   // std:: 생략 가능하게 함

// long long 타입 숫자 n을 받아
// 각 자리수를 내림차순 정렬한 숫자를 반환하는 함수
long long solution(long long n)
{
    // 숫자를 문자열로 변환
    // 예:
    // 118372 -> "118372"
    string str = to_string(n);

    // 문자열 정렬

    // str.begin()
    // 문자열의 시작 위치

    // str.end()
    // 문자열의 끝 위치

    // greater<char>()
    // 큰 문자부터 정렬(내림차순)

    // 결과:
    // "118372" -> "873211"
    sort(str.begin(), str.end(), greater<char>());

    // 문자열을 다시 long long 숫자로 변환
    // "873211" -> 873211
    return stoll(str);
}
```

| --- | --- | --- |
| **코드** | **읽는 법** | **의미** |

| sort | 소트 | 정렬 |
| begin | 비긴 | 시작 위치 |
| end | 엔드 | 끝 위치 |
| greater<char> | 그레이터 차 | 큰 값부터 정렬 |
| stoll | 에스티오엘엘 | string → long long 변환 |
