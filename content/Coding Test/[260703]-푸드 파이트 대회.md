**[ 풀이 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

string solution(vector<int> food) {
    string left = "";

    for (int i = 1; i < food.size(); i++) {

        int total = food[i];
        int useCount = total / 2;

        for (int j = 0; j < useCount; j++) {
            char foodChar = '0' + i;
            left += foodChar;
        }
    }

    string middle = "0";

    string right = left;
    reverse(right.begin(), right.end());

    string answer = left + middle + right;

    return answer;
}
```

**[ 해설 ]**
```cpp
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

string solution(vector<int> food) {
    // 왼쪽 선수(왼쪽에서 오른쪽으로 먹는 사람)의 음식 배열
    string left = "";

    // 0번은 물이기 때문에 1번 음식부터 시작
    for (int i = 1; i < food.size(); i++) {

        // 해당 음식의 전체 개수
        int total = food[i];

        // 양쪽 선수가 공평하게 나눠 먹어야 하므로 절반만 사용
        // 예: 7개면 3개씩만 사용 (1개는 버려짐)
        int useCount = total / 2;

        // useCount만큼 해당 음식 번호를 문자열에 추가
        for (int j = 0; j < useCount; j++) {

            // 숫자 i를 문자로 변환
            // '0'은 ASCII 기준 48이라서 숫자 + '0' 하면 문자로 변환됨
            char foodChar = '0' + i;

            // 왼쪽 문자열에 음식 추가
            left += foodChar;
        }
    }

    // 가운데에는 항상 물(0)이 들어감
    string middle = "0";

    // 오른쪽 선수는 왼쪽과 완전히 반대 방향으로 먹기 때문에
    // 왼쪽 문자열을 뒤집어서 오른쪽을 만든다
    string right = left;
    reverse(right.begin(), right.end());

    // 최종 구조:
    // 왼쪽 + 물 + 오른쪽
    string answer = left + middle + right;

    return answer;
}
```
**[ 타 답안 ]**
```cpp
#include <string>
#include <vector>

using namespace std;

string solution(vector<int> food) {
    
    // 처음 중앙에 물(0)을 넣고 시작
    // 최종 문자열 구조의 기준점이 됨
    string answer = "0";

    // food[i]를 절반씩 나눠서 양쪽에 배치해야 하므로 사용
    int check;

    // 가장 큰 음식 번호부터 1번까지 역순으로 처리
    // (0번은 물이라 제외)
    for(int i = food.size() - 1; i > 0; i--) {

        // 해당 음식의 절반만 사용 (양쪽 선수에게 나눔)
        check = food[i] / 2;

        // check만큼 반복해서 양쪽에 같은 숫자를 붙임
        while(check) {

            // 핵심 아이디어:
            // 현재 answer의 "양쪽"에 동시에 i를 붙임

            // 예:
            // answer = "0"
            // i = 2
            // → "2" + "0" + "2" = "202"

            answer = to_string(i) + answer + to_string(i);

            // 하나 사용했으므로 감소
            check--;
        }
    }

    return answer;
}
```
