풀이 코드

```cpp
#include <vector>      // vector 사용을 위한 헤더
#include <algorithm>   // sort 함수 사용을 위한 헤더

using namespace std;   // std::vector, std::sort 등을 간단히 쓰기 위해 사용


/*
    solution 함수

    매개변수:
    arr      -> 숫자들이 들어있는 배열
    divisor  -> 나누는 숫자

    반환값:
    조건에 맞는 숫자들을 담은 vector<int>
*/
vector<int> solution(vector<int> arr, int divisor)
{
    /*
        결과를 저장할 vector 생성

        vector<int>
        -> int형 숫자를 여러 개 저장할 수 있는 배열

        처음 상태:
        answer = {}

        아직 아무 값도 없음
    */
    vector<int> answer;


    /*
        range-based for문

        형식:
        for(자료형 변수명 : 배열)

        의미:
        배열 안의 값을 하나씩 자동으로 꺼내라

        ------------------------------------------------

        예시:

        arr = {5, 9, 7, 10}

        반복 진행 과정:

        1번 반복:
        num = 5

        2번 반복:
        num = 9

        3번 반복:
        num = 7

        4번 반복:
        num = 10

        ------------------------------------------------

        기존 방식:

        for(int i = 0; i < arr.size(); i++)

        는 index 번호(i)를 이용해서 접근

        ------------------------------------------------

        현재 방식:

        for(int num : arr)

        는 값 자체를 바로 가져옴

        초보자가 읽기 더 쉬운 편
    */
    for(int num : arr)
    {
        /*
            % 연산자
            -> 나머지를 구하는 연산자

            ------------------------------------------------

            예시:

            10 % 5 = 0
            -> 나누어 떨어짐

            11 % 5 = 1
            -> 나누어 떨어지지 않음

            ------------------------------------------------

            즉:

            num % divisor == 0

            의 뜻은

            "num이 divisor로 나누어 떨어지는가?"

            라는 의미
        */
        if(num % divisor == 0)
        {
            /*
                push_back()

                vector 맨 뒤에 값 추가

                ------------------------------------------------

                예시:

                answer = {}

                answer.push_back(5);

                결과:
                {5}

                ------------------------------------------------

                또 추가:

                answer.push_back(10);

                결과:
                {5, 10}

                ------------------------------------------------

                현재 반복문에서는
                나누어 떨어지는 숫자만 저장하고 있음
            */
            answer.push_back(num);
        }
    }


    /*
        empty()

        vector가 비어있는지 확인하는 함수

        ------------------------------------------------

        비어있으면:
        true

        값이 있으면:
        false

        ------------------------------------------------

        예시:

        {}

        -> 비어있음
        -> true

        ------------------------------------------------

        {5, 10}

        -> 값 존재
        -> false

        ------------------------------------------------

        즉:

        answer.empty()

        의 뜻은

        "저장된 숫자가 하나도 없는가?"
    */
    if(answer.empty())
    {
        /*
            조건에 맞는 숫자가 하나도 없을 경우

            문제 조건:
            -1을 담아서 반환해야 함

            ------------------------------------------------

            {-1}

            -> 숫자 -1 하나가 들어있는 vector 생성

            결과:

            {-1}
        */
        return {-1};
    }


    /*
        sort()

        배열을 정렬하는 함수

        ------------------------------------------------

        형식:

        sort(시작위치, 끝위치)

        ------------------------------------------------

        answer.begin()
        -> 배열 시작 위치

        answer.end()
        -> 배열 마지막 다음 위치

        ------------------------------------------------

        기본 정렬:
        오름차순(작은 수 -> 큰 수)

        ------------------------------------------------

        예시:

        {10, 5, 3}

        정렬 후:

        {3, 5, 10}
    */
    sort(answer.begin(), answer.end());


    /*
        최종 결과 반환

        예시 결과:

        {5, 10}

        또는

        {-1}
    */
    return answer;
}
```
