
제공해주신 공중 발판 이동 및 코인 수집 게임(`P3P4`) 베이스 코드를 바탕으로, 기획하신 **'P5' 프로젝트(멀티 웨이브 시스템, 3단계 레벨 전환, 3종 특수 디버프 아이템, HUD 및 UI 애니메이션 연동)** 조건에 맞춰 완벽하게 확장 및 재설계한 C++ 코드와 블루프린트 제작 가이드를 정리해 드립니다.

프로젝트 이름을 `P5_API` 및 `P5`로 매크로와 헤더 경로를 일괄 변경하여 바로 복사-붙여넣기 하실 수 있도록 정밀 교정했습니다.

# 1. 아키텍처 및 파일 구조 설계 (P5)

기존 폰(`AOkPawn`)의 자체 물리 기반 이동(LineTrace 방식)을 그대로 유지하면서, 요구 조건인 **디버프 실시간 중첩 및 해제, 컨트롤 반전, 시야 제한, 그리고 레벨 내 3단계 Wave 관리 시스템**을 탑재하기 위해 다음과 같이 구조를 확장합니다.

- **`UOKGameInstance`**: 전체 3개 레벨을 이동할 때 점수(`TotalScore`)와 레벨 인덱스를 전역적으로 기억.
    
- **`AOKGameState`**: 하나의 레벨 안에서 **Wave 1~3단계**를 타이머로 조절하며, 난이도(제한 시간, 아이템 스폰)를 제어하고 화면에 환경 변화 메시지를 전송.
    
- **`AOkPawn`**: 기존 이동/카메라 로직에 **컨트롤 반전, 마우스 회전 제한(Blind), 이동 속도 50% 감소 디버프** 및 **UI 표기용 중첩 카운트 스택** 장착. 머리 위 3D 위젯과 마우스 클릭 상호작용 지원.
    
- **`ABaseItem`** 및 **특수 디버프 아이템 3종**: 폰에 오버랩 시 고유 디버프 타이머를 실행하고 UI 전송.

// P5GameState.cpp
#include "P5GameState.h"

#include "Kismet/GameplayStatics.h"

#include "Blueprint/UserWidget.h"

AP5GameState::AP5GameState()
{
    CurrentWaveIndex = 0;
    MaxWaves = 3;
    MaxLevels = 3;
    WaveDuration = 30.0f;
}

void AP5GameState::BeginPlay()
{
    Super::BeginPlay();
    
    if (UP5GameInstance* GI = Cast<UP5GameInstance>(GetGameInstance()))
    {
        CurrentLevelIndex = GI->CurrentLevelIndex;
    }
    
    StartWave();
    GetWorldTimerManager().SetTimer(HUDUpdateTimerHandle, this, &AP5GameState::UpdateHUD, 0.1f, true);
}

void AP5GameState::StartWave()
{
    CurrentWaveIndex++;
    
    // 💡 [조건 구현] 웨이브 난이도 조정을 위해 웨이브별 시간 배리에이션 설정
    if (CurrentWaveIndex == 1) WaveDuration = 40.0f;
    else if (CurrentWaveIndex == 2) WaveDuration = 30.0f;
    else if (CurrentWaveIndex == 3) WaveDuration = 20.0f;

    // 💡 [조건 구현] GEngine 및 UI 연동용 웨이브 알림 / 맵 환경 변화 출력
    GEngine->AddOnScreenDebugMessage(-1, 5.0f, FColor::Green, FString::Printf(TEXT("Wave %d 시작!"), CurrentWaveIndex));

    if (CurrentWaveIndex == 2)
    {
        GEngine->AddOnScreenDebugMessage(-1, 5.0f, FColor::Orange, TEXT("이벤트: 맵에 새로운 장애물이 추가되었습니다!"));
    }
    else if (CurrentWaveIndex == 3)
    {
        GEngine->AddOnScreenDebugMessage(-1, 5.0f, FColor::Red, TEXT("이벤트: 파도 함정이 발동되었습니다...!"));
    }

    GetWorldTimerManager().SetTimer(WaveTimerHandle, this, &AP5GameState::OnWaveTimeUp, WaveDuration, false);
}

void AP5GameState::OnWaveTimeUp()
{
    if (CurrentWaveIndex < MaxWaves)
    {
        StartWave(); // 레벨 전환 없이 내부 다음 단계 웨이브 진행
    }
    else
    {
        EndLevel(); // 3웨이브 모두 종료 시 다음 레벨로 전환
    }
}

void AP5GameState::EndLevel()
{
    GetWorldTimerManager().ClearTimer(WaveTimerHandle);
    CurrentLevelIndex++;

    if (UP5GameInstance* GI = Cast<UP5GameInstance>(GetGameInstance()))
    {
        GI->CurrentLevelIndex = CurrentLevelIndex;
        
        if (CurrentLevelIndex >= MaxLevels)
        {
            OnGameOver();
            return;
        }
        
        if (LevelMapNames.IsValidIndex(CurrentLevelIndex))
        {
            UGameplayStatics::OpenLevel(GetWorld(), LevelMapNames[CurrentLevelIndex]);
        }
        else
        {
            OnGameOver();
        }
    }
}

void AP5GameState::OnGameOver()
{
    if (APlayerController* PC = GetWorld()->GetFirstPlayerController())
    {
        PC->SetPause(true);
        // 블루프린트 UI에서 후처리할 수 있도록 마우스 커서 및 UI모드 활성화 코드는 컨트롤러 UI 오픈부에서 처리합니다.

C++

```cpp
// OkPawn.cpp
#include "Pawn/OkPawn.h"
#include "Controller/OkPlayerController.h"
#include "Components/CapsuleComponent.h"
#include "Components/SkeletalMeshComponent.h"
#include "GameFramework/SpringArmComponent.h"
#include "Camera/CameraComponent.h"
#include "Components/WidgetComponent.h"
#include "EnhancedInputComponent.h"

AOkPawn::AOkPawn()
{
    PrimaryActorTick.bCanEverTick = true;
    
    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));
    SetRootComponent(CapsuleComp);
    CapsuleComp->InitCapsuleSize(34.0f, 88.0f);

    MeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("MeshComp"));
    MeshComp->SetupAttachment(RootComponent);

    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));
    SpringArmComp->SetupAttachment(RootComponent);
    SpringArmComp->TargetArmLength = 300.0f;

    CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComp"));
    CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);

    // 💡 3D 위젯 컴포넌트 초기화 및 스크린 스페이스 셋업
    OverheadWidget = CreateDefaultSubobject<UWidgetComponent>(TEXT("OverheadWidget"));
    OverheadWidget->SetupAttachment(RootComponent);
    OverheadWidget->SetWidgetSpace(EWidgetSpace::Screen);

    MaxHealth = 100.0f;
    Health = MaxHealth;

    bIsSlowed = false; bIsControlReversed = false; bIsBlind = false;
    SlowStack = 0; ReverseStack = 0; BlindStack = 0;
}

void AOkPawn::BeginPlay()
{
    Super::BeginPlay();
}

void AOkPawn::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);
    
    // 기존 LineTrace 기반 Grounded 및 중력 시스템 유지
    FHitResult HitResult;
    FVector StartLocation = GetActorLocation();
    float TraceLength = CapsuleComp->GetScaledCapsuleHalfHeight() + 5.0f;
    FVector EndLocation = StartLocation - FVector(0.0f, 0.0f, TraceLength);

    FCollisionQueryParams TraceParams;
    TraceParams.AddIgnoredActor(this);

    bool bHit = GetWorld()->LineTraceSingleByChannel(HitResult, StartLocation, EndLocation, ECC_Visibility, TraceParams);

    if (bHit)
    {
        bIsGrounded = true;
        if (VerticalVelocity < 0.0f) VerticalVelocity = 0.0f;
    }
    else
    {
        bIsGrounded = false;
        VerticalVelocity += Gravity * DeltaTime;
    }

    // 💡 [조건 구현] 슬로우 디버프 활성화 시 속도 50% 감속 멀티플라이어 적용
    float CurrentSpeed = MoveSpeed;
    if (bIsSlowed)
    {
        CurrentSpeed *= 0.5f;
    }
    if (!bIsGrounded)
    {
        CurrentSpeed *= 0.4f;
    }

    FVector InstantMove = MoveInputVector * CurrentSpeed * DeltaTime;
    AddActorLocalOffset(InstantMove, true);
    AddActorWorldOffset(FVector(0.0f, 0.0f, VerticalVelocity * DeltaTime), true);

    // 💡 [조건 구현] Blind 디버프 활성화 시 카메라 회전(Look 입력)을 완벽 차단 및 락 처리
    if (!bIsBlind)
    {
        FRotator PawnRotation = FRotator(0.0f, LookInputVector.X, LookInputVector.Z);
        AddActorLocalRotation(PawnRotation);

        FRotator NewArmRotation = SpringArmComp->GetRelativeRotation();
        NewArmRotation.Pitch = FMath::Clamp(NewArmRotation.Pitch + LookInputVector.Y, -60.0f, 60.0f); 
        SpringArmComp->SetRelativeRotation(NewArmRotation);
    }

    MoveInputVector = FVector::ZeroVector;
    LookInputVector = FVector::ZeroVector;
}

void AOkPawn::Move(const FInputActionValue& Value)
{
    FVector2D Input = Value.Get<FVector2D>();
    
    // 💡 [조건 구현] 컨트롤 반전 디버프 활성화 시 입력 방향 축을 전면 반전(-1.0f)
    if (bIsControlReversed)
    {
        Input.X *= -1.0f;
        Input.Y *= -1.0f;
    }

    MoveInputVector.X = Input.Y; 
    MoveInputVector.Y = Input.X;
}

// 디버프 적용 공통 시스템 (중첩 가능 및 타이머 실시간 갱신)
void AOkPawn::ApplySlow(float Duration)
{
    bIsSlowed = true;
    SlowStack++;
    GetWorldTimerManager().SetTimer(SlowTimerHandle, this, &AOkPawn::ClearSlow, Duration, false);
}
void AOkPawn::ClearSlow() { bIsSlowed = false; SlowStack = 0; }

void AOkPawn::ApplyReverse(float Duration)
{
    bIsControlReversed = true;
    ReverseStack++;
    GetWorldTimerManager().SetTimer(ReverseTimerHandle, this, &AOkPawn::ClearReverse, Duration, false);
}
void AOkPawn::ClearReverse() { bIsControlReversed = false; ReverseStack = 0; }

void AOkPawn::ApplyBlind(float Duration)
{
    bIsBlind = true;
    BlindStack++;
    GetWorldTimerManager().SetTimer(BlindTimerHandle, this, &AOkPawn::ClearBlind, Duration, false);
}
void AOkPawn::ClearBlind() { bIsBlind = false; BlindStack = 0; }

void AOkPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    Super::SetupPlayerInputComponent(PlayerInputComponent);
    if (UEnhancedInputComponent* EnhancedInput = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        if (AOkPlayerController* PC = Cast<AOkPlayerController>(GetController()))
        {
            if (PC->MoveAction) EnhancedInput->BindAction(PC->MoveAction, ETriggerEvent::Triggered, this, &AOkPawn::Move);
            if (PC->LookAction) EnhancedInput->BindAction(PC->LookAction, ETriggerEvent::Triggered, this, &AOkPawn::Look);
            if (PC->MoveUpAction) EnhancedInput->BindAction(PC->MoveUpAction, ETriggerEvent::Triggered, this, &AOkPawn::MoveUp);
            if (PC->MoveDownAction) EnhancedInput->BindAction(PC->MoveDownAction, ETriggerEvent::Triggered, this, &AOkPawn::MoveDown);
            if (PC->RollAction) EnhancedInput->BindAction(PC->RollAction, ETriggerEvent::Triggered, this, &AOkPawn::RollAxis);
        }
    }
}
void AOkPawn::MoveUp(const FInputActionValue& Value) { MoveInputVector.Z = 1.0f; }
void AOkPawn::MoveDown(const FInputActionValue& Value) { MoveInputVector.Z = -1.0f; }
void AOkPawn::Look(const FInputActionValue& Value) { FVector2D Input = Value.Get<FVector2D>(); LookInputVector.X = Input.X; LookInputVector.Y = -Input.Y; }
void AOkPawn::RollAxis(const FInputActionValue& Value) { LookInputVector.Z = Value.Get<float>() * 5.0f; }
```

## ④ 특수 디버프 아이템 클래스 (SlowingItem, ReverseItem, BlindItem 베이스)

기존 `RotatingActor`처럼 상공에 스폰되어 돌며, 플레이어에게 실시간 상태 이상 효과를 전송하는 아이템 예시 소스코드입니다.

C++

```cpp
// P5DebuffItems.h
#pragma once
#include "CoreMinimal.h"
#include "GameFramework/Actor.h"
#include "P5DebuffItems.generated.h"

UENUM(BlueprintType)
enum class EDebuffType : uint8
{
    Slow,
    Reverse,
    Blind
};

UCLASS()
class P5_API AP5DebuffItem : public AActor
{
    GENERATED_BODY()
public:
    AP5DebuffItem();

    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    class USphereComponent* CollisionComp;
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    UStaticMeshComponent* MeshComp;

    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Item Settings")
    EDebuffType ItemDebuffType;
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Item Settings")
    float DebuffDuration;

    UFUNCTION()
    void OnOverlap(UPrimitiveComponent* OverlappedComp, AActor* OtherActor, UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult);
};

// P5DebuffItems.cpp
#include "P5DebuffItems.h"
#include "Components/SphereComponent.h"
#include "Pawn/OkPawn.h"

AP5DebuffItem::AP5DebuffItem()
{
    PrimaryActorTick.bCanEverTick = false;
    CollisionComp = CreateDefaultSubobject<USphereComponent>(TEXT("CollisionComp"));
    SetRootComponent(CollisionComp);
    CollisionComp->SetCollisionProfileName(TEXT("OverlapAllDynamic"));

    MeshComp = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("MeshComp"));
    MeshComp->SetupAttachment(CollisionComp);

    DebuffDuration = 6.0f;
    ItemDebuffType = EDebuffType::Slow;
    CollisionComp->OnComponentBeginOverlap.AddDynamic(this, &AP5DebuffItem::OnOverlap);
}

void AP5DebuffItem::OnOverlap(UPrimitiveComponent* OverlappedComp, AActor* OtherActor, UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
{
    if (AOkPawn* PlayerPawn = Cast<AOkPawn>(OtherActor))
    {
        // 💡 [조건 구현] 디버프 종류에 맞춰 중첩 호출 처리 및 타이머 갱신
        switch (ItemDebuffType)
        {
        case EDebuffType::Slow:
            PlayerPawn->ApplySlow(DebuffDuration);
            break;
        case EDebuffType::Reverse:
            PlayerPawn->ApplyReverse(DebuffDuration);
            break;
        case EDebuffType::Blind:
            PlayerPawn->ApplyBlind(DebuffDuration);
            break;
        }
        Destroy(); // 💡 람다 댕글링 포인터 방지 및 안전 파괴
    }
}
```

### 📋 1. WBP_HUD 분석 및 배치 구조

- **구조 설계**: 루트 `Canvas Panel` 아래에 정보를 명확히 구획화하여 단일 화면에 올인원 배치합니다.
    
    - **좌측 상단 (`Horizontal Box`)**: 플레이어의 상태 레이아웃. 아이콘과 함께 `ProgressBar`를 사용하여 `Health / MaxHealth` 비율을 바인딩합니다.
        
    - **상단 중앙 (`Vertical Box`)**: `TextBlock` 두 개를 수직 배치하여 하나는 남은 시간 타이머(`Time: 00.0s`), 다른 하나는 레벨과 웨이브 정보(`Level 1 - Wave 2`)를 볼 수 있도록 가독성을 향상시킵니다.
        
    - **우측 상단 (`Vertical Box`)**: 디버프 전용 표시 패널. 폰(`AOkPawn`)의 `SlowStack`, `ReverseStack`, `BlindStack` 변수와 바인딩 링크를 걸어, 디버프가 활성화되거나 중첩될 경우 텍스트 및 경고 아이콘의 색상이 붉은색 계열로 활성화되도록 구성합니다.
        
- **디자인 완성도**: 글꼴 외곽선(Outline: 2)을 주어 발판 맵 배경과 겹치더라도 명확하게 노출되도록 디자인을 통일합니다.
    

### 📋 2. WBP_MainMenu 및 게임 오버 메뉴 UI 재설계

- **시각적 연출 및 블러 효과**: 전체 화면을 채우는 반투명 블랙 `Image` 컴포넌트를 생성하고 하단에 **`Background Blur`** 컴포넌트를 배치하여 `Blur Strength: 15.0`을 부여합니다. 이를 통해 게임 화면이 부드럽게 뒤로 아웃포커싱되는 고급스러운 연출이 완성됩니다.
    
- **버튼 디자인 인터랙션**:
    
    - `시작`, `종료`, `재시작`, `메인메뉴` 버튼들의 기본 색상(Tint)을 플랫한 다크 톤으로 맞추고, 사용자가 마우스를 올리는 **`Hovered` 상태일 때 채도가 높은 청록색(#00FFFF) 또는 네온 골드 컬러로 변환**되도록 세팅하여 클릭 욕구를 향상시킵니다.
        
- **Widget Animation 적용**: UI 컴포넌트 타임라인 창에서 `FadeInAnimation`을 추가합니다. 위젯이 생성될 때(Construct) `Play Animation` 노드를 연동하여 0.5초 동안 Canvas의 알파(Opacity) 값이 0에서 1로 차오르고 동시에 슬라이드 업되는 모션을 연출합니다.
    

### 📋 3. 머리 위 3D 체력바 및 월드 스페이스 상호작용

- `AOkPawn`에 장착된 `OverheadWidget` 컴포넌트의 클래스로 체력바 전용 미니 위젯인 `WBP_OverheadHP`를 할당합니다.
    
- **Screen Space 모드**로 작동하기 때문에 캐릭터가 공중 발판을 점프하며 회전하더라도 UI가 왜곡되지 않고 항상 플레이어 카메라 정면을 바라보게 되어 시인성을 지켜줍니다.
    
- 3D 아이템이나 수집형 보물상자 액터의 경우 컴포넌트 디테일 항목 내의 **`Receive Hardware Input`** 체크박스를 활성화하고 마우스 커서 쇼 비주얼 기능을 켜주면, 공중에서 3D 공간 상에 떠 있는 위젯 버튼을 플레이어가 직접 조준 사격하거나 클릭하여 상호작용하는 기능을 완벽히 구현해낼 수 있습니다.
