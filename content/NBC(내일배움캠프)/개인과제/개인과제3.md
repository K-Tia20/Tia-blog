*필수조건 상세*
### 서로 다른 Actor 클래스 2개 이상 구현
- 각 `StaticMeshComponent` 포함 : 맵에 배치 가능한 형태
- 각 클래스 : 서로 다른 동작 로직(회전용 / 이동용 등) 수행

### Tick 함수 기반 동적 Transform 변경
회전 기능 (Rotating Actor)
- `Tick(float DeltaTime)` 에서 `AddActorLocalRotation()`로 **매 프레임 회전** 적용
이동 기능 (Moving Platform Actor)
- `Tick(float DeltaTime)`에서 위치를 변경하여 **왕복 이동** 구현
- `StartLocation`(시작 위치) 저장/기준점으로 사용
- `MoveSpeed`(이동 속도) 적용
- `MaxRange`(왕복 범위) 기준으로 이동 제한
프레임 독립성
- 이동/회전 시 반드시 `DeltaTime` 을 활용
- 하드웨어 성능에 관계 없이 일정한 움직임을 보장해야합니다.
- 회전하는 톱니바퀴, 엘리베이터 등으로 자유롭게 확장 가능합니다.

### 리플렉션 적용
- 주요 변수 (회전 속도, 이동 속도, 이동 범위 등)를 `UPROPERTY`로 선언하여 **에디터에서 조정**
- `EditAnywhere`, `BlueprintReadWrite`, `Category` 등을 활용하여 **Details 패널**에서 편집 가능하게 만들기
- 플레이 중 Details 패널에서 값 변경 시 **즉시 반영**되는지 확인합니다.
- C++ 클래스의 자식 블루프린트를 만들어 시각적 효과 등의 추가 로직을 Blueprint에서 구현해도 좋습니다.

### 오브젝트 배치 및 테스트
- 레벨에 제작한 회전 발판과 이동 플랫폼을 여러 개 배치
- 각각 다른 속도, 이동범위, 회전값 적용
- 플레이 모드에서 오브젝트가 의도대로 움직이는지 확인
- 속도 변경이 실시간 반영되는지 확인

## 1. 회전 기능 액터 (ARotatingActor)

### RotatingActor.h

```cpp
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Actor.h"
#include "RotatingActor.generated.h"

UCLASS()
class SPARTAPROJECT_API ARotatingActor : public AActor
{
    GENERATED_BODY()
	
public:	
    ARotatingActor();

protected:
    virtual void BeginPlay() override;

public:	
    virtual void Tick(float DeltaTime) override;

protected:
    // 루트 컴포넌트
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    USceneComponent* SceneRoot;

    // 외형을 담당할 스태틱 메쉬
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    UStaticMeshComponent* StaticMeshComp;

    // 회전 속도 (초당 회전 각도, Details 패널에서 수정 가능)
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Properties|Rotation")
    FRotator RotationSpeed;
};

```

### RotatingActor.cpp
```cpp
#include "RotatingActor.h"

ARotatingActor::ARotatingActor()
{
    PrimaryActorTick.bCanEverTick = true;

    // 컴포넌트 생성 및 계층 구조 설정
    SceneRoot = CreateDefaultSubobject<USceneComponent>(TEXT("SceneRoot"));
    SetRootComponent(SceneRoot);

    StaticMeshComp = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("StaticMesh"));
    StaticMeshComp->SetupAttachment(SceneRoot);

    // 기본 회전 속도 설정 (Yaw 방향으로 초당 90도 회전)
    RotationSpeed = FRotator(0.0f, 90.0f, 0.0f);
}

void ARotatingActor::BeginPlay()
{
    Super::BeginPlay();
}

void ARotatingActor::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);

    // DeltaTime을 곱해 프레임 독립적인 회전 구현
    FRotator NewRotation = RotationSpeed * DeltaTime;
    AddActorLocalRotation(NewRotation);
}

```


## 2. 이동 기능 액터 (AMovingActor)

### MovingActor.h

```cpp

#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Actor.h"
#include "MovingActor.generated.h"

UCLASS()
class SPARTAPROJECT_API AMovingActor : public AActor
{
    GENERATED_BODY()
	
public:	
    AMovingActor();

protected:
    virtual void BeginPlay() override;

public:	
    virtual void Tick(float DeltaTime) override;

protected:
    // 루트 컴포넌트
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    USceneComponent* SceneRoot;

    // 외형을 담당할 스태틱 메쉬
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    UStaticMeshComponent* StaticMeshComp;

    // 이동 속도 (초당 이동 거리)
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Properties|Movement")
    float MoveSpeed;

    // 최대 왕복 범위 (시작 위치로부터 이동할 수 있는 최대 거리)
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Properties|Movement")
    float MaxRange;

    // 이동 방향 단위 벡터 (예: X축 이동 = 1, 0, 0 / Z축 엘리베이터 = 0, 0, 1)
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Properties|Movement")
    FVector MoveDirection;

private:
    // 시작 위치 보관용 변수 (기준점)
    FVector StartLocation;

    // 현재 이동 방향 부호 (1: 정방향, -1: 역방향)
    float DirectionSign;
};
```
### MovingActor.cpp

```cpp
#include "MovingActor.h"

AMovingActor::AMovingActor()
{
    PrimaryActorTick.bCanEverTick = true;

    // 컴포넌트 생성 및 계층 구조 설정
    SceneRoot = CreateDefaultSubobject<USceneComponent>(TEXT("SceneRoot"));
    SetRootComponent(SceneRoot);

    StaticMeshComp = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("StaticMesh"));
    StaticMeshComp->SetupAttachment(SceneRoot);

    // 변수 기본값 초기화
    MoveSpeed = 200.0f;          // 초당 200유닛 이동(이동속도)
    MaxRange = 500.0f;           // 500유닛만큼 왕복
    MoveDirection = FVector(1.0f, 0.0f, 0.0f); // 기본값: X축으로 이동
    DirectionSign = 1.0f;        // 처음에 정방향으로 출발
}

void AMovingActor::BeginPlay()
{
    Super::BeginPlay();

    // 게임이 시작될 때 현재 위치를 시작 위치(기준점)로 저장
    StartLocation = GetActorLocation();
}

void AMovingActor::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);

    // 1. 현재 위치 구하기
    FVector CurrentLocation = GetActorLocation();

    // 2. 이동할 크기 계산 (속도 * 프레임 시간 * 방향)
    // MoveDirection을 정규화(Normalize)하여 속도 계산의 일관성 유지
    FVector NormalDirection = MoveDirection.GetSafeNormal();
    FVector DeltaMovement = NormalDirection * MoveSpeed * DeltaTime * DirectionSign;

    // 3. 새 위치 적용
    FVector NewLocation = CurrentLocation + DeltaMovement;
    SetActorLocation(NewLocation);

    // 4. 시작 위치로부터 얼마나 멀어졌는지 거리 계산
    float DistanceMoved = FVector::Dist(StartLocation, NewLocation);

    // 5. 범위를 벗어났다면 방향을 반대로 전환
    if (DistanceMoved >= MaxRange)
    {
        // 완벽한 왕복을 위해 범위를 초과한 만큼 위치를 강제로 MaxRange 끝점에 고정
        FVector TargetEndLocation = StartLocation + (NormalDirection * MaxRange * DirectionSign);
        SetActorLocation(TargetEndLocation);

        // 방향 전환 (1 -> -1 또는 -1 -> 1)
        DirectionSign *= -1.0f;
    }
}

```

### 💡 과제 제출 및 에디터 테스트 팁

1. **빌드 후 에디터 확인**: 코드를 저장하고 언리얼 에디터에서 라이브 코딩을 켜거나(Ctrl+Alt+F11) 컴파일합니다.
    
2. **블루프린트 자식 생성 (권장)**: 과제 요구사항에 써있듯, 배치할 때는 이 C++ 클래스를 그대로 배치하기보다 콘텐츠 브라우저에서 우클릭하여 '자식 블루프린트 클래스 생성'을 한 뒤 배치하는 것이 좋습니다.
    
3. **스태틱 메쉬 설정**: 블루프린트 에디터를 열고 `StaticMeshComp`를 선택한 뒤, **Details 패널에서 원하는 모양(큐브, 톱니바퀴 등)의 메쉬를 등록**해야 화면에 보입니다.
    
4. **실시간 반영 테스트**: 게임을 플레이(PIE) 모드로 실행한 상태에서 배치한 액터를 선택하고, Details 패널의 `MoveSpeed`나 `RotationSpeed` 값을 바꾸면 오브젝트가 즉시 빨라지거나 느려지는 것을 확인할 수 있습니다.


혹시 로그 추가 하고 싶으면
```cpp
void AMovingActor::BeginPlay()
{
    Super::BeginPlay();
    // 게임 시작 시, 로그 창에 흰색(Log) 글씨로 출력됩니다.
    UE_LOG(LogTemp, Log, TEXT("Moving Actor가 스폰되었습니다!"));
}
```

```cpp
public:
```

```cpp
#include "MovingActor.h"

AMovingActor::AMovingActor()
{
    PrimaryActorTick.bCanEverTick = true;

    SceneRoot = CreateDefaultSubobject<USceneComponent>(TEXT("SceneRoot"));
    SetRootComponent(SceneRoot);

    StaticMeshComp = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("StaticMesh"));
    StaticMeshComp->SetupAttachment(SceneRoot);

    // 🚨 에디터에서 수정을 못 하므로 변수 기본값이 곧 최종 작동 값이 됩니다.
    MoveSpeed = 200.0f;          // 초당 200유닛 이동 고정
    MaxRange = 500.0f;           // 500유닛 범위 고정
    MoveDirection = FVector(1.0f, 0.0f, 0.0f); // X축 이동 고정
    DirectionSign = 1.0f;
}

void AMovingActor::BeginPlay()
{
    Super::BeginPlay();
    StartLocation = GetActorLocation();
}

void AMovingActor::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);

    // 왕복 이동 및 범위 제한 로직 자체는 리플렉션 유무와 상관없이 동일하게 작동합니다.
    FVector CurrentLocation = GetActorLocation();
    FVector NormalDirection = MoveDirection.GetSafeNormal();
    FVector DeltaMovement = NormalDirection * MoveSpeed * DeltaTime * DirectionSign;

    FVector NewLocation = CurrentLocation + DeltaMovement;
    SetActorLocation(NewLocation);

    float DistanceMoved = FVector::Dist(StartLocation, NewLocation);

    if (DistanceMoved >= MaxRange)
    {
        FVector TargetEndLocation = StartLocation + (NormalDirection * MaxRange * DirectionSign);
        SetActorLocation(TargetEndLocation);
        DirectionSign *= -1.0f;
    }
}
```

### RotatingActor.cpp
```cpp
#include "RotatingActor.h" // ARotatingActor 클래스의 헤더 파일 포함

// 생성자
ARotatingActor::ARotatingActor()
{
    // Tick 함수를 매 프레임 호출하도록 설정
    PrimaryActorTick.bCanEverTick = true;

    // 루트 컴포넌트 생성
    // 액터의 위치, 회전, 크기의 기준점 역할
    SceneRoot = CreateDefaultSubobject<USceneComponent>(TEXT("SceneRoot"));

    // 생성한 SceneRoot를 액터의 루트 컴포넌트로 지정
    SetRootComponent(SceneRoot);

    // 스태틱 메시 컴포넌트 생성
    // 액터의 외형을 화면에 표시하는 역할
    StaticMeshComp = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("StaticMesh"));

    // StaticMeshComp를 SceneRoot의 자식으로 부착
    // SceneRoot가 움직이거나 회전하면 함께 영향을 받음
    StaticMeshComp->SetupAttachment(SceneRoot);

    // 기본 회전 속도 설정
    // Pitch(X축) = 0도
    // Yaw(Z축) = 초당 90도
    // Roll(Y축) = 0도
    RotationSpeed = FRotator(0.0f, 90.0f, 0.0f);
}

// 게임 시작 시 한 번 호출
void ARotatingActor::BeginPlay()
{
    // 부모 클래스(AActor)의 BeginPlay 실행
    Super::BeginPlay();
}

// 매 프레임마다 호출
void ARotatingActor::Tick(float DeltaTime)
{
    // 부모 클래스(AActor)의 Tick 실행
    Super::Tick(DeltaTime);

    // 현재 프레임에서 회전해야 할 각도 계산
    // DeltaTime을 곱해 프레임 수가 달라도 같은 속도로 회전하도록 처리
    FRotator NewRotation = RotationSpeed * DeltaTime;

    // 액터의 로컬 좌표 기준으로 회전 적용
    AddActorLocalRotation(NewRotation);
}
```

```cpp
#include "MovingActor.h" // AMovingActor 클래스의 헤더 파일 포함

// 생성자
AMovingActor::AMovingActor()
{
    // Tick 함수를 매 프레임 호출하도록 설정
    PrimaryActorTick.bCanEverTick = true;

    // 루트 컴포넌트 생성
    // 액터의 위치, 회전, 크기의 기준이 되는 컴포넌트
    SceneRoot = CreateDefaultSubobject<USceneComponent>(TEXT("SceneRoot"));

    // 생성한 SceneRoot를 액터의 루트 컴포넌트로 지정
    SetRootComponent(SceneRoot);

    // 액터의 외형을 담당하는 스태틱 메시 컴포넌트 생성
    StaticMeshComp = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("StaticMesh"));

    // StaticMeshComp를 SceneRoot의 자식으로 연결
    StaticMeshComp->SetupAttachment(SceneRoot);

    // ==========================
    // 이동 관련 변수 기본값 설정
    // ==========================

    // 초당 200유닛 속도로 이동
    MoveSpeed = 200.0f;

    // 시작 위치 기준 최대 500유닛까지 이동
    MaxRange = 500.0f;

    // 기본 이동 방향을 X축 방향으로 설정
    MoveDirection = FVector(1.0f, 0.0f, 0.0f);

    // 처음에는 정방향(+) 이동
    // 1 = 정방향
    // -1 = 역방향
    DirectionSign = 1.0f;
}

// 게임 시작 시 한 번 호출
void AMovingActor::BeginPlay()
{
    // 부모 클래스의 BeginPlay 실행
    Super::BeginPlay();

    // 현재 위치를 시작 위치로 저장
    // 이후 왕복 이동 거리 계산의 기준점이 됨
    StartLocation = GetActorLocation();
}

// 매 프레임마다 호출
void AMovingActor::Tick(float DeltaTime)
{
    // 부모 클래스의 Tick 실행
    Super::Tick(DeltaTime);

    // ==========================
    // 현재 위치 가져오기
    // ==========================
    FVector CurrentLocation = GetActorLocation();

    // ==========================
    // 이동 방향 벡터 정규화
    // ==========================
    // (10,0,0) 과 (1,0,0)이 같은 방향으로 처리되도록
    // 길이를 1로 맞춤
    FVector NormalDirection = MoveDirection.GetSafeNormal();

    // ==========================
    // 이번 프레임 이동량 계산
    // ==========================
    // 이동방향
    // × 이동속도
    // × 프레임시간
    // × 현재 이동방향 상태(정방향/역방향)
    FVector DeltaMovement =
        NormalDirection *
        MoveSpeed *
        DeltaTime *
        DirectionSign;

    // ==========================
    // 새 위치 계산
    // ==========================
    FVector NewLocation =
        CurrentLocation +
        DeltaMovement;

    // 액터 위치 적용
    SetActorLocation(NewLocation);

    // ==========================
    // 시작 위치로부터 이동 거리 계산
    // ==========================
    float DistanceMoved =
        FVector::Dist(
            StartLocation,
            NewLocation
        );

    // ==========================
    // 최대 이동 범위 도달 여부 확인
    // ==========================
    if (DistanceMoved >= MaxRange)
    {
        // 최대 범위를 초과하지 않도록
        // 정확한 끝 지점으로 위치 보정

        FVector TargetEndLocation =
            StartLocation +
            (
                NormalDirection *
                MaxRange *
                DirectionSign
            );

        // 보정 위치 적용
        SetActorLocation(TargetEndLocation);

        // ==========================
        // 이동 방향 반전
        // ==========================
        //  1 → -1
        // -1 →  1
        DirectionSign *= -1.0f;
    }
}
```
