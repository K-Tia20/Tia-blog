// MyPawn.cpp
#include "MyPawn.h"

// 컴포넌트 헤더
#include "Components/CapsuleComponent.h"
#include "Components/SkeletalMeshComponent.h"
#include "GameFramework/SpringArmComponent.h"
#include "Camera/CameraComponent.h"

// Enhanced Input 헤더
#include "EnhancedInputComponent.h"
#include "EnhancedInputSubsystems.h"
#include "InputActionValue.h"

// ─────────────────────────────────────────────────────────────────────────────
//  생성자
// ─────────────────────────────────────────────────────────────────────────────
AMyPawn::AMyPawn()
{
    PrimaryActorTick.bCanEverTick = true;

    // ── 1) CapsuleComponent → RootComponent ──────────────────────────────────
    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));
    SetRootComponent(CapsuleComp);

    // 물리 시뮬레이션 OFF — 코드로 직접 제어
    CapsuleComp->SetSimulatePhysics(false);

    // 콜리전 프로파일: 필요에 따라 변경 가능
    CapsuleComp->SetCollisionProfileName(TEXT("Pawn"));

    // 캡슐 기본 크기 (Height 절반, Radius)
    //CapsuleComp->InitCapsuleSize(42.f, 96.f);
	//여기위에 ㅈ

    // ── 2) SkeletalMeshComponent → Capsule에 부착 ────────────────────────────
    SkeletalMeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("SkeletalMeshComp"));
    SkeletalMeshComp->SetupAttachment(CapsuleComp);

    // 물리 시뮬레이션 OFF
    SkeletalMeshComp->SetSimulatePhysics(false);

    // 메시를 캡슐 하단에 맞게 오프셋 (에디터에서 메시 교체 후 조정 권장)
    //SkeletalMeshComp->SetRelativeLocationAndRotation(
    //    FVector(0.f, 0.f, -96.f),          // Z 오프셋
    //   FRotator(0.f, -90.f, 0.f)           // 메시 방향 보정
    //);

    // ── 3) SpringArmComponent → Capsule에 부착 ───────────────────────────────
    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));
    SpringArmComp->SetupAttachment(CapsuleComp);
    SpringArmComp->TargetArmLength = 400.f;             // 붐 길이
    SpringArmComp->SetRelativeLocation(FVector(0.f, 0.f, 60.f)); // 시점 높이

    // Pawn의 Yaw 회전을 SpringArm이 따라가지 않도록 설정
    // (Pawn 자체를 회전시키므로 상속만으로 충분)
    SpringArmComp->bUsePawnControlRotation = false;
    //SpringArmComp->bInheritPitch           = true;
    //SpringArmComp->bInheritYaw             = true;
    //SpringArmComp->bInheritRoll            = false;

    // ── 4) CameraComponent → SpringArm 끝에 부착 ─────────────────────────────
    CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComp"));
    CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);
    CameraComp->bUsePawnControlRotation = false;        // 카메라 자체는 회전 안 함

    // ── 5) 엔진 기본 컨트롤러 회전 연동 OFF ──────────────────────────────────
    //       (직접 AddActorLocalRotation으로 처리하므로 모두 끔)
    //bUseControllerRotationPitch = false;
    //bUseControllerRotationYaw   = false;
    //bUseControllerRotationRoll  = false;
}

// ─────────────────────────────────────────────────────────────────────────────
//  BeginPlay — Enhanced Input IMC 등록
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::BeginPlay()
{
    Super::BeginPlay();

    // LocalPlayer의 EnhancedInputSubsystem에 IMC를 우선순위 0으로 추가
    if (APlayerController* PC = Cast<APlayerController>(GetController()))
    {
        if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
            ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PC->GetLocalPlayer()))
        {
            Subsystem->AddMappingContext(DefaultMappingContext, 0);
        }
    }
}

// ─────────────────────────────────────────────────────────────────────────────
//  Tick
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);
    // 이동·회전 로직은 입력 콜백(HandleMove / HandleLook)에서 처리합니다.
    // Tick에서 추가 처리가 필요하다면 여기에 작성하세요.
}

// ─────────────────────────────────────────────────────────────────────────────
//  SetupPlayerInputComponent — Enhanced Input 바인딩
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    Super::SetupPlayerInputComponent(PlayerInputComponent);

    // EnhancedInputComponent로 캐스트
    if (UEnhancedInputComponent* EIC = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        // IA_Move — 키를 누르고 있는 동안(Triggered) 매 프레임 호출
        EIC->BindAction(MoveAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleMove);

        // IA_Look — 마우스가 움직이는 동안(Triggered) 매 프레임 호출
        EIC->BindAction(LookAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleLook);
    }
}

// ─────────────────────────────────────────────────────────────────────────────
//  HandleMove — WASD 이동
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::HandleMove(const FInputActionValue& Value)
{
    // IA_Move 는 Vector2D 타입
    //   X : 좌우 (A=-1, D=+1)
    //   Y : 전후 (S=-1, W=+1)
    const FVector2D MovementInput = Value.Get<FVector2D>();

    if (MovementInput.IsNearlyZero())
    {
        return;
    }

    // DeltaTime은 Tick이 아닌 콜백에서는 직접 구할 수 없으므로
    // GetWorld()->GetDeltaSeconds() 로 가져옵니다.
    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // Pawn의 Forward / Right 벡터 기준으로 로컬 오프셋 계산
    const FVector ForwardDelta = GetActorForwardVector() * MovementInput.Y * MoveSpeed * DeltaTime;
    const FVector RightDelta   = GetActorRightVector()   * MovementInput.X * MoveSpeed * DeltaTime;

    // bSweep = true → 충돌 감지를 활성화한 채로 이동
    AddActorWorldOffset(ForwardDelta + RightDelta, /*bSweep=*/true);
}

// ─────────────────────────────────────────────────────────────────────────────
//  HandleLook — 마우스 회전 (Yaw + Pitch)
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::HandleLook(const FInputActionValue& Value)
{
    // IA_Look 는 Vector2D 타입
    //   X : 마우스 수평 이동량 → Yaw
    //   Y : 마우스 수직 이동량 → Pitch
    const FVector2D LookInput = Value.Get<FVector2D>();

    if (LookInput.IsNearlyZero())
    {
        return;
    }

    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // ── Yaw (좌우 회전) ─────────────────────────────────────────────────────
    //    Pawn 자체를 Z축 기준으로 회전시킵니다.
    {
        const float YawDelta = LookInput.X * LookSensitivity.X;   // DeltaTime 불필요 (마우스 델타값 자체가 프레임 독립적)
        AddActorLocalRotation(FRotator(0.f, YawDelta, 0.f));
    }

    // ── Pitch (상하 회전) ────────────────────────────────────────────────────
    //    SpringArm을 Y축 기준으로 회전시켜 카메라 고개 숙임/들기를 구현합니다.
    //    Pawn 전체를 Pitch 회전시키면 이동 방향도 틀어지므로 SpringArm만 회전시킵니다.
    {
        // 마우스 Y 입력 반전 여부: 일반적으로 Y 양수 = 위 → Pitch 감소
        const float PitchDelta = -LookInput.Y * LookSensitivity.Y;

        // 누적값 계산 후 클램프
        CurrentPitch = FMath::Clamp(CurrentPitch + PitchDelta, PitchMin, PitchMax);

        // SpringArm의 절대 Pitch를 설정 (누적값으로 덮어씀)
        FRotator ArmRotation = SpringArmComp->GetRelativeRotation();
        ArmRotation.Pitch    = CurrentPitch;
        SpringArmComp->SetRelativeRotation(ArmRotation);
    }
}
