*필수조건 체크 리스트*

- [ ] Pawn 클래스 생성: Pawn을 상속받는 새로운 C++ 클래스를 생성합니다.
- [ ] 컴포넌트 추가: 아래 컴포넌트들을 Pawn 클래스에 추가합니다.
- [ ] CapsuleComponent(또는 Box/Sphere 중 택 1)
- [ ] SkeletalMeshComponent
- [ ] SpringArmComponent
- [ ] CameraComponent
- [ ] 계층 구조 설정: 충돌 컴포넌트를 RootComponent로 설정하고, 나머지 컴포넌트들을 부착합니다.
- [ ] defaultPawn 설정: GameMode 클래스에서 DefaultPawnClass 를 본인이 만든 Pawn으로 지정합니다.
- [ ] Physics 설정: 루트 충돌체 및 Mesh의 `Simulate Physics`를 **false**로 설정합니다. (물리 대신 코드로 직접 제어)
- [ ] 입력 액션(IA) 생성: 아래 두 가지 액션을 생성합니다. (타입: Vector2D)
- [ ] IA_Move(WASD 이동용)
- [ ] IA_Look(마우스 회전용)
- [ ] IMC 매핑: 생성한 액션들을 Input Mapping Context에 등록하고 키를 할당합니다.
- [ ] 액션 바인딩: SetupPlayerInputComponent()에서 입력 처리 함수와 액션들을 바인딩합니다.
- [ ] 프레임 독립성: DeltaTime을 사용하여 프레임 속도와 관계없이 일정한 속도로 움직이도록 구현합니다.
- [ ] 이동 구현: AddActorLocalOffset()등을 활용해 WASD 입력에 따라 Pawn이 움직이도록 작성합니다.
- [ ] 이동 방향은 Pawn의 Forward/Right 벡터를 기준으로 결정됩니다.
- [ ] 회전 구현: AddActorLocalRotation()등을 활용해 마우스 입력에 따라 회전하도록 작성합니다.
- [ ] 마우스 입력값으로 Yaw와 Pitch를 직접 계산하여 구현합니다.
- [ ] 주의: AddControllerYawInput(),AddControllerPitchInput()등 엔진 기본 제공 함수는 사용하지 않습니다.
- [ ] 제한 사항: 평면 상에서의 이동과 회전만 처리하며, 중력이나 낙하 효과는 고려하지 않습니다.

#include "BasePawn.h"  
  
#include "Camera/CameraComponent.h"  
#include "Components/CapsuleComponent.h"

ABasePawn::ABasePawn()  
{  
    PrimaryActorTick.bCanEverTick = true;  
  
    // Capsule  
    CapsuleComp =  
       CreateDefaultSubobject<UCapsuleComponent>(TEXT("Capsule"));  
  
    RootComponent = CapsuleComp;  
  
    // Mesh  
    MeshComp =  
       CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("Mesh"));  
  
    MeshComp->SetupAttachment(RootComponent);  
  
    // SpringArm  
    SpringArmComp =  
       CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArm"));  
  
    SpringArmComp->SetupAttachment(RootComponent);  
    SpringArmComp->TargetArmLength = 300.f;  
  
    // Camera  
    CameraComp =  
       CreateDefaultSubobject<UCameraComponent>(TEXT("Camera"));  
  
    CameraComp->SetupAttachment(SpringArmComp);  
  
    // Physics Off  
    CapsuleComp->SetSimulatePhysics(false);  
    MeshComp->SetSimulatePhysics(false);  
}  
  
void ABasePawn::BeginPlay()  
{  
    Super::BeginPlay();  
  
    if (APlayerController* PC = Cast<APlayerController>(GetController()))  
    {       if (ULocalPlayer* LocalPlayer = PC->GetLocalPlayer())  
       {          if (UEnhancedInputLocalPlayerSubsystem* Subsystem =  
             LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())  
          {             if (IMC_Default)  
             {                Subsystem->AddMappingContext(IMC_Default, 0);  
             }          }       }    }}  
  
void ABasePawn::Tick(float DeltaTime)  
{  
    Super::Tick(DeltaTime);  
  
    // Move  
    FVector MoveDelta =  
       (GetActorForwardVector() * MoveInput.X +  
        GetActorRightVector() * MoveInput.Y)  
       * MoveSpeed  
       * DeltaTime;  
  
    //기존코드  
    //AddActorLocalOffset(MoveDelta, true);  
        //수정코드  
    AddActorWorldOffset(MoveDelta, true);  
    //기존 코드   
/*  
    FRotator RotationDelta;  
    RotationDelta.Pitch = LookInput.Y * RotationSpeed * DeltaTime;    RotationDelta.Yaw = LookInput.X * RotationSpeed * DeltaTime;  
    AddActorLocalRotation(RotationDelta);    */        //신규 코드  
    // Yaw - Pawn 전체 회전  
    FRotator YawDelta(0.f, LookInput.X * RotationSpeed * DeltaTime, 0.f);  
    AddActorLocalRotation(YawDelta);  
  
    // Pitch - SpringArm만 회전  
    FRotator PitchDelta(LookInput.Y * RotationSpeed * DeltaTime, 0.f, 0.f);  
    FRotator NewSpringArmRot = SpringArmComp->GetRelativeRotation() + PitchDelta;  
    NewSpringArmRot.Pitch = FMath::Clamp(NewSpringArmRot.Pitch, -60.f, 60.f);  
    SpringArmComp->SetRelativeRotation(NewSpringArmRot);  
    UE_LOG(LogTemp, Warning, TEXT("Forward: %s"), *GetActorForwardVector().ToString());  
    //신규 코드  
    FVector VerticalDelta = FVector(0.f, 0.f, VerticalInput * MoveSpeed * DeltaTime);  
    AddActorWorldOffset(VerticalDelta, true);  
  
    }  
  
void ABasePawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)  
{  
    Super::SetupPlayerInputComponent(PlayerInputComponent);  
  
    if (UEnhancedInputComponent* EIC =  
       Cast<UEnhancedInputComponent>(PlayerInputComponent))  
    {       if (IA_Move)  
       {          EIC->BindAction(  
             IA_Move,  
             ETriggerEvent::Triggered,  
             this,  
             &ABasePawn::Move);  
                    EIC->BindAction(  
             IA_Move,  
             ETriggerEvent::Completed,  
             this,  
             &ABasePawn::StopMove);  
       }  
       if (IA_Look)  
       {          EIC->BindAction(  
             IA_Look,  
             ETriggerEvent::Triggered,  
             this,  
             &ABasePawn::Look);  
                    EIC->BindAction(  
             IA_Look,  
             ETriggerEvent::Completed,  
             this,  
             &ABasePawn::StopLook);  
       }       if (IA_UpDown)  
       {          EIC->BindAction(  
             IA_UpDown,  
             ETriggerEvent::Triggered,  
             this,  
             &ABasePawn::Vertical);  
          EIC->BindAction(  
             IA_UpDown,  
             ETriggerEvent::Completed,  
             this,  
             &ABasePawn::StopVertical);  
  
       }    }  
}

#include "BasePawn.h"   // ABasePawn 클래스의 헤더 포함

#include "Camera/CameraComponent.h"                 // 카메라 컴포넌트

#include "EnhancedInputSubsystems.h"                // Enhanced Input 서브시스템

#include "InputAction.h"                            // Input Action
#include "InputMappingContext.h"                    // Input Mapping Context

#include "Engine/LocalPlayer.h"                     // LocalPlayer 클래스

ABasePawn::ABasePawn()
{
    // Tick 함수 사용 가능
    PrimaryActorTick.bCanEverTick = true;

    // ==========================
    // Capsule 생성
    // ==========================

    CapsuleComp =
        CreateDefaultSubobject<UCapsuleComponent>(TEXT("Capsule"));
    // "Capsule"이라는 이름의 캡슐 컴포넌트 생성

    RootComponent = CapsuleComp;
    // Pawn의 루트 컴포넌트로 설정

    // ==========================
    // Mesh 생성
    // ==========================

    MeshComp =
        CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("Mesh"));
    // 캐릭터 외형이 될 스켈레탈 메시 생성

    MeshComp->SetupAttachment(RootComponent);
    // Capsule 밑에 붙임

    // ==========================
    // SpringArm 생성
    // ==========================

    SpringArmComp =
        CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArm"));
    // 카메라 거리 유지를 위한 스프링암 생성

    SpringArmComp->SetupAttachment(RootComponent);
    // Capsule에 부착

    SpringArmComp->TargetArmLength = 300.f;
    // 카메라와 Pawn 거리 300

    // ==========================
    // Camera 생성
    // ==========================

    CameraComp =
        CreateDefaultSubobject<UCameraComponent>(TEXT("Camera"));
    // 실제 화면을 보여줄 카메라 생성

    CameraComp->SetupAttachment(SpringArmComp);
    // 스프링암 끝에 부착

    // ==========================
    // Physics 비활성화
    // ==========================

    CapsuleComp->SetSimulatePhysics(false);
    // 캡슐 물리 사용 안 함

    MeshComp->SetSimulatePhysics(false);
    // 메시 물리 사용 안 함
}

void ABasePawn::BeginPlay()
{
    Super::BeginPlay();
    // 부모 BeginPlay 먼저 실행

    if (APlayerController* PC = Cast<APlayerController>(GetController()))
    // 현재 Pawn을 조종하는 Controller 가져오기
    {
        if (ULocalPlayer* LocalPlayer = PC->GetLocalPlayer())
        // LocalPlayer 가져오기
        {
            if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
                LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
            // Enhanced Input Subsystem 가져오기
            {
                if (IMC_Default)
                // IMC_Default가 비어있지 않으면
                {
                    Subsystem->AddMappingContext(IMC_Default, 0);
                    // 입력 맵 등록
                }
            }
        }
    }
}

    FVector MoveDelta =
        (
            GetActorForwardVector() * MoveInput.X +
            GetActorRightVector() * MoveInput.Y
        )
        * MoveSpeed
        * DeltaTime;

    // GetActorForwardVector()
    // Pawn이 바라보는 앞 방향 벡터

    // MoveInput.X
    // W,S 입력값

    // GetActorRightVector()
    // Pawn의 오른쪽 방향 벡터

    // MoveInput.Y
    // A,D 입력값

    // MoveSpeed
    // 이동 속도 적용

    // DeltaTime
    // 프레임 독립적인 이동

    AddActorWorldOffset(MoveDelta, true);
    // 월드 기준으로 이동
    // true = 충돌 검사

    // ==========================
    // 좌우 회전(Yaw)
    // ==========================

    FRotator YawDelta(
        0.f,
        LookInput.X * RotationSpeed * DeltaTime,
        0.f
    );

    // Pitch = 0
    // Yaw = 마우스 좌우
    // Roll = 0

    AddActorLocalRotation(YawDelta);
    // Pawn 자체 회전

    // ==========================
    // 상하 회전(Pitch)
    // ==========================

    FRotator PitchDelta(
        LookInput.Y * RotationSpeed * DeltaTime,
        0.f,
        0.f
    );

    // Pitch만 변경

    FRotator NewSpringArmRot =
        SpringArmComp->GetRelativeRotation() + PitchDelta;

    // 현재 스프링암 회전에 Pitch 추가

    NewSpringArmRot.Pitch =
        FMath::Clamp(
            NewSpringArmRot.Pitch,
            -60.f,
            60.f
        );

    // -60 ~ 60도 제한

    SpringArmComp->SetRelativeRotation(NewSpringArmRot);
    // 스프링암 회전 적용

    UE_LOG(
        LogTemp,
        Warning,
        TEXT("Forward: %s"),
        *GetActorForwardVector().ToString()
    );

    // 현재 Forward Vector 출력
    // 디버깅용

    // ==========================
    // 위아래 이동
    // ==========================

    FVector VerticalDelta(
        0.f,
        0.f,
        VerticalInput * MoveSpeed * DeltaTime
    );

    // X = 0
    // Y = 0
    // Z만 이동

    AddActorWorldOffset(VerticalDelta, true);
    // 월드 기준 위/아래 이동
}

// 1. 평면 이동 (WASD 입력 처리)
void ABasePawn::Move(const FInputActionValue& Value)
{
    if (!Controller) return;
    const FVector2D MoveInput = Value.Get<FVector2D>();

    // 마우스가 쳐다보는 방향(Yaw)을 기준으로 십자 방향 계산
    const FRotator Rotation = Controller->GetControlRotation();
    const FRotator YawRotation(0, Rotation.Yaw, 0);

    // 앞 / 뒤
    if (!FMath::IsNearlyZero(MoveInput.X))
    {
        const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
        AddMovementInput(ForwardDirection, MoveInput.X);
    }

    // 좌 / 우
    if (!FMath::IsNearlyZero(MoveInput.Y))
    {
        const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
        AddMovementInput(RightDirection, MoveInput.Y);
    }
}

void ABasePawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    Super::SetupPlayerInputComponent(PlayerInputComponent);

    if (UEnhancedInputComponent* EnhancedInput = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        if (ASpartaPlayerController* PlayerController = Cast<ASpartaPlayerController>(GetController()))
        {
            // 1. WASD 평면 이동 바인딩
            if (PlayerController->MoveAction)
            {
                EnhancedInput->BindAction(PlayerController->MoveAction, ETriggerEvent::Triggered, this, &ABasePawn::Move);
            }
            
            // 2. 마우스 시선 전환 바인딩
            if (PlayerController->LookAction)
            {
                EnhancedInput->BindAction(PlayerController->LookAction, ETriggerEvent::Triggered, this, &ABasePawn::Look);
            }
            
            // 3. 위/아래 수직 이동 바인딩 (Q/E)
            if (PlayerController->UpMoveAction)
            {
                EnhancedInput->BindAction(PlayerController->UpMoveAction, ETriggerEvent::Triggered, this, &ABasePawn::FlyUpDown);
            }
        }
    }
}

// 1. 평면 이동 (WASD 입력 즉시 처리)
void ABasePawn::Move(const FInputActionValue& Value)
{
    if (!Controller) return;
    const FVector2D MoveInput = Value.Get<FVector2D>();

    const FRotator Rotation = Controller->GetControlRotation();
    const FRotator YawRotation(0, Rotation.Yaw, 0);

    // 컴퓨터 사양(프레임)에 상관없이 일정한 속도를 유지하기 위한 DeltaTime 확보
    float DeltaTime = GetWorld()->GetDeltaSeconds();

    // 앞 / 뒤 이동
    if (!FMath::IsNearlyZero(MoveInput.X))
    {
        const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
        // 위치를 즉시 변경 (600.f = 이동 속도)
        SetActorLocation(GetActorLocation() + (ForwardDirection * MoveInput.X * 600.f * DeltaTime), true);
    }

    // 좌 / 우 이동
    if (!FMath::IsNearlyZero(MoveInput.Y))
    {
        const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
        SetActorLocation(GetActorLocation() + (RightDirection * MoveInput.Y * 600.f * DeltaTime), true);
    }
}

// 2. 시선 전환 (마우스 회전 처리)
void ABasePawn::Look(const FInputActionValue& Value)
{
    const FVector2D LookInput = Value.Get<FVector2D>();

    AddControllerYawInput(LookInput.X);   // 마우스 좌우 -> 폰 회전
    AddControllerPitchInput(LookInput.Y); // 마우스 상하 -> 카메라 시선 회전
}

// 3. 위/아래 수직 비행 (IA_UpMove 입력 즉시 처리)
void ABasePawn::FlyUpDown(const FInputActionValue& Value)
{
    const float VerticalValue = Value.Get<float>();
    float DeltaTime = GetWorld()->GetDeltaSeconds();

    if (!FMath::IsNearlyZero(VerticalValue))
    {
        // 월드 위쪽(Z축) 방향으로 즉시 변경 (600.f = 이동 속도)
        SetActorLocation(GetActorLocation() + (FVector::UpVector * VerticalValue * 600.f * DeltaTime), true);
    }
}

void ABasePawn::BeginPlay()
{
	Super::BeginPlay();

	if (APlayerController* PC = Cast<APlayerController>(GetController()))
	{
		if (ULocalPlayer* LocalPlayer = PC->GetLocalPlayer())
		{
			if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
				LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
			{
				Subsystem->AddMappingContext(InputMappingContext, 0);
			}
		}
	}
}

void ABasePawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);

	if (UEnhancedInputComponent* EnhancedInput =
		Cast<UEnhancedInputComponent>(PlayerInputComponent))
	{
		if (MoveAction)
		{
			EnhancedInput->BindAction(
				MoveAction,
				ETriggerEvent::Triggered,
				this,
				&ABasePawn::Move);
		}

		if (LookAction)
		{
			EnhancedInput->BindAction(
				LookAction,
				ETriggerEvent::Triggered,
				this,
				&ABasePawn::Look);
		}
	}
}

void ABasePawn::Move(const FInputActionValue& Value)
{
	MoveInput = Value.Get<FVector2D>();
}

void ABasePawn::Look(const FInputActionValue& Value)

	LookInput = Value.Get<FVector2D>();

void ABasePawn::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

	FVector MoveDirection =
		GetActorForwardVector() * MoveInput.Y +
		GetActorRightVector() * MoveInput.X;

	AddActorLocalOffset(
		MoveDirection * MoveSpeed * DeltaTime,
		true);

	FRotator DeltaRotation(
		0.f,
		LookInput.X * RotationSpeed * DeltaTime,
		0.f);

	AddActorLocalRotation(DeltaRotation);
}
