```cpp
UCLASS()
class STAGE01_API ATest1Actor : public AActor
{
	GENERATED_BODY()
	
public:	
	// Sets default values for this actor's properties
	ATest1Actor();

protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;

public:	
	// Called every frame
	virtual void Tick(float DeltaTime) override;
	
public:

	UPROPERTY(VisiableAnywhere, BluprintReadOnly, Category = "Test1Actor|Components")
	UScenceComponent* RootGOGOGO;
	
	UPROPERTY(VisiableAnywhere, BluprintReadOnly, Category = "Test1Actor|Components")
	UStaticMeshComponent* MeshOk;
	
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Test1Actor|TestRotattion" )
	FRotator Rotatatatat;
	
};


```

### 회전 연습.cpp
```cpp
// Fill out your copyright notice in the Description page of Project Settings.


#include "Test1Actor.h"

// Sets default values
ATest1Actor::ATest1Actor()
{
 	// Set this actor to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;

	RootGOGOGO = CreateDefaultSubobject<UScenceComponent>(TEXT("Ok1"))
	
	SetRootComponent(RootGOGOGO);

	MeshOk = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("Ok2"))

	MeshOk->SetupAttachment(RootGOGOGO);
	
	Rotatatatat = FRotator(0.0f, 90.0f, 0.0f);
	
}

// Called when the game starts or when spawned
void ATest1Actor::BeginPlay()
{

	Super::BeginPlay();
	
}

// Called every frame
void ATest1Actor::Tick(float DeltaTime)
{

	Super::Tick(DeltaTime);
	
	FRotator SSSD = Rotatatatat * DeltaTime;
	
	AddActorLocalRotation(SSSD);

}

```
