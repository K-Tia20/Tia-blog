```cpp
// MyPawn.h
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Pawn.h"
#include "MyPawn.generated.h"

// 전방 선언 (Forward Declarations)
class UCapsuleComponent;
class USkeletalMeshComponent;
class USpringArmComponent;
class UCameraComponent;
class UInputMappingContext;
class UInputAction;
struct FInputActionValue;

UCLASS()
class MYPROJECT_API AMyPawn : public APawn
{
    GENERATED_BODY()

public:
    AMyPawn();

protected:
    virtual void BeginPlay() override;

public:
    virtual void Tick(float DeltaTime) override;
    virtual void SetupPlayerInputComponent(class UInputComponent* PlayerInputComponent) override;

    // ─────────────────────────────────────────
    //  컴포넌트 (Components)
    // ─────────────────────────────────────────

    /** 루트 충돌체 — CapsuleComponent */
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    TObjectPtr<UCapsuleComponent> CapsuleComp;

    /** 스켈레탈 메시 */
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    TObjectPtr<USkeletalMeshComponent> SkeletalMeshComp;

    /** 스프링 암 (카메라 붐) */
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    TObjectPtr<USpringArmComponent> SpringArmComp;

    /** 카메라 */
    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    TObjectPtr<UCameraComponent> CameraComp;

    // ─────────────────────────────────────────
    //  Enhanced Input — IMC & IA
    // ─────────────────────────────────────────

    /** 기본 Input Mapping Context */
    UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input")
    TObjectPtr<UInputMappingContext> DefaultMappingContext;

    /** 이동 입력 액션 (Vector2D) */
    UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input")
    TObjectPtr<UInputAction> MoveAction;

    /** 시점 회전 입력 액션 (Vector2D) */
    UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input")
    TObjectPtr<UInputAction> LookAction;

    // ─────────────────────────────────────────
    //  이동 / 회전 파라미터
    // ─────────────────────────────────────────

    /** 이동 속도 (cm/s) */
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Movement")
    float MoveSpeed = 600.f;

    /**
     * 마우스 감도
     * X : Yaw 감도, Y : Pitch 감도
     */
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Movement")
    FVector2D LookSensitivity = FVector2D(1.5f, 1.5f);

    /** Pitch 상한 (도) */
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Movement")
    float PitchMax = 70.f;

    /** Pitch 하한 (도, 음수) */
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Movement")
    float PitchMin = -70.f;

private:
    // ─────────────────────────────────────────
    //  입력 콜백 (Input Callbacks)
    // ─────────────────────────────────────────

    /** WASD 이동 처리 */
    void HandleMove(const FInputActionValue& Value);

    /** 마우스 시점 회전 처리 */
    void HandleLook(const FInputActionValue& Value);

    // ─────────────────────────────────────────
    //  내부 상태 (Internal State)
    // ─────────────────────────────────────────

    /**
     * 누적 Pitch 값 — Pawn 자체에 직접 적용하므로
     * 매 프레임 클램프를 위해 별도 추적
     */
    float CurrentPitch = 0.f;
};
```

### MyPawn.cpp

```cpp

// MyPawn.cpp
#include "MyPawn.h"

// 컴포넌트 헤더
#include "Components/CapsuleComponent.h"
#include "Components/SkeletalMeshComponent.h"
#include "GameFramework/SpringArmComponent.h"
#include "Camera/CameraComponent.h"

// Enhanced Input 헤더
#include "EnhancedInputComponent.h"
#include "EnhancedInputSubsystems.h"
#include "InputActionValue.h"

// ─────────────────────────────────────────────────────────────────────────────
//  생성자
// ─────────────────────────────────────────────────────────────────────────────
AMyPawn::AMyPawn()
{
    PrimaryActorTick.bCanEverTick = true;

    // ── 1) CapsuleComponent → RootComponent ──────────────────────────────────
    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));
    SetRootComponent(CapsuleComp);

    // 물리 시뮬레이션 OFF — 코드로 직접 제어
    CapsuleComp->SetSimulatePhysics(false);

    // 콜리전 프로파일: 필요에 따라 변경 가능
    CapsuleComp->SetCollisionProfileName(TEXT("Pawn"));

    // 캡슐 기본 크기 (Height 절반, Radius)
    CapsuleComp->InitCapsuleSize(42.f, 96.f);

    // ── 2) SkeletalMeshComponent → Capsule에 부착 ────────────────────────────
    SkeletalMeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("SkeletalMeshComp"));
    SkeletalMeshComp->SetupAttachment(CapsuleComp);

    // 물리 시뮬레이션 OFF
    SkeletalMeshComp->SetSimulatePhysics(false);

    // 메시를 캡슐 하단에 맞게 오프셋 (에디터에서 메시 교체 후 조정 권장)
    SkeletalMeshComp->SetRelativeLocationAndRotation(
        FVector(0.f, 0.f, -96.f),          // Z 오프셋
        FRotator(0.f, -90.f, 0.f)           // 메시 방향 보정
    );

    // ── 3) SpringArmComponent → Capsule에 부착 ───────────────────────────────
    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));
    SpringArmComp->SetupAttachment(CapsuleComp);
    SpringArmComp->TargetArmLength = 400.f;             // 붐 길이
    SpringArmComp->SetRelativeLocation(FVector(0.f, 0.f, 60.f)); // 시점 높이

    // Pawn의 Yaw 회전을 SpringArm이 따라가지 않도록 설정
    // (Pawn 자체를 회전시키므로 상속만으로 충분)
    SpringArmComp->bUsePawnControlRotation = false;
    SpringArmComp->bInheritPitch           = true;
    SpringArmComp->bInheritYaw             = true;
    SpringArmComp->bInheritRoll            = false;

    // ── 4) CameraComponent → SpringArm 끝에 부착 ─────────────────────────────
    CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComp"));
    CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);
    CameraComp->bUsePawnControlRotation = false;        // 카메라 자체는 회전 안 함

    // ── 5) 엔진 기본 컨트롤러 회전 연동 OFF ──────────────────────────────────
    //       (직접 AddActorLocalRotation으로 처리하므로 모두 끔)
    bUseControllerRotationPitch = false;
    bUseControllerRotationYaw   = false;
    bUseControllerRotationRoll  = false;
}

// ─────────────────────────────────────────────────────────────────────────────
//  BeginPlay — Enhanced Input IMC 등록
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::BeginPlay()
{
    Super::BeginPlay();

    // LocalPlayer의 EnhancedInputSubsystem에 IMC를 우선순위 0으로 추가
    if (APlayerController* PC = Cast<APlayerController>(GetController()))
    {
        if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
            ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PC->GetLocalPlayer()))
        {
            Subsystem->AddMappingContext(DefaultMappingContext, 0);
        }
    }
}

// ─────────────────────────────────────────────────────────────────────────────
//  Tick
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);
    // 이동·회전 로직은 입력 콜백(HandleMove / HandleLook)에서 처리합니다.
    // Tick에서 추가 처리가 필요하다면 여기에 작성하세요.
}

// ─────────────────────────────────────────────────────────────────────────────
//  SetupPlayerInputComponent — Enhanced Input 바인딩
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    Super::SetupPlayerInputComponent(PlayerInputComponent);

    // EnhancedInputComponent로 캐스트
    if (UEnhancedInputComponent* EIC = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        // IA_Move — 키를 누르고 있는 동안(Triggered) 매 프레임 호출
        EIC->BindAction(MoveAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleMove);

        // IA_Look — 마우스가 움직이는 동안(Triggered) 매 프레임 호출
        EIC->BindAction(LookAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleLook);
    }
}

// ─────────────────────────────────────────────────────────────────────────────
//  HandleMove — WASD 이동
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::HandleMove(const FInputActionValue& Value)
{
    // IA_Move 는 Vector2D 타입
    //   X : 좌우 (A=-1, D=+1)
    //   Y : 전후 (S=-1, W=+1)
    const FVector2D MovementInput = Value.Get<FVector2D>();

    if (MovementInput.IsNearlyZero())
    {
        return;
    }

    // DeltaTime은 Tick이 아닌 콜백에서는 직접 구할 수 없으므로
    // GetWorld()->GetDeltaSeconds() 로 가져옵니다.
    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // Pawn의 Forward / Right 벡터 기준으로 로컬 오프셋 계산
    const FVector ForwardDelta = GetActorForwardVector() * MovementInput.Y * MoveSpeed * DeltaTime;
    const FVector RightDelta   = GetActorRightVector()   * MovementInput.X * MoveSpeed * DeltaTime;

    // bSweep = true → 충돌 감지를 활성화한 채로 이동
    AddActorWorldOffset(ForwardDelta + RightDelta, /*bSweep=*/true);
}

// ─────────────────────────────────────────────────────────────────────────────
//  HandleLook — 마우스 회전 (Yaw + Pitch)
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::HandleLook(const FInputActionValue& Value)
{
    // IA_Look 는 Vector2D 타입
    //   X : 마우스 수평 이동량 → Yaw
    //   Y : 마우스 수직 이동량 → Pitch
    const FVector2D LookInput = Value.Get<FVector2D>();

    if (LookInput.IsNearlyZero())
    {
        return;
    }

    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // ── Yaw (좌우 회전) ─────────────────────────────────────────────────────
    //    Pawn 자체를 Z축 기준으로 회전시킵니다.
    {
        const float YawDelta = LookInput.X * LookSensitivity.X;   // DeltaTime 불필요 (마우스 델타값 자체가 프레임 독립적)
        AddActorLocalRotation(FRotator(0.f, YawDelta, 0.f));
    }

    // ── Pitch (상하 회전) ────────────────────────────────────────────────────
    //    SpringArm을 Y축 기준으로 회전시켜 카메라 고개 숙임/들기를 구현합니다.
    //    Pawn 전체를 Pitch 회전시키면 이동 방향도 틀어지므로 SpringArm만 회전시킵니다.
    {
        // 마우스 Y 입력 반전 여부: 일반적으로 Y 양수 = 위 → Pitch 감소
        const float PitchDelta = -LookInput.Y * LookSensitivity.Y;

        // 누적값 계산 후 클램프
        CurrentPitch = FMath::Clamp(CurrentPitch + PitchDelta, PitchMin, PitchMax);

        // SpringArm의 절대 Pitch를 설정 (누적값으로 덮어씀)
        FRotator ArmRotation = SpringArmComp->GetRelativeRotation();
        ArmRotation.Pitch    = CurrentPitch;
        SpringArmComp->SetRelativeRotation(ArmRotation);
    }
}
```

`.h/.cpp` 넣은 뒤 **에디터 닫고** → `.sln` 우클릭 → **Generate Visual Studio project files** → Visual Studio에서 빌드

### 2. 에디터에서 해야 할 것 (순서대로)

**① IA 에셋 2개 생성** 콘텐츠 브라우저 빈 곳 우클릭 → `Input > Input Action`

- `IA_Move` : Value Type → `Axis2D (Vector2D)`
- `IA_Look` : Value Type → `Axis2D (Vector2D)`

**② IMC 생성 및 키 매핑** 우클릭 → `Input > Input Mapping Context` → `IMC_Default`

| 액션      | 키                | Modifier    |
| ------- | ---------------- | ----------- |
| IA_Move | W                | 없음          |
| IA_Move | S                | Negate (Y축) |
| IA_Move | D                | 없음          |
| IA_Move | A                | Negate (X축) |
| IA_Look | Mouse XY 2D-Axis | 없음          |

**③ BP_MyPawn 생성** `MyPawn` C++ 클래스 우클릭 → `Create Blueprint class based on MyPawn` → `BP_MyPawn`

BP_MyPawn 열고 디테일 패널에서:

- `Default Mapping Context` → `IMC_Default`
- `Move Action` → `IA_Move`
- `Look Action` → `IA_Look`
- `Skeletal Mesh` 슬롯에 마네킹 등 메시 지정

**④ GameMode 설정** `BP_GameMode` 생성 (없으면) → `Default Pawn Class` → `BP_MyPawn`

프로젝트 세팅 → Maps & Modes → Default GameMode → `BP_GameMode`

- [ ] 마우스 좌우로 Yaw 회전이 되는가

- [ ] Pitch가 70도 이상 꺾이지 않는가 (클램프 확인)
- [ ] 이동 중 프레임 낮춰도 속도 일정한가 (`t.MaxFPS 10` 콘솔 명령으로 테스트)

### 4. 자주 나오는 에러 & 해결

|에러|원인|해결|
|---|---|---|
|`EnhancedInput` 컴파일 에러|Build.cs 누락|`"EnhancedInput"` 모듈 추가|

```cpp
// MyPawn.h
// 해당 헤더 파일이 컴파일될 때 딱 한 번만 포함되도록 보장하는 지시어입니다. (중복 정의 방지)

// 언리얼 엔진의 가장 핵심적인 데이터 타입(FString, FVector 등)과 기본 기능들을 모아둔 헤더입니다.

// 실제 상세한 기능(멤버 함수 등)은 .cpp 파일에서 해당 헤더를 직접 include하여 사용합니다.

// 언리얼 에디터와 엔진이 이 클래스를 인식하고 관리할 수 있도록 만드는 매크로입니다.

// AMyPawn 클래스 정의입니다. APawn 클래스를 상속받습니다.

public:

    AMyPawn();

protected:

    virtual void BeginPlay() override;

    // 플레이어의 입력 장치와 C++ 함수를 연결해주는 함수입니다. 부모 클래스의 함수를 재정의합니다.

    // 변수의 메모리 관리는 언리얼 엔진 5의 표준 오브젝트 포인터 방식인 TObjectPtr로 래핑합니다.

    UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Components")
    TObjectPtr<USpringArmComponent> SpringArmComp;

    // 에디터에서 할당 가능하고, 블루프린트에서 읽기 전용이며, "Input" 카테고리로 분류합니다.
    UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input")

    // 에디터 수정 가능, 블루프린트 읽기/쓰기 가능, "Movement" 카테고리, 기본값 600.0 (초당 6미터)
```

```cpp

// MyPawn.cpp
#include "MyPawn.h"

// ──────────────────────────────────────────────
// 컴포넌트 헤더 인클루드
// ──────────────────────────────────────────────

// UCapsuleComponent 사용을 위한 헤더.
// 캐릭터의 물리 충돌 범위를 캡슐 형태로 정의한다.
#include "Components/CapsuleComponent.h"

// USkeletalMeshComponent 사용을 위한 헤더.
// 애니메이션이 적용되는 3D 캐릭터 메시를 담당한다.
#include "Components/SkeletalMeshComponent.h"

// USpringArmComponent 사용을 위한 헤더.
// 카메라와 Pawn 사이에 일정 거리를 유지하는 "붐대" 역할을 한다.
// 벽 등 장애물이 있으면 카메라를 자동으로 당겨준다.
#include "GameFramework/SpringArmComponent.h"

// UCameraComponent 사용을 위한 헤더.
// 플레이어가 실제로 보는 시점(뷰)을 렌더링하는 카메라다.
#include "Camera/CameraComponent.h"

// ──────────────────────────────────────────────
// Enhanced Input 헤더 인클루드
// ──────────────────────────────────────────────

// UEnhancedInputComponent 사용을 위한 헤더.
// 기존 InputComponent를 대체하며, InputAction과 트리거 이벤트를 바인딩한다.
#include "EnhancedInputComponent.h"

// UEnhancedInputLocalPlayerSubsystem 사용을 위한 헤더.
// IMC(Input Mapping Context)를 런타임에 추가/제거하는 서브시스템이다.
#include "EnhancedInputSubsystems.h"

// FInputActionValue 사용을 위한 헤더.
// 입력 콜백 함수의 파라미터 타입으로, bool / float / Vector2D / Vector3D 값을 담는다.
#include "InputActionValue.h"


// ─────────────────────────────────────────────────────────────────────────────
//  생성자
// ─────────────────────────────────────────────────────────────────────────────
AMyPawn::AMyPawn()
{
    // 이 Pawn이 매 프레임 Tick()을 호출받도록 활성화한다.
    // false로 설정하면 Tick이 실행되지 않아 CPU를 아낄 수 있다.
    PrimaryActorTick.bCanEverTick = true;

    // ── 1) CapsuleComponent → RootComponent ──────────────────────────────────

    // UCapsuleComponent를 "CapsuleComp"라는 이름으로 생성한다.
    // CreateDefaultSubobject는 생성자 안에서만 호출 가능하며,
    // 에디터 디테일 패널에 해당 컴포넌트를 노출시킨다.
    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));

    // 방금 만든 CapsuleComp를 이 Actor의 RootComponent로 지정한다.
    // RootComponent는 트랜스폼(위치·회전·스케일)의 기준점이 된다.
    SetRootComponent(CapsuleComp);

    // 캡슐에 UE 물리 엔진(피직스) 시뮬레이션을 끈다.
    // 중력·충격 등 물리 연산을 엔진에 맡기지 않고,
    // 코드(HandleMove 등)로 직접 위치를 제어하기 위함이다.
    CapsuleComp->SetSimulatePhysics(false);

    // 충돌 프로파일을 "Pawn"으로 설정한다.
    // "Pawn" 프로파일은 캐릭터끼리의 기본 충돌 반응이 미리 정의되어 있다.
    // 프로젝트 필요에 따라 커스텀 프로파일로 변경 가능하다.
    CapsuleComp->SetCollisionProfileName(TEXT("Pawn"));

    // 캡슐의 크기를 초기화한다.
    // 첫 번째 인자(42.f): 캡슐 반지름(Radius) — 좌우 충돌 범위
    // 두 번째 인자(96.f): 캡슐 절반 높이(HalfHeight) — 전체 높이는 192cm
    CapsuleComp->InitCapsuleSize(42.f, 96.f);

    // ── 2) SkeletalMeshComponent → Capsule에 부착 ────────────────────────────

    // USkeletalMeshComponent를 "SkeletalMeshComp"라는 이름으로 생성한다.
    SkeletalMeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("SkeletalMeshComp"));

    // SkeletalMeshComp를 CapsuleComp에 자식으로 부착한다.
    // 이후 캡슐이 움직이면 메시도 함께 따라간다.
    SkeletalMeshComp->SetupAttachment(CapsuleComp);

    // 메시에도 물리 시뮬레이션을 끈다.
    // 켜두면 Ragdoll처럼 물리 연산을 받아 의도치 않게 쓰러질 수 있다.
    SkeletalMeshComp->SetSimulatePhysics(false);

    // 메시의 상대 위치와 회전을 설정한다.
    // - FVector(0, 0, -96): 캡슐 중심이 메시의 허리 높이이므로,
    //   Z를 -96(캡슐 HalfHeight)만큼 내려 메시 발바닥이 캡슐 바닥에 닿게 맞춘다.
    // - FRotator(0, -90, 0): UE 기본 좌표계에서 메시가 옆을 향하는 경우가 많아,
    //   Yaw를 -90도 돌려 Forward 방향(+X)과 일치시킨다.
    //   실제 메시에 따라 에디터에서 재조정이 필요할 수 있다.
    SkeletalMeshComp->SetRelativeLocationAndRotation(
        FVector(0.f, 0.f, -96.f),
        FRotator(0.f, -90.f, 0.f)
    );

    // ── 3) SpringArmComponent → Capsule에 부착 ───────────────────────────────

    // USpringArmComponent를 "SpringArmComp"라는 이름으로 생성한다.
    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));

    // SpringArmComp를 CapsuleComp에 부착한다.
    // Pawn이 움직이면 SpringArm도 함께 이동한다.
    SpringArmComp->SetupAttachment(CapsuleComp);

    // SpringArm(붐대)의 길이를 400 유닛(약 4m)으로 설정한다.
    // 값이 클수록 카메라가 캐릭터에서 멀어진다. (TPS 거리 조절)
    SpringArmComp->TargetArmLength = 400.f;

    // SpringArm의 기준 위치를 캡슐 중심보다 Z축으로 60 유닛 위로 올린다.
    // 카메라가 캐릭터의 어깨/머리 높이 근처에서 바라보게 하기 위함이다.
    SpringArmComp->SetRelativeLocation(FVector(0.f, 0.f, 60.f));

    // SpringArm이 Pawn의 컨트롤러 회전을 따라가지 않도록 끈다.
    // 이 코드에서는 Pawn 자체를 직접 회전시키므로(HandleLook의 AddActorLocalRotation),
    // 컨트롤러 회전 연동은 불필요하다.
    SpringArmComp->bUsePawnControlRotation = false;

    // SpringArm이 부모(CapsuleComp)의 Pitch(앞뒤 기울기)를 상속받도록 한다.
    // true이면 Pawn이 Pitch 회전할 때 SpringArm도 함께 기울어진다.
    SpringArmComp->bInheritPitch = true;

    // SpringArm이 부모의 Yaw(좌우 회전)를 상속받도록 한다.
    // Pawn 자체를 Yaw 회전시키므로 SpringArm도 자연스럽게 따라간다.
    SpringArmComp->bInheritYaw = true;

    // SpringArm이 부모의 Roll(측면 기울기)은 상속받지 않는다.
    // Roll 상속은 카메라가 화면을 기울이는 어색한 결과를 낳으므로 끈다.
    SpringArmComp->bInheritRoll = false;

    // ── 4) CameraComponent → SpringArm 끝에 부착 ─────────────────────────────

    // UCameraComponent를 "CameraComp"라는 이름으로 생성한다.
    CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComp"));

    // CameraComp를 SpringArm의 소켓 끝(SocketName)에 부착한다.
    // USpringArmComponent::SocketName은 붐대 끝 소켓의 이름 상수다.
    // 이렇게 해야 카메라가 항상 붐대 끝에 위치하며 장애물 충돌 보정을 받는다.
    CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);

    // 카메라 자체가 컨트롤러 회전을 따라 독립적으로 회전하지 않도록 끈다.
    // 카메라의 회전은 부모인 SpringArm을 통해 간접적으로 이루어진다.
    CameraComp->bUsePawnControlRotation = false;

    // ── 5) 엔진 기본 컨트롤러 회전 연동 OFF ──────────────────────────────────

    // 컨트롤러의 Pitch 값이 Pawn의 Pitch에 자동으로 적용되지 않도록 끈다.
    // Pitch는 SpringArm에서만 처리하므로 Pawn 자체에는 필요 없다.
    bUseControllerRotationPitch = false;

    // 컨트롤러의 Yaw 값이 Pawn의 Yaw에 자동으로 적용되지 않도록 끈다.
    // Yaw는 HandleLook에서 AddActorLocalRotation으로 직접 처리한다.
    bUseControllerRotationYaw = false;

    // 컨트롤러의 Roll 값이 Pawn의 Roll에 자동으로 적용되지 않도록 끈다.
    // Roll 회전은 이 Pawn에서 사용하지 않는다.
    bUseControllerRotationRoll = false;
}


// ─────────────────────────────────────────────────────────────────────────────
//  BeginPlay — Enhanced Input IMC 등록
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::BeginPlay()
{
    // 부모 클래스(APawn)의 BeginPlay를 먼저 호출한다.
    // UE 컴포넌트 초기화, 오버랩 이벤트 등록 등 엔진 내부 로직이 여기서 실행된다.
    // Super 호출을 빠뜨리면 컴포넌트가 정상 작동하지 않을 수 있다.
    Super::BeginPlay();

    // 이 Pawn을 소유한 컨트롤러를 APlayerController로 캐스트한다.
    // AI 컨트롤러일 경우 캐스트가 실패해 nullptr이 되므로,
    // Enhanced Input은 플레이어 전용 기능임을 암묵적으로 보장한다.
    if (APlayerController* PC = Cast<APlayerController>(GetController()))
    {
        // 로컬 플레이어에 연결된 EnhancedInputLocalPlayerSubsystem을 가져온다.
        // 이 서브시스템이 Input Mapping Context(IMC)의 추가/제거를 관리한다.
        // GetLocalPlayer()가 nullptr이면 서브시스템도 nullptr이 된다.
        if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
            ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PC->GetLocalPlayer()))
        {
            // DefaultMappingContext(IMC 에셋)를 우선순위 0으로 서브시스템에 등록한다.
            // 숫자가 높을수록 우선순위가 높으며, 같은 키에 여러 IMC가 겹칠 때 우선 처리된다.
            // 0은 가장 낮은 우선순위로, 기본 입력 매핑에 적합하다.
            Subsystem->AddMappingContext(DefaultMappingContext, 0);
        }
    }
}


// ─────────────────────────────────────────────────────────────────────────────
//  Tick
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::Tick(float DeltaTime)
{
    // 부모 클래스의 Tick을 먼저 호출한다.
    // 컴포넌트 Tick, 타이머 등 엔진 내부 갱신 처리가 여기서 이루어진다.
    Super::Tick(DeltaTime);

    // 이동·회전 로직은 입력 콜백(HandleMove / HandleLook)에서 처리합니다.
    // 매 프레임 추가로 처리할 게임 로직이 있다면 이 아래에 작성하세요.
}


// ─────────────────────────────────────────────────────────────────────────────
//  SetupPlayerInputComponent — Enhanced Input 바인딩
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    // 부모 클래스의 SetupPlayerInputComponent를 먼저 호출한다.
    // 엔진이 기본 입력 컴포넌트를 초기화하는 과정이 포함되어 있다.
    Super::SetupPlayerInputComponent(PlayerInputComponent);

    // 전달된 PlayerInputComponent를 UEnhancedInputComponent로 캐스트한다.
    // 프로젝트 세팅에서 DefaultInputComponentClass가 UEnhancedInputComponent로
    // 설정되어 있어야 캐스트가 성공한다. 그렇지 않으면 nullptr이 된다.
    if (UEnhancedInputComponent* EIC = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        // MoveAction(IA_Move InputAction 에셋)에 ETriggerEvent::Triggered 이벤트를 연결한다.
        // Triggered: 키를 누르고 있는 매 프레임마다 HandleMove가 호출된다.
        // (Started는 최초 1회, Completed는 키를 뗄 때만 호출됨)
        EIC->BindAction(MoveAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleMove);

        // LookAction(IA_Look InputAction 에셋)에 ETriggerEvent::Triggered 이벤트를 연결한다.
        // 마우스가 움직이는 동안 매 프레임 HandleLook이 호출된다.
        EIC->BindAction(LookAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleLook);
    }
}


// ─────────────────────────────────────────────────────────────────────────────
//  HandleMove — WASD 이동
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::HandleMove(const FInputActionValue& Value)
{
    // FInputActionValue에서 Vector2D 타입으로 실제 입력값을 꺼낸다.
    // IA_Move의 Value Type이 Axis2D(Vector2D)로 설정되어 있어야 한다.
    //   X 축: A키 = -1.0, D키 = +1.0 (좌우)
    //   Y 축: S키 = -1.0, W키 = +1.0 (전후)
    const FVector2D MovementInput = Value.Get<FVector2D>();

    // 입력값이 거의 0에 가깝다면(데드존) 이동 처리를 건너뛴다.
    // 부동소수점 오차로 인한 미세 이동을 방지한다.
    if (MovementInput.IsNearlyZero())
    {
        return;
    }

    // 이 콜백은 Tick이 아니므로 DeltaTime 파라미터가 없다.
    // GetWorld()->GetDeltaSeconds()로 현재 프레임의 경과 시간을 가져온다.
    // DeltaTime을 곱해야 프레임레이트와 무관하게 일정한 속도로 이동한다.
    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // Pawn의 Forward 벡터(앞 방향) × 전후 입력 × 이동속도 × DeltaTime
    // 결과: 이번 프레임에 앞뒤로 이동할 월드 오프셋 벡터
    const FVector ForwardDelta = GetActorForwardVector() * MovementInput.Y * MoveSpeed * DeltaTime;

    // Pawn의 Right 벡터(오른쪽 방향) × 좌우 입력 × 이동속도 × DeltaTime
    // 결과: 이번 프레임에 좌우로 이동할 월드 오프셋 벡터
    const FVector RightDelta = GetActorRightVector() * MovementInput.X * MoveSpeed * DeltaTime;

    // 두 벡터를 합산한 최종 오프셋만큼 Actor를 월드 좌표로 이동시킨다.
    // bSweep = true: 이동 경로에 충돌 검사를 수행한다.
    //   벽이나 바닥에 파묻히지 않고 표면에서 멈추게 된다.
    //   false로 바꾸면 충돌 없이 텔레포트처럼 순간 이동한다.
    AddActorWorldOffset(ForwardDelta + RightDelta, /*bSweep=*/true);
}


// ─────────────────────────────────────────────────────────────────────────────
//  HandleLook — 마우스 회전 (Yaw + Pitch)
// ─────────────────────────────────────────────────────────────────────────────
void AMyPawn::HandleLook(const FInputActionValue& Value)
{
    // FInputActionValue에서 Vector2D 타입으로 마우스 이동량(델타)을 꺼낸다.
    // IA_Look의 Value Type이 Axis2D(Vector2D)여야 한다.
    //   X: 마우스 수평 이동량 → Yaw(좌우 회전)에 사용
    //   Y: 마우스 수직 이동량 → Pitch(상하 회전)에 사용
    const FVector2D LookInput = Value.Get<FVector2D>();

    // 마우스 입력이 거의 없으면 회전 처리를 건너뛴다.
    if (LookInput.IsNearlyZero())
    {
        return;
    }

    // Tick이 아닌 콜백이므로 직접 DeltaTime을 가져온다.
    // (아래 Yaw 계산에서는 실제로 사용하지 않지만, 필요 시 참고용으로 남긴다.)
    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // ── Yaw (좌우 회전) ─────────────────────────────────────────────────────
    {
        // 마우스 X 이동량에 수평 감도를 곱해 이번 프레임의 Yaw 회전량을 구한다.
        // 마우스 델타값은 이미 프레임 시간에 비례하므로 DeltaTime을 곱하지 않는다.
        // (곱하면 프레임레이트가 높을수록 회전이 느려지는 버그가 발생한다.)
        const float YawDelta = LookInput.X * LookSensitivity.X;

        // Pawn 자체를 로컬 Z축 기준으로 YawDelta만큼 회전시킨다.
        // Pawn이 회전하면 SpringArm과 Camera도 함께 돌아가므로
        // 시점 방향과 이동 방향(ForwardVector)이 자동으로 일치한다.
        AddActorLocalRotation(FRotator(0.f, YawDelta, 0.f));
    }

    // ── Pitch (상하 회전) ────────────────────────────────────────────────────
    {
        // 마우스 Y 이동량에 수직 감도를 곱하고 부호를 반전한다.
        // UE에서 마우스를 위로 올리면 Y가 양수이지만,
        // 카메라는 위를 바라볼 때 Pitch가 음수(−)여야 한다.
        // 따라서 -1을 곱해 방향을 맞춘다.
        const float PitchDelta = -LookInput.Y * LookSensitivity.Y;

        // 현재까지 누적된 Pitch에 이번 프레임의 변화량을 더한다.
        // FMath::Clamp로 PitchMin~PitchMax 범위를 벗어나지 못하게 막아
        // 카메라가 완전히 뒤집히는 것을 방지한다.
        CurrentPitch = FMath::Clamp(CurrentPitch + PitchDelta, PitchMin, PitchMax);

        // SpringArm의 현재 상대 회전값(FRotator)을 가져온다.
        // Yaw(Pawn이 관리)와 Roll은 건드리지 않고 Pitch만 수정하기 위해
        // 기존 값을 먼저 읽어온다.
        FRotator ArmRotation = SpringArmComp->GetRelativeRotation();

        // Pitch 값만 누적된 CurrentPitch로 덮어쓴다.
        // 매 프레임 절댓값으로 설정하기 때문에 오차가 누적되지 않는다.
        ArmRotation.Pitch = CurrentPitch;

        // 수정된 회전값을 SpringArm에 적용한다.
        // 결과적으로 카메라만 상하로 회전하고 Pawn 이동 방향은 바뀌지 않는다.
        SpringArmComp->SetRelativeRotation(ArmRotation);
    }
}
```

```cpp
// ── Enhanced Input (향상된 입력 시스템) 헤더 ───────────────────────────────
// 플레이어의 입력 신호와 실제 C++ 함수를 바인딩(연결)해주는 컴포넌트 헤더입니다.

#include "EnhancedInputSubsystems.h"
// 입력 장치(키보드, 마우스 등)로부터 들어오는 데이터 값(디지털, 1D, 2D, 3D)을 담는 구조체 헤더입니다.

// ─────────────────────────────────────────────────────────────────────────────
//  생성자 (Constructor) : 객체가 메모리에 생성될 때 초기 설정을 하는 곳

// AMyPawn 클래스의 기본 생성자 정의입니다.
AMyPawn::AMyPawn()
{
    // 이 액터가 매 프레임마다 Tick() 함수를 호출할 수 있도록 허용합니다. (true = 매 프레임 실행)
    PrimaryActorTick.bCanEverTick = true;

    // ── 1) CapsuleComponent → RootComponent ──────────────────────────────────
    // 캡슐 컴포넌트를 기본 서브 오브젝트로 생성하고 내부 이름을 "CapsuleComp"로 지정합니다.
    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));
    // 생성한 캡슐 컴포넌트를 이 Pawn 액터의 최상위 루트(기본 중심점) 컴포넌트로 설정합니다.
    SetRootComponent(CapsuleComp);

    // 캡슐의 언리얼 물리 엔진(중력, 힘) 시뮬레이션을 끕니다. (이동을 코드로 직접 완벽히 제어하기 위함)
    CapsuleComp->SetSimulatePhysics(false);

    // 충돌 규칙(프로파일) 이름을 'Pawn'으로 설정하여 플레이어 캐릭터에 맞는 충돌 반응을 켜줍니다.
    CapsuleComp->SetCollisionProfileName(TEXT("Pawn"));

    // 캡슐의 크기를 설정합니다. 반지름(Radius) 42.0, 절반 높이(Half Height) 96.0으로 지정합니다.
    CapsuleComp->InitCapsuleSize(42.f, 96.f);

    // ── 2) SkeletalMeshComponent → Capsule에 부착 ────────────────────────────
    // 스켈레탈 메쉬 컴포넌트를 생성하고 내부 이름을 "SkeletalMeshComp"로 지정합니다.
    SkeletalMeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("SkeletalMeshComp"));
    // 이 메쉬 컴포넌트를 루트 컴포넌트인 캡슐 컴포넌트의 자식으로 부착합니다.
    SkeletalMeshComp->SetupAttachment(CapsuleComp);

    // 캐릭터 메쉬 역시 자체적인 물리 시뮬레이션에 의해 흐느적거리지 않도록 물리를 끕니다.
    SkeletalMeshComp->SetSimulatePhysics(false);

    // 캡슐 중심에 떠 있는 메쉬를 캡슐의 바닥 높이와 정면 방향에 맞게 위치와 회전값을 조정합니다.
    SkeletalMeshComp->SetRelativeLocationAndRotation(
        FVector(0.f, 0.f, -96.f),          // Z축으로 -96만큼 내려서 캡슐 바닥에 발을 맞춥니다.
        FRotator(0.f, -90.f, 0.f)           // 언리얼 기준 정면(X축)을 바라보도록 90도 회전시킵니다.
    );

    // ── 3) SpringArmComponent → Capsule에 부착 ───────────────────────────────
    // 스프링 암 컴포넌트를 생성하고 내부 이름을 "SpringArmComp"로 지정합니다.
    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));
    // 스프링 암을 루트 컴포넌트인 캡슐 컴포넌트의 자식으로 부착합니다.
    SpringArmComp->SetupAttachment(CapsuleComp);
    // 캐릭터와 카메라 사이의 거리(붐 길이)를 400.0 유닛(cm)으로 설정합니다.
    SpringArmComp->TargetArmLength = 400.f;             
    // 스프링 암의 시작 위치를 캐릭터의 중심보다 약간 위(Z축으로 60cm)인 목/어깨 높이로 올립니다.
    SpringArmComp->SetRelativeLocation(FVector(0.f, 0.f, 60.f)); 

    // 컨트롤러의 회전 값을 스프링 암이 직접 사용하지 않도록 강제로 끕니다.
    SpringArmComp->bUsePawnControlRotation = false;
    // 부모(Pawn)의 Pitch(상하 고개 끄덕임) 회전 값이 스프링 암에 그대로 상속되도록 설정합니다.
    SpringArmComp->bInheritPitch           = true;
    // 부모(Pawn)의 Yaw(좌우 회전) 회전 값이 스프링 암에 그대로 상속되도록 설정합니다.
    SpringArmComp->bInheritYaw             = true;
    // 부모(Pawn)의 Roll(좌우 기울어짐) 회전 값은 상속받지 않도록 차단합니다. (화면이 기우는 것 방지)
    SpringArmComp->bInheritRoll            = false;

    // 카메라 컴포넌트를 생성하고 내부 이름을 "CameraComp"로 지정합니다.

    CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);
    // 카메라 자체가 컨트롤러 회전값을 직접 받지 않게 합니다. (앞의 스프링 암이 회전하므로 카메라는 가만히 지켜봄)
    CameraComp->bUsePawnControlRotation = false;        

    // ── 5) 엔진 기본 컨트롤러 회전 연동 OFF ──────────────────────────────────
    // 플레이어가 마우스를 돌릴 때 컨트롤러의 Pitch 회전이 Pawn 자체를 상하로 뒤집지 못하게 끕니다.
    bUseControllerRotationPitch = false;
    // 플레이어가 마우스를 돌릴 때 컨트롤러의 Yaw 회전이 Pawn 자체를 강제로 회전시키지 못하게 끕니다. (코드로 직접 제어 예정)
    bUseControllerRotationYaw   = false;
    // 플레이어가 마우스를 돌릴 때 컨트롤러의 Roll 회전이 Pawn을 옆으로 눕히지 못하게 끕니다.
    bUseControllerRotationRoll  = false;
}

// ─────────────────────────────────────────────────────────────────────────────
//  BeginPlay — 게임이 시작되거나 액터가 배치될 때 1회 호출되는 함수
// ─────────────────────────────────────────────────────────────────────────────

void AMyPawn::BeginPlay()
{
    // 부모 클래스(APawn)의 BeginPlay 내부 로직을 먼저 안전하게 실행합니다.
    Super::BeginPlay();

    // 이 Pawn을 조종하고 있는 플레이어 컨트롤러(APlayerController)가 존재하는지 캐스팅하여 확인합니다.
    if (APlayerController* PC = Cast<APlayerController>(GetController()))
    {
        // 로컬 플레이어로부터 Enhanced Input 시스템을 다루는 핵심 서브시스템을 찾아옵니다.
        if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
            ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PC->GetLocalPlayer()))
        {
            // 찾은 서브시스템에 우리가 헤더에서 지정한 'DefaultMappingContext(입력 매핑 데이터)'를 우선순위 0으로 등록합니다.
            Subsystem->AddMappingContext(DefaultMappingContext, 0);
        }
    }
}

// ─────────────────────────────────────────────────────────────────────────────
//  Tick — 매 프레임마다 반복 호출되는 함수 (게임의 심장박동 역할)
// ─────────────────────────────────────────────────────────────────────────────

void AMyPawn::Tick(float DeltaTime)
{
    // 부모 클래스의 Tick 내부 로직을 먼저 실행합니다.
    Super::Tick(DeltaTime);

    // 이동·회전 로직은 입력 콜백(HandleMove / HandleLook)에서 처리합니다.
    // Tick에서 추가 처리가 필요하다면 여기에 작성하세요.
}

// ─────────────────────────────────────────────────────────────────────────────
//  SetupPlayerInputComponent — 입력 컴포넌트 세팅 및 기능 바인딩
// ─────────────────────────────────────────────────────────────────────────────

void AMyPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    // 부모 클래스의 기본 입력 컴포넌트 설정을 먼저 수행합니다.
    Super::SetupPlayerInputComponent(PlayerInputComponent);

    // 기본 입력 컴포넌트(PlayerInputComponent)를 확장된 시스템인 'EnhancedInputComponent'로 캐스팅합니다.
    if (UEnhancedInputComponent* EIC = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        // MoveAction(IA_Move)이 실행되는 상태(Triggered: 키를 누르고 있는 중)일 때, 이 클래스(this)의 HandleMove 함수를 실행하도록 연결합니다.
        EIC->BindAction(MoveAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleMove);

        // LookAction(IA_Look)이 실행되는 상태(Triggered: 마우스를 움직이는 중)일 때, 이 클래스(this)의 HandleLook 함수를 실행하도록 연결합니다.
        EIC->BindAction(LookAction, ETriggerEvent::Triggered, this, &AMyPawn::HandleLook);
    }
}

// ─────────────────────────────────────────────────────────────────────────────
//  HandleMove — WASD 입력에 따른 실제 캐릭터 이동 처리
// ─────────────────────────────────────────────────────────────────────────────

void AMyPawn::HandleMove(const FInputActionValue& Value)
{
    // IA_Move 로부터 전달받은 입력 데이터 값을 Vector2D 형태로 꺼내옵니다. (X: 좌우값, Y: 전후값)
    const FVector2D MovementInput = Value.Get<FVector2D>();

    // 만약 입력값(키보드 누름)이 전혀 없거나 0에 아주 가깝다면, 연산을 수행하지 않고 함수를 즉시 종료합니다.
    if (MovementInput.IsNearlyZero())
    {
        return;
    }

    // 한 프레임이 지나간 시간(초)을 월드(World)로부터 가져옵니다. (프레임률이 달라도 이동 속도를 일정하게 만들기 위함)
    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // 현재 캐릭터가 바라보는 정면 벡터(방향)에 플레이어의 전후 입력값(Y)과 이동 속도(MoveSpeed), 프레임 시간을 곱해 전진/후진 거리를 계산합니다.
    const FVector ForwardDelta = GetActorForwardVector() * MovementInput.Y * MoveSpeed * DeltaTime;
    // 현재 캐릭터의 우측 벡터(방향)에 플레이어의 좌우 입력값(X)과 이동 속도(MoveSpeed), 프레임 시간을 곱해 좌/우 이동 거리를 계산합니다.
    const FVector RightDelta   = GetActorRightVector()   * MovementInput.X * MoveSpeed * DeltaTime;

    // 계산된 전후/좌우 벡터를 더해 최종 이동 벡터를 만든 뒤, 월드 좌표 기준으로 캐릭터를 이동시킵니다.
    // bSweep=true 로 설정되어 있어 이동 경로에 벽이 있으면 뚫지 못하고 멈춰 서게 됩니다.
    AddActorWorldOffset(ForwardDelta + RightDelta, /*bSweep=*/true);
}

// ─────────────────────────────────────────────────────────────────────────────
//  HandleLook — 마우스 이동에 따른 카메라 시점 회전 처리 (Yaw + Pitch)
// ─────────────────────────────────────────────────────────────────────────────

void AMyPawn::HandleLook(const FInputActionValue& Value)
{
    // IA_Look 으로부터 전달받은 입력 데이터 값을 Vector2D 형태로 꺼내옵니다. (X: 마우스 가로 이동량, Y: 마우스 세로 이동량)
    const FVector2D LookInput = Value.Get<FVector2D>();

    // 마우스 움직임이 전혀 없다면, 불필요한 회전 연산을 하지 않도록 함수를 즉시 종료합니다.
    if (LookInput.IsNearlyZero())
    {
        return;
    }

    // 한 프레임이 지나간 시간(초)을 월드로부터 가져옵니다. (여기서는 아래 Yaw 연산 주석대로 사용되지는 않습니다)
    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    // ── Yaw (좌우 회전) ─────────────────────────────────────────────────────
    //    마우스를 좌우로 흔들었을 때 캐릭터 본체 자체를 Z축 기준으로 회전시키는 블록입니다.
    {
        // 마우스의 가로 이동량(X)에 사용자가 설정한 가로 감도(LookSensitivity.X)를 곱해 회전할 각도를 계산합니다.
        const float YawDelta = LookInput.X * LookSensitivity.X;   
        // 계산된 회전 각도를 가지고 캐릭터(Pawn) 자체의 로컬 Yaw 회전값에 더해 주어 캐릭터를 좌/우로 돌립니다.
        AddActorLocalRotation(FRotator(0.f, YawDelta, 0.f));
    }

    // ── Pitch (상하 회전) ────────────────────────────────────────────────────
    //    마우스를 위아래로 움직일 때 캐릭터 본체는 놔두고, 카메라(스프링암)만 상하로 움직이게 하는 블록입니다.
    {
        // 마우스 Y 입력을 반전(-) 시킵니다. (일반적으로 마우스를 위로 밀면 화면이 위를 보아야 하므로 Pitch 값을 감소시켜 고개를 들게 만듦)
        const float PitchDelta = -LookInput.Y * LookSensitivity.Y;

        // 기존의 현재 누적 Pitch 값에 방금 움직인 PitchDelta를 더한 후, 사용자가 지정한 최소/최대 각도(PitchMin ~ PitchMax)를 벗어나지 못하게 가둡니다.
        CurrentPitch = FMath::Clamp(CurrentPitch + PitchDelta, PitchMin, PitchMax);

        // 현재 스프링 암이 가지고 있는 상대 회전값(값의 뭉치)을 가져옵니다.
        FRotator ArmRotation = SpringArmComp->GetRelativeRotation();
        // 회전값 중 고개 숙임/들기를 담당하는 Pitch 부분만 우리가 새로 제한(Clamp)하여 계산한 CurrentPitch 값으로 교체합니다.
        ArmRotation.Pitch    = CurrentPitch;
        // 변경된 최종 회전 값을 스프링 암 컴포넌트에 다시 적용하여 카메라가 상하로 부드럽게 움직이도록 만듭니다.
        SpringArmComp->SetRelativeRotation(ArmRotation);
    }
}
```

## 🚀 추천 작업 프로세스

### 1단계: 엔진 에디터에서 기초 에셋 생성 (지금 하려는 단계)

코드에서 입력 액션(`UInputAction`)과 매핑 컨텍스트(`UInputMappingContext`)를 참조하거나 블루프린트에서 할당해야 하므로, 에셋을 먼저 만들어 두는 것이 좋습니다.

1. **C++ 클래스 생성**:
    
    - 에디터의 [Tools] -> [New C++ Class] 메뉴를 통해 **Pawn**을 상속받는 클래스(예: `MyPawn`)와 **GameModeBase**를 상속받는 클래스(예: `MyGameMode`)를 생성합니다.
        
2. **입력(Enhanced Input) 에셋 생성**:
    
    - 콘텐츠 브라우저에서 우클릭 -> [Input] 메뉴를 통해 아래 에셋들을 생성합니다.
        
    - `IA_Move` (Input Action, Value Type: **Axis2D**)
        
    - `IA_Look` (Input Action, Value Type: **Axis2D**)
        
    - `IMC_Default` (Input Mapping Context)를 생성하고 내부에서 `IA_Move`(WASD 키 및 Modifiers 설정)와 `IA_Look`(마우스 XY축 설정)을 매핑합니다.
        

### 2단계: Visual Studio (또는 Rider)에서 C++ 코드 작성

이제 에디터를 끄거나 라이브 코딩을 켠 상태로 코드를 작성합니다.

- `MyPawn.h / .cpp`: Capsule, SkeletalMesh, SpringArm, Camera 컴포넌트들을 생성하고 계층 구조를 설정합니다. (`SetupAttachment`)
    
- `Simulate Physics`를 `false`로 설정합니다.
    
- `SetupPlayerInputComponent`에서 `IA_Move`와 `IA_Look`에 반응할 함수를 바인딩합니다.
    
- `Tick` 함수 내부에 `DeltaTime`을 곱한 **프레임 독립적 이동(`AddActorLocalOffset`) 및 회전 로직**을 구현합니다.
    
- `MyGameMode.cpp`: 생성자에서 `DefaultPawnClass = AMyPawn::StaticClass();`를 지정합니다.
    

### 3단계: 엔진 에디터에서 조립 및 마무리

코딩이 끝났다면 컴파일(빌드) 후 엔진으로 돌아와 마지막 연결 작업을 합니다.

1. **파생 블루프린트 생성 (강장 중요한 팁!)**:
    
    - C++로 만든 `MyPawn` 클래스를 우클릭하여 **블루프린트 클래스**를 생성합니다 (예: `BP_MyPawn`).
        
    - 이유: C++ 코드에 `UProperty(EditAnywhere)`로 뚫어놓은 `MoveAction`, `LookAction`, `DefaultMappingContext` 변수에 **1단계에서 만든 에셋들을 마우스 드롭으로 편하게 할당**하기 위해서입니다.
        
2. **월드 세팅 변경**:
    
    - [World Settings] 패널에서 `GameMode Override`를 본인이 만든 `MyGameMode`로 변경합니다.
        
    - 만약 GameMode도 블루프린트로 파생시켰다면 해당 BP를 지정하고, 그 내부에서 `Default Pawn Class`를 `BP_MyPawn`으로 설정합니다.

```cpp

#include "BasePawn.h"
#include "Components/CapsuleComponent.h"
#include "Components/SkeletalMeshComponent.h"
#include "GameFramework/SpringArmComponent.h"
#include "Camera/CameraComponent.h"
#include "EnhancedInputComponent.h"
#include "EnhancedInputSubsystems.h"

ABasePawn::ABasePawn()
{
    PrimaryActorTick.bCanEverTick = true;

    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));
    RootComponent = CapsuleComp;

    MeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("MeshComp"));
    MeshComp->SetupAttachment(RootComponent);

    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));
    SpringArmComp->SetupAttachment(RootComponent);

    CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComp"));
    CameraComp->SetupAttachment(SpringArmComp);

    CapsuleComp->SetSimulatePhysics(false);
    MeshComp->SetSimulatePhysics(false);

    MoveInput = FVector2D::ZeroVector;
    LookInput = FVector2D::ZeroVector;
}

void ABasePawn::BeginPlay()
{
    Super::BeginPlay();

    if (APlayerController* PlayerController = Cast<APlayerController>(GetController()))
    {
        if (UEnhancedInputLocalPlayerSubsystem* Subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PlayerController->GetLocalPlayer()))
        {
            if (DefaultIMC)
            {
                Subsystem->AddMappingContext(DefaultIMC, 0);
            }
        }
    }
}

void ABasePawn::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);

    if (!MoveInput.IsNearlyZero())
    {
        FVector ForwardDirection = GetActorForwardVector();
        FVector RightDirection = GetActorRightVector();
        FVector Direction = (ForwardDirection * MoveInput.Y) + (RightDirection * MoveInput.X);
        Direction.Normalize();

        AddActorLocalOffset(Direction * MoveSpeed * DeltaTime, true);
        MoveInput = FVector2D::ZeroVector;
    }

    if (!LookInput.IsNearlyZero())
    {
        FRotator NewRotation = FRotator(LookInput.Y * RotationSpeed * DeltaTime, LookInput.X * RotationSpeed * DeltaTime, 0.0f);
        AddActorLocalRotation(NewRotation);
        LookInput = FVector2D::ZeroVector;
    }
}

void ABasePawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
    Super::SetupPlayerInputComponent(PlayerInputComponent);

    if (UEnhancedInputComponent* EnhancedInputComponent = Cast<UEnhancedInputComponent>(PlayerInputComponent))
    {
        if (IAMove)
        {
            EnhancedInputComponent->BindAction(IAMove, ETriggerEvent::Triggered, this, &ABasePawn::Move);
        }
        if (IALook)
        {
            EnhancedInputComponent->BindAction(IALook, ETriggerEvent::Triggered, this, &ABasePawn::Look);
        }
    }
}

void ABasePawn::Move(const FInputActionValue& Value)
{
    MoveInput = Value.Get<FVector2D>();
}

void ABasePawn::Look(const FInputActionValue& Value)
{
    LookInput = Value.Get<FVector2D>();
}
```
