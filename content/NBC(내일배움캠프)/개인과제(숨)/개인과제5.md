#pragma once

#include "CoreMinimal.h"
#include "GameFramework/GameState.h"

// 과제 조건: 각 웨이브마다 제한 시간, 아이템 스폰 횟수의 베리에이션 설정
USTRUCT(BlueprintType)
struct FWaveData
{
	GENERATED_BODY()

	UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Wave")
	float WaveDuration = 30.0f;

	UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Wave")
	int32 SpawnItemCount = 40;
};

UCLASS()
class SPARTAPROJECT_API ASpartaGameState : public AGameState
{
	GENERATED_BODY()

public:
	ASpartaGameState();
	virtual void BeginPlay() override;

	UPROPERTY(VisibleAnywhere, BlueprintReadWrite, Category = "Score")
	int32 Score;
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Coin")
	int32 SpawnedCoinCount;
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Coin")
	int32 CollectedCoinCount;
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Level")
	float LevelDuration;
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Level")
	int32 CurrentLevelIndex;
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Level")
	int32 MaxLevels;
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Level")
	TArray<FName> LevelMapNames;

	// --- 기존 코드 기반 웨이브 시스템 변수 추가 ---
	UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Wave")
	TArray<FWaveData> WaveSettings;
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Wave")
	int32 CurrentWaveIndex;
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Wave")
	int32 MaxWaves;

	FTimerHandle LevelTimerHandle;
	FTimerHandle HUDUpdateTimerHandle;

	int32 GetScore() const;
	void AddScore(int32 Amount);
	
	void StartLevel();
	void StartWave(); // 웨이브 제어 핵심 로직
	void EndWave();   // 웨이브 분기 및 완료 처리
	void EndLevel();
	void OnLevelTimeUp();
	void OnCoinCollected();
	void OnGameOver();
	void UpdateHUD();
};

SpartaGameState.cpp
```cpp

#include "SpartaGameState.h"
#include "SpartaGameInstance.h"
#include "SpartaPlayerController.h"
#include "SpartaCharacter.h"
#include "SpawnVolume.h"
#include "CoinItem.h"
#include "Kismet/GameplayStatics.h"
#include "Components/TextBlock.h"
#include "Blueprint/UserWidget.h"

ASpartaGameState::ASpartaGameState()
{
	Score = 0;
	SpawnedCoinCount = 0;
	CollectedCoinCount = 0;
	LevelDuration = 30.0f;
	CurrentLevelIndex = 0;
	MaxLevels = 3;

	// 기존 데이터 양식에 맞춘 웨이브 초기화
	CurrentWaveIndex = 0;
	MaxWaves = 3;

	// 기획 베리에이션 세팅 (시간은 줄어들고, 아이템은 늘어남)
	FWaveData W1; W1.WaveDuration = 40.0f; W1.SpawnItemCount = 20;
	FWaveData W2; W2.WaveDuration = 30.0f; W2.SpawnItemCount = 35;
	FWaveData W3; W3.WaveDuration = 20.0f; W3.SpawnItemCount = 50;
	WaveSettings.Add(W1);
	WaveSettings.Add(W2);
	WaveSettings.Add(W3);
}

void ASpartaGameState::BeginPlay()
{
	Super::BeginPlay();
	
	StartLevel();
	
	GetWorldTimerManager().SetTimer(
		HUDUpdateTimerHandle,
		this,
		&ASpartaGameState::UpdateHUD,
		0.1f,
		true
	);
}

int32 ASpartaGameState::GetScore() const
{
	return Score;
}

void ASpartaGameState::AddScore(int32 Amount)
{
	if (UGameInstance* GameInstance = GetGameInstance())
	{
		USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
		if (SpartaGameInstance)
		{
			SpartaGameInstance->AddToScore(Amount);
		}
	}
}

void ASpartaGameState::StartLevel()
{
	if (UGameInstance* GameInstance = GetGameInstance())
	{
		USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
		if (SpartaGameInstance)
		{
			CurrentLevelIndex = SpartaGameInstance->CurrentLevelIndex;
		}
	}

	CurrentWaveIndex = 0; // 레벨 전환 시 웨이브 인덱스 초기화
	StartWave();
}

void ASpartaGameState::StartWave()
{
	// 기존 HUD 호출 규칙 유지
	if (APlayerController* PlayerController = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr)
	{
		if (ASpartaPlayerController* SpartaPlayerController = Cast<ASpartaPlayerController>(PlayerController))
		{
			SpartaPlayerController->ShowGameHUD();
		}
	}

	// [조건 충족] 알림 출력 규칙 적용
	FString WaveAnnounce = FString::Printf(TEXT("Wave %d 시작!"), CurrentWaveIndex + 1);
	if (GEngine)
	{
		GEngine->AddOnScreenDebugMessage(-1, 4.0f, FColor::Orange, WaveAnnounce);
	}
	UE_LOG(LogTemp, Warning, TEXT("%s"), *WaveAnnounce);

	if (WaveSettings.IsValidIndex(CurrentWaveIndex))
	{
		LevelDuration = WaveSettings[CurrentWaveIndex].WaveDuration;
		SpawnedCoinCount = 0;
		CollectedCoinCount = 0;

		TArray<AActor*> FoundVolumes;
		UGameplayStatics::GetAllActorsOfClass(GetWorld(), ASpawnVolume::StaticClass(), FoundVolumes);

		// 베리에이션 개수 적용
		int32 ItemToSpawn = WaveSettings[CurrentWaveIndex].SpawnItemCount;
	
		if (FoundVolumes.Num() > 0)
		{
			ASpawnVolume* SpawnVolume = Cast<ASpawnVolume>(FoundVolumes[0]);
			if (SpawnVolume)
			{
				for (int32 i = 0; i < ItemToSpawn; i++)
				{
					AActor* SpawnedActor = SpawnVolume->SpawnRandomItem();
					if (SpawnedActor && SpawnedActor->IsA(ACoinItem::StaticClass()))
					{
						SpawnedCoinCount++;
					}
				}
			}
		}
	}

	GetWorldTimerManager().SetTimer(
		LevelTimerHandle,
		this,
		&ASpartaGameState::OnLevelTimeUp,
		LevelDuration,
		false
	);

	UpdateHUD();
}

void ASpartaGameState::OnLevelTimeUp()
{
	EndWave();
}

void ASpartaGameState::OnCoinCollected()
{
	CollectedCoinCount++;
	
	if (SpawnedCoinCount > 0 && CollectedCoinCount >= SpawnedCoinCount)
	{
		EndWave();
	}
}

void ASpartaGameState::EndWave()
{
	GetWorldTimerManager().ClearTimer(LevelTimerHandle);

	CurrentWaveIndex++;
	if (CurrentWaveIndex < MaxWaves)
	{
		StartWave(); // 레벨 전환 없이 다음 웨이브 연속 진행
	}
	else
	{
		EndLevel(); // 3웨이브 모두 완료 시 레벨 전환
	}
}

void ASpartaGameState::EndLevel()
{
	if (UGameInstance* GameInstance = GetGameInstance())
	{
		USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
		if (SpartaGameInstance)
		{
			AddScore(Score);
			CurrentLevelIndex++;
			SpartaGameInstance->CurrentLevelIndex = CurrentLevelIndex;
			
			if (CurrentLevelIndex >= MaxLevels)
			{
				OnGameOver();
				return;
			}
			
			if (LevelMapNames.IsValidIndex(CurrentLevelIndex))
			{
				UGameplayStatics::OpenLevel(GetWorld(), LevelMapNames[CurrentLevelIndex]);
			}
			else
			{
				OnGameOver();
			}
		}
	}
}

void ASpartaGameState::OnGameOver()
{
	if (APlayerController* PlayerController = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr)
	{
		if (ASpartaPlayerController* SpartaPlayerController = Cast<ASpartaPlayerController>(PlayerController))
		{
			SpartaPlayerController->SetPause(true);
			SpartaPlayerController->ShowMainMenu(true);
		}
	}
}

void ASpartaGameState::UpdateHUD()
{
	if (APlayerController* PlayerController = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr)
	{
		ASpartaPlayerController* SpartaPlayerController = Cast<ASpartaPlayerController>(PlayerController);
		if (SpartaPlayerController)
		{
			if (UUserWidget* HUDWidget = SpartaPlayerController->GetHUDWidget())
			{
				// 1. 제한 시간 출력
				if (UTextBlock* TimeText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("Time"))))
				{
					float RemainingTime = GetWorldTimerManager().GetTimerRemaining(LevelTimerHandle);
					TimeText->SetText(FText::FromString(FString::Printf(TEXT("Time: %.1f"), RemainingTime)));
				}
				
				// 2. 점수 출력
				if (UTextBlock* ScoreText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("Score"))))
				{
					if (UGameInstance* GameInstance = GetGameInstance())
					{
						USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
						if (SpartaGameInstance)
						{
							ScoreText->SetText(FText::FromString(FString::Printf(TEXT("Score: %d"), SpartaGameInstance->TotalScore)));
						}
					}
				}
				
				// 3. 레벨 및 웨이브 통합출력 변경
				if (UTextBlock* LevelIndexText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("Level"))))
				{
					LevelIndexText->SetText(FText::FromString(FString::Printf(TEXT("Level: %d (Wave %d)"), CurrentLevelIndex + 1, CurrentWaveIndex + 1)));
				}

				// 4. [과제 조건 완비] 메인 HUD에 실시간 플레이어 체력(HP) 정보 연동
				if (UTextBlock* HPText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("HPText"))))
				{
					if (ASpartaCharacter* PlayerChar = Cast<ASpartaCharacter>(PlayerController->GetPawn()))
					{
						HPText->SetText(FText::FromString(FString::Printf(TEXT("HP: %.0f / 100"), PlayerChar->GetHealth())));
					}
				}
			}
		}
	}
}

```
- **`WBP_HUD` 위젯**: 텍스트 컴포넌트를 하나 추가하시고, 이름을 디테일 패널 가장 상단에서 `HPText`로 변경해 주셔야 C++의 `UpdateHUD`가 에러 없이 체력을 실시간으로 그려줍니다.
    
- **`LevelMapNames` 설정**: `SpartaGameState` 블루프린트 자식 클래스의 디테일 패널에서 `LevelMapNames` 배열을 펼치고 플러스(+) 버튼을 눌러 인덱스 3개를 만든 뒤, 이동할 맵 이름(`BasicLevel`, `Level2`, `Level3` 등)을 기입해 주셔야 웨이브가 끝난 후 다음 레벨로 정상 이동합니다.

## 1단계: C++ 소스 코드 교체 및 빌드

가장 먼저 C++ 레이어에 웨이브 시스템과 HUD 체력 연동 시스템을 이식해야 합니다.

1. **`SpartaGameState.h` 열기**
    
    - 기존 코드를 전부 지우고, 바로 이전 답변에서 제공해 드린 `SpartaGameState.h` 최종 교정본 코드를 그대로 복사해서 붙여넣습니다.
        
2. **`SpartaGameState.cpp` 열기**
    
    - 기존 코드를 전부 지우고, 이전 답변의 `SpartaGameState.cpp` 최종 교정본 코드를 복사해서 붙여넣습니다.
        
3. **라이브 코딩 컴파일 또는 빌드**
    
    - 언리얼 에디터로 돌아와 화면 우측 하단의 라이브 코딩 버튼(단축키: `Ctrl + Alt + F11`)을 눌러 컴파일을 완료합니다.
        

## 2단계: 레벨 및 웨이브 베리에이션 에디터 세팅

C++에서 선언한 배열 데이터를 블루프린트 에디터에서 채워주어야 정상적인 레벨 이동과 웨이브가 구동됩니다.

1. 에셋 폴더에서 `ASpartaGameState`를 상속받아 만든 게임스테이트 블루프린트(예: `BP_SpartaGameState`)를 더블클릭하여 엽니다.
    
2. 왼쪽 성분 패널이나 **디테일(Details) 패널**에서 아래 항목들을 찾아 설정합니다.
    

### 1. Level Map Names (전체 게임 진행 3번의 레벨 전환)

- `Level Map Names` 항목의 `+` 버튼을 3번 눌러 엘리먼트를 3개 생성합니다.
    
- 각 칸에 실제 제작하신 맵 이름을 정확히 입력합니다. (예: `BasicLevel`, `SecondLevel`, `ThirdLevel`)
    
- **주의**: 메인 메뉴가 있는 맵 이름은 여기에 넣지 마세요.
    

### 2. Wave Settings (각 레벨 안에서 3단계 웨이브 베리에이션)

- `Wave Settings` 항목 역시 기본적으로 C++에서 3개를 채워두었지만, 에디터에서 직접 수치를 조정할 수 있습니다.
    
- 각 인덱스를 펼쳐 과제 조건에 맞게 난이도를 다르게 조정합니다.
    
    - **Index 0 (Wave 1)**: `Wave Duration = 40.0`, `Spawn Item Count = 20` (쉬움)
        
    - **Index 1 (Wave 2)**: `Wave Duration = 30.0`, `Spawn Item Count = 35` (보통)
        
    - **Index 2 (Wave 3)**: `Wave Duration = 20.0`, `Spawn Item Count = 50` (어려움 - 제한시간은 짧고 템은 많음)
        

## 3단계: 인게임 HUD 위젯 재디자인 (`WBP_HUD`)

과제 조건인 "점수, 시간, 체력을 전부 한 화면에서 보기 좋게 배치"하고 **`HPText`** 바인딩을 완성하는 단계입니다.

1. 기존에 사용하던 HUD 위젯 블루프린트를 엽니다.
    
2. **계층구조(Hierarchy)** 패널을 아래와 같이 정돈합니다.
    
    - `Canvas Panel` (루트)
        
        - `Vertical Box` (화면 상단 중앙 앵커)
            
            - `TextBlock` (이름: **`Time`**) -> 타이머용
                
            - `TextBlock` (이름: **`Level`**) -> 레벨 및 웨이브 표시용
                
        - `Horizontal Box` (화면 좌측 상단 앵커)
            
            - `TextBlock` (이름: **`Score`**) -> 점수용
                
        - `Horizontal Box` (화면 하단 중앙 또는 좌측 하단 앵커 - **추가 필수**)
            
            - `TextBlock` (이름: **`HPText`**) -> ❗ **중요**: 이름을 반드시 정확히 `HPText`로 변경해야 C++에서 인식합니다.
                
3. **텍스트 스타일 디자인 업그레이드**:
    
    - `Time`, `Level`, `Score`, `HPText` 컴포넌트를 각각 클릭하고 우측 디테일 패널의 **`Font`** 항목을 확장합니다.
        
    - 가독성을 위해 `Outline Size(테두리)`를 `2`로 설정하고, Outline Color를 검은색으로 지정합니다.
        
    - 색상(`Color and Opacity`)을 시각적으로 직관적이게 변경합니다. (예: Time은 연한 주황색, Score는 황금색, HPText는 네온 그린)
        

## 4단계: 고급 메뉴 UI 재디자인 (메인 메뉴 및 게임 오버)

과제 조건인 "반투명 블러 효과"와 "버튼 Hover/Clicked 인터랙션 효과"를 적용하는 방법입니다.

### 1. 시각적 요소 개선 (반투명 블러 배경)

1. 메인 메뉴 위젯 또는 게임 오버 위젯 블루프린트를 엽니다.
    
2. 좌측 팔레트(Palette) 창에서 **`Background Blur`** 컴포넌트를 검색하여 `Canvas Panel` 바로 아래로 드래그합니다.
    
3. 계층구조에서 `Background Blur`가 모든 버튼과 텍스트보다 가장 위(배경 위치)에 오도록 배치하고, 화면 전체를 덮도록 계층을 조절(화면 전체 앵커 설정)합니다.
    
4. `Background Blur` 컴포넌트의 디테일 패널에서 **`Blur Strength`** 값을 **`7.0`** 정도로 설정합니다. 이렇게 하면 게임 오버나 메뉴 진입 시 뒷배경인 게임 화면이 고급스럽게 흐려집니다.
    

### 2. 버튼 인터랙션 효과 (Hover, Clicked) 적용

1. 위젯에 배치된 버튼(`StartButton`, `RestartButton`, `QuitButton` 등)을 선택합니다.
    
2. 우측 디테일 패널에서 **`Style`** 탭을 확장하면 `Normal`, `Hovered`, `Pressed` 상태가 나타납니다.
    
3. 각 상태의 `Tint` (색상 및 투명도)나 이미지를 변경하여 피드백을 줍니다.
    
    - **Normal**: 버튼의 기본 상태입니다. 약간 반투명한 어두운 회색 계열로 지정합니다.
        
    - **Hovered**: 마우스를 올렸을 때입니다. 플레이어가 "내가 이 버튼을 조준했다"는 것을 알 수 있게 **밝은 네온 컬러나 화이트 계열**로 지정하여 버튼이 밝아지도록 만듭니다.
        
    - **Pressed**: 버튼을 클릭하는 순간입니다. 어둡고 진한 색상(또는 주황색)으로 변경하여 실제로 꾹 눌린 듯한 느낌을 줍니다.
        

## 5단계: 버튼 클릭 시 C++ 및 블루프린트 함수 호출 연동

1. 메뉴 위젯의 **그래프(Graph)** 모드로 전환합니다.
    
2. 각 버튼의 **`On Clicked`** 이벤트를 생성하여 기능을 연결합니다.
    

### 게임 시작 / 재시작 버튼

- `On Clicked (StartButton)` 이벤트 뒤에 **`Get Player Controller`** -> **`Cast to SpartaPlayerController`** 노드를 배치합니다.
    
- 캐스팅 성공 노드 뒤에서 C++로 작성해 두신 **`Start Game`** 함수 노드를 검색하여 연결합니다. (점수 초기화 및 첫 레벨 오픈이 C++ 단에서 안전하게 실행됩니다.)
    

### 게임 종료 버튼 (과제 조건: 게임 종료 등 버튼 클릭 시 함수 호출)

- `On Clicked (QuitButton)` 이벤트 노드 뒤에 곧바로 블루프린트 빌트인 함수인 **`Quit Game`** 노드를 연결합니다. (특정 C++ 로직을 거칠 필요 없이 깔끔하게 클라이언트가 종료됩니다.)

	void OnGameOver(); // 플레이어가 사망하거나 모든 레벨의 최종 웨이브를 클리어했을 때 호출되며, 월드를 일시정지(Pause)시키고 컨트롤러를 통해 메인/결과 메뉴 UI를 띄웁니다.
	void UpdateHUD(); // HUDUpdateTimerHandle에 의해 실시간 루프를 돌며 PlayerController의 HUD 위젯을 찾아 화면의 남은시간 text, 점수 text, 레벨/웨이브 text, 플레이어 실시간 체력 text를 지속적으로 갱신합니다.
};

	FWaveData W3; W3.WaveDuration = 20.0f; W3.SpawnItemCount = 50; // 3웨이브 데이터: 제한시간 20초, 스폰 아이템 50개 세팅
	WaveSettings.Add(W1); // 1웨이브 기획 데이터를 WaveSettings 동적 배열의 0번 인덱스에 추가합니다.
	WaveSettings.Add(W2); // 2웨이브 기획 데이터를 WaveSettings 동적 배열의 1번 인덱스에 추가합니다.
	WaveSettings.Add(W3); // 3웨이브 기획 데이터를 WaveSettings 동적 배열의 2번 인덱스에 추가합니다.
}

// 게임 레벨이 본격적으로 가동될 때(액터의 초기화가 완료된 직후) 엔진에 의해 호출되는 시작 함수입니다.
void ASpartaGameState::BeginPlay()
{
	Super::BeginPlay(); // 부모 클래스(AGameState)의 BeginPlay 내부 로직을 먼저 실행하여 엔진 기본 기능을 보장합니다.
	
	StartLevel(); // 게임이 켜지자마자 첫 번째 레벨 및 첫 번째 웨이브 시스템을 가동시키는 함수를 호출합니다.
	
	// HUDUpdateTimerHandle을 통해 0.1초마다 반복(true)해서 UpdateHUD 함수를 강제로 실행시키는 타이머를 작동시킵니다.
	GetWorldTimerManager().SetTimer(
		HUDUpdateTimerHandle,
		this,
		&ASpartaGameState::UpdateHUD,
		0.1f,
		true
	);
}

// 외부(예: 캐릭터 등)에서 현재 GameState에 누적된 내부 점수(Score)를 조회할 때 사용하는 안전한 getter 함수입니다.
int32 ASpartaGameState::GetScore() const
{
	return Score; // 현재 가지고 있는 Score 변수값을 그대로 반환합니다.
}

// 플레이어가 아이템을 먹었을 때 전역 영구 저장소인 GameInstance에 점수를 누적하여 반영하는 함수입니다.
void ASpartaGameState::AddScore(int32 Amount)
{
	// 언리얼 엔진의 전역 싱글톤 객체인 GameInstance를 안전하게 가져옵니다.
	if (UGameInstance* GameInstance = GetGameInstance())
	{
		// 가져온 GameInstance를 프로젝트 전용 게임 인스턴스인 USpartaGameInstance 타입으로 형변환(Cast)합니다.
		USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
		if (SpartaGameInstance) // 형변환에 성공했다면 (유효한 포인터라면)
		{
			SpartaGameInstance->AddToScore(Amount); // 게임 인스턴스 내부의 전역 누적 점수 변수에 매개변수로 받은 Amount 만큼을 더해줍니다.
		}
	}
}

// 새로운 맵(레벨)이 열릴 때마다 호출되어, GameInstance로부터 이전에 저장된 레벨 인덱스를 동기화하는 함수입니다.
void ASpartaGameState::StartLevel()
{
	// 전역 영구 데이터 보관소인 GameInstance를 가져옵니다.
	if (UGameInstance* GameInstance = GetGameInstance())
	{
		// 우리가 커스텀 정의한 USpartaGameInstance 클래스 타입이 맞는지 체크 및 형변환을 진행합니다.
		USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
		if (SpartaGameInstance) // 형변환에 성공했다면
		{
			CurrentLevelIndex = SpartaGameInstance->CurrentLevelIndex; // 영구 보관되어 있던 레벨 인덱스 값을 가져와 현재 GameState의 Level 인덱스에 대입합니다.
		}
	}

	CurrentWaveIndex = 0; // 새로운 레벨 스테이지에 진입했으므로 내부 진행 웨이브 단계는 무조건 1단계(0번 인덱스)부터 초기화합니다.
	StartWave(); // 변수 정돈이 끝났으니 즉시 첫 번째 웨이브를 실시간으로 가동시킵니다.
}

// 실제 아이템을 스폰하고, 타이머를 켜며, 화면에 웨이브 시작 알림 메시지를 출력하는 핵심 제어 함수입니다.
void ASpartaGameState::StartWave()
{
	// 현재 월드의 첫 번째 로컬 플레이어 제어권자인 PlayerController 포인터를 안전하게 조회합니다.
	if (APlayerController* PlayerController = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr)
	{
		// 프로젝트 규칙용 커스텀 컨트롤러인 ASpartaPlayerController 클래스 타입이 맞는지 형변환합니다.
		if (ASpartaPlayerController* SpartaPlayerController = Cast<ASpartaPlayerController>(PlayerController))
		{
			SpartaPlayerController->ShowGameHUD(); // 레벨이 바뀌었거나 메뉴에서 돌아왔을 때 꺼져있을 수 있는 인게임 메인 HUD UI를 화면에 다시 띄웁니다.
		}
	}

	// [과제 조건 충족] 웨이브 시작 시 스크린(GEngine) 및 로그(UE_LOG)에 알림 문구를 즉시 출력하는 규칙을 수행합니다.
	FString WaveAnnounce = FString::Printf(TEXT("Wave %d 시작!"), CurrentWaveIndex + 1); // 현재 0번 인덱스에 +1을 하여 사람이 보기 편하게 "Wave 1 시작!" 형태의 문자열을 포맷팅합니다.
	if (GEngine) // 엔진 그래픽 출력 시스템이 정상 작동 중인지 체크합니다.
	{
		// 화면 좌측 상단에 -1(매번 새로운 줄 추가), 4초 동안 지속, 주황색 글씨로 조립한 웨이브 시작 알림 텍스트를 출력합니다.
		GEngine->AddOnScreenDebugMessage(-1, 4.0f, FColor::Orange, WaveAnnounce);
	}
	UE_LOG(LogTemp, Warning, TEXT("%s"), *WaveAnnounce); // 출력 로그 창에도 경고 레벨(Warning)의 노란색 텍스트로 웨이브 시작 로그를 상세 기록합니다.

	// 배열 범위(0 ~ 2 인덱스) 내에 현재 웨이브 번호가 정상적으로 위치하는지 안전성(유효성)을 검사합니다.
	if (WaveSettings.IsValidIndex(CurrentWaveIndex))
	{
		// 생성자에서 세팅한 기획 배열(WaveSettings)로부터 이번 단계의 지속 시간을 추출하여 제한 시간 변수에 대입합니다.
		LevelDuration = WaveSettings[CurrentWaveIndex].WaveDuration;
		SpawnedCoinCount = 0; // 아이템을 새로 뿌리기 전에 이번 웨이브에서 관리할 스폰 카운트를 0으로 완전히 리셋합니다.
		CollectedCoinCount = 0; // 플레이어가 이번 웨이브에서 먹은 코인 카운트도 0으로 완전히 청소합니다.

		TArray<AActor*> FoundVolumes; // 월드에 배치된 스폰 볼륨들을 찾아 임시 보관할 액터 포인터 배열을 선언합니다.
		// 월드 전체를 탐색하여 ASpawnVolume 클래스 및 이를 상속받은 모든 스폰 볼륨 액터들을 FoundVolumes 배열에 수집합니다.
		UGameplayStatics::GetAllActorsOfClass(GetWorld(), ASpawnVolume::StaticClass(), FoundVolumes);

		// 기획 베리에이션 데이터 적용: 이번 웨이브 인덱스에 정의된 아이템 스폰 갯수를 추출하여 로컬 변수에 저장합니다.
		int32 ItemToSpawn = WaveSettings[CurrentWaveIndex].SpawnItemCount;
	
		if (FoundVolumes.Num() > 0) // 월드에 스폰 볼륨이 최소 1개 이상 존재한다면
		{
			ASpawnVolume* SpawnVolume = Cast<ASpawnVolume>(FoundVolumes[0]); // 가장 첫 번째로 발견된 스폰 볼륨 하나를 가져와 형변환합니다.
			if (SpawnVolume) // 스폰 볼륨이 성공적으로 바인딩 되었다면
			{
				for (int32 i = 0; i < ItemToSpawn; i++) // 기획에서 지정한 스폰 개수(ItemToSpawn)만큼 루프(반복문)를 돌립니다.
				{
					AActor* SpawnedActor = SpawnVolume->SpawnRandomItem(); // 스폰 볼륨 클래스의 멤버 함수를 호출하여 임의의 위치에 무작위 아이템을 생성합니다.
					if (SpawnedActor && SpawnedActor->IsA(ACoinItem::StaticClass())) // 성공적으로 스폰되었고, 그 액터의 실제 클래스 종류가 코인(ACoinItem)이 맞는지 검사합니다.
					{
						SpawnedCoinCount++; // 생성된 액터가 코인 아이템인 것이 확실하므로, 관리용 변수인 SpawnedCoinCount를 1 증가시킵니다.
					}
				}
			}
		}
	}

	// 이번 웨이브의 제한 시간(LevelDuration)이 만료되었을 때, 자동으로 OnLevelTimeUp() 함수를 호출해 줄 타이머 핸들을 가동합니다. (반복 없음 = false)
	GetWorldTimerManager().SetTimer(
		LevelTimerHandle,
		this,
		&ASpartaGameState::OnLevelTimeUp,
		LevelDuration,
		false
	);

	UpdateHUD(); // 웨이브 정보와 제한 시간이 새롭게 리셋되었으므로 화면 UI를 즉각 최신 정보로 갱신 호출합니다.
}

// LevelTimerHandle 타이머에 의해 지정된 웨이브 버티기 제한 시간이 완전히 끝났을(0초가 되었을 때) 때 엔진에 의해 자동 실행되는 함수입니다.
void ASpartaGameState::OnLevelTimeUp()
{
	EndWave(); // 시간이 다 소모되었으므로 현재 진행 중인 웨이브를 종료하고 다음 분기로 넘기는 함수를 실행합니다.
}

// 플레이어 캐릭터가 월드의 코인 아이템과 충돌(Overlap)하여 코인을 성공적으로 획득할 때마다 외부에서 호출해 주는 이벤트 함수입니다.
void ASpartaGameState::OnCoinCollected()
{
	CollectedCoinCount++; // 플레이어가 획득한 코인의 숫자를 실시간으로 1 증가시킵니다.
	
	// 만약 월드에 정상적으로 코인이 생성되었고(>0), 플레이어가 먹은 코인 개수가 스폰된 총 코인 개수 이상이 되었다면 (전량 수집 완료 조건)
	if (SpawnedCoinCount > 0 && CollectedCoinCount >= SpawnedCoinCount)
	{
		EndWave(); // 제한 시간이 아직 한참 남아있더라도 지체 없이 즉시 현재 웨이브를 조기 종료(클리어) 처리합니다.
	}
}

// 한 웨이브가 종료(타임업 또는 전량 수집)되었을 때 호출되어, 다음 단계를 실행할지 혹은 레벨을 전환할지 분기하는 핵심 제어부입니다.
void ASpartaGameState::EndWave()
{
	GetWorldTimerManager().ClearTimer(LevelTimerHandle); // 웨이브가 조기 클리어 되었을 상황을 대비하여 가동 중이던 제한 시간 타이머(LevelTimerHandle)를 안전하게 중지 및 해제합니다.

	CurrentWaveIndex++; // 다음 단계를 진행하기 위해 웨이브 진행 상태 인덱스 번호를 1 증가시킵니다.
	if (CurrentWaveIndex < MaxWaves) // 증가된 인덱스가 최대 웨이브 수(3단계)보다 작다면 (즉, 아직 이번 레벨에서 버텨야 할 웨이브가 남아있다면)
	{
		StartWave(); // 과제 조건 충족: 화면 전환(OpenLevel)을 전혀 하지 않고, 동일 맵 내에서 연속해서 다음 단계 웨이브(StartWave)를 곧바로 기동합니다.
	}
	else // 3단계 웨이브(Wave 1, 2, 3)를 한 판도 빠짐없이 모두 안전하게 클리어 완료했다면
	{
		EndLevel(); // 비로소 현재 스테이지를 완전히 끝내고 다음 맵으로 전환하기 위한 레벨 종료 함수를 호출합니다.
	}
}

// 한 레벨 스테이지의 3웨이브를 전부 끝마쳤을 때 호출되어, 총점 저장 및 다음 레벨 오픈 혹은 전체 게임 클리어를 처리하는 함수입니다.
void ASpartaGameState::EndLevel()
{
	// 전역 데이터 보관소인 GameInstance 인스턴스를 월드로부터 안전하게 획득합니다.
	if (UGameInstance* GameInstance = GetGameInstance())
	{
		// 프로젝트용 전용 클래스인 USpartaGameInstance 타입이 확실한지 형변환 및 유효 검사를 진행합니다.
		USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
		if (SpartaGameInstance) // 형변환에 정상적으로 성공했다면
		{
			AddScore(Score); // 이번 스테이지에서 얻은 Score 점수 데이터를 GameInstance 전 전역 통합 스코어 공간에 최종 병합 누적합니다.
			CurrentLevelIndex++; // 다음 맵으로 넘어가기 위해 레벨 인덱스를 1 증가시킵니다.
			SpartaGameInstance->CurrentLevelIndex = CurrentLevelIndex; // 인스턴스 보관소의 레벨 인덱스도 함께 최신화하여 다음 맵이 열려도 이 숫자가 기억되게 만듭니다.
			
			if (CurrentLevelIndex >= MaxLevels) // 만약 증가된 레벨 인덱스가 최대 제한 레벨 수(3개 레벨) 이상이 되었다면 (최종 스테이지까지 전부 우승 클리어 시)
			{
				OnGameOver(); // 더 이상 갈 맵이 없으므로 게임 승리 및 최종 종료 화면을 띄우기 위해 온게임오버 함수를 실행하고 끝냅니다.
				return; // 아래의 레벨 오픈 처리 코드로 내려가지 않도록 함수를 즉시 중단 및 탈출합니다.
			}
			
			// 에디터에서 설정한 레벨 배열(LevelMapNames) 범위 내에 다음 스테이지의 맵 이름이 유효하게 존재하는지 체크합니다.
			if (LevelMapNames.IsValidIndex(CurrentLevelIndex))
			{
				// 과제 조건 완비: 3번의 레벨 전환을 위해, 배열에 들어있는 다음 스테이지 맵 이름(FName)을 읽어와 언리얼 새로운 레벨을 완전히 오픈합니다.
				UGameplayStatics::OpenLevel(GetWorld(), LevelMapNames[CurrentLevelIndex]);
			}
			else // 만약 기획자가 에디터에서 맵 이름을 실수로 빼먹어 인덱스가 유효하지 않다면
			{
				OnGameOver(); // 예외 처리를 위해 게임을 즉시 중단시키고 결과/종료 창으로 유도합니다.
			}
		}
	}
}

// 플레이어가 체력을 모두 잃어 사망했거나, 전체 3레벨의 모든 최종 웨이브를 격파하고 완전히 우승했을 때 게임을 멈추는 종료 함수입니다.
void ASpartaGameState::OnGameOver()
{
	// 월드의 첫 번째 로컬 플레이어 컨트롤러 포인터를 유효한지 검사하며 안전하게 가져옵니다.
	if (APlayerController* PlayerController = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr)
	{
		// 프로젝트용 커스텀 플레이어 컨트롤러(ASpartaPlayerController) 클래스 타입이 맞는지 형변환을 진행합니다.
		if (ASpartaPlayerController* SpartaPlayerController = Cast<ASpartaPlayerController>(PlayerController))
		{
			SpartaPlayerController->SetPause(true); // 더 이상 플레이어나 물리 연산이 움직이지 못하도록 게임 내부 전역 세상을 '일시정지' 상태로 잠가버립니다.
			SpartaPlayerController->ShowMainMenu(true); // 컨트롤러 내부 함수를 호출하여 결과 화면 기능 및 마우스 커서가 포함된 메인/종료 메뉴 UI 위젯을 화면 전면에 배치합니다.
		}
	}
}

// 0.1초 타이머 주기 주기로 무한 반복 호출되며 화면 UMG UI의 텍스트 컴포넌트들을 실시간 동기화 갱신하는 렌더링 지원 함수입니다.
void ASpartaGameState::UpdateHUD()
{
	// 현재 월드의 로컬 플레이어 컨트롤러를 유효성 검증과 동시에 수집합니다.
	if (APlayerController* PlayerController = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr)
	{
		// 가독성 및 HUDWidget 참조를 위해 컨트롤러를 커스텀 아키텍처 클래스인 ASpartaPlayerController 타입으로 형변환합니다.
		ASpartaPlayerController* SpartaPlayerController = Cast<ASpartaPlayerController>(PlayerController);
		if (SpartaPlayerController) // 형변환 처리가 무사히 끝났다면
		{
			// 플레이어 컨트롤러가 현재 화면에 띄워둔 인게임 메인 HUD 위젯 객체의 메모리 주소를 안전하게 참조해옵니다.
			if (UUserWidget* HUDWidget = SpartaPlayerController->GetHUDWidget())
			{
				// 1. 제한 시간 실시간 UI 연동 및 출력
				// UMG 위젯 내부에서 디자인 컴포넌트 이름이 정확히 "Time"으로 지정된 텍스트 블록 에셋을 검색하여 가져옵니다.
				if (UTextBlock* TimeText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("Time"))))
				{
					// 가동 중인 LevelTimerHandle 타이머로부터 현재 0초를 향해 떨어지고 있는 '실시간 남은 초(시간)'를 계산해 추출합니다.
					float RemainingTime = GetWorldTimerManager().GetTimerRemaining(LevelTimerHandle);
					// 소수점 아래 첫째 자리까지 표시되도록 포맷팅하여 "Time: 24.5" 형태의 텍스트 문자열로 화면의 글자를 즉시 변경합니다.
					TimeText->SetText(FText::FromString(FString::Printf(TEXT("Time: %.1f"), RemainingTime)));
				}
				
				// 2. 누적 점수 실시간 UI 연동 및 출력
				// UMG 위젯 내 디자인 계층 구조에서 이름이 "Score"로 명명된 텍스트 블록 컴포넌트를 정밀 검색하여 가져옵니다.
				if (UTextBlock* ScoreText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("Score"))))
				{
					// 전역 보관소인 GameInstance 인터페이스를 재조회합니다.
					if (UGameInstance* GameInstance = GetGameInstance())
					{
						USpartaGameInstance* SpartaGameInstance = Cast<USpartaGameInstance>(GameInstance);
						if (SpartaGameInstance) // 형변환에 문제 없이 성공했다면
						{
							// 영구 누적 점수 필드인 TotalScore 값을 정수형 양식으로 추출하여 UI 스코어 텍스트 칸의 글자를 최신화합니다.
							ScoreText->SetText(FText::FromString(FString::Printf(TEXT("Score: %d"), SpartaGameInstance->TotalScore)));
						}
					}
				}
				
				// 3. 과제 조건 충족: 현재 몇 레벨의 몇 단계 웨이브 상태인지 한 화면에 통합 문자열로 실시간 출력 변경
				// 위젯 디자이너 툴에서 이름을 "Level"로 설정해 둔 텍스트 블록 컴포넌트를 계층 구조 내부에서 찾아 형변환합니다.
				if (UTextBlock* LevelIndexText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("Level"))))
				{
					// 인덱스 특성상 0부터 시작하므로 각각 +1을 처리하여 화면 상에는 직관적으로 "Level: 1 (Wave 2)" 같은 형태로 문자를 조립해 투사합니다.
					LevelIndexText->SetText(FText::FromString(FString::Printf(TEXT("Level: %d (Wave %d)"), CurrentLevelIndex + 1, CurrentWaveIndex + 1)));
				}

				// 4. [과제 조건 완비] 점수, 시간과 더불어 실시간 플레이어의 남은 체력(HP) 수치 데이터를 메인 HUD의 한 화면에 통합 연동합니다.
				// 위젯 디자이너 가이드에 따라 이름을 정확히 "HPText"로 지정해 둔 텍스트 블록 컴포넌트를 정밀 조준해 가져옵니다.
				if (UTextBlock* HPText = Cast<UTextBlock>(HUDWidget->GetWidgetFromName(TEXT("HPText"))))
				{
					// 플레이어 컨트롤러가 현재 빙의(Possess)하여 실시간으로 키보드/마우스 입력을 먹고 움직이는 메인 전사 캐릭터(Pawn) 객체를 획득합니다.
					if (ASpartaCharacter* PlayerChar = Cast<ASpartaCharacter>(PlayerController->GetPawn()))
					{
						// 캐릭터 클래스 내부 멤버 함수인 GetHealth()를 실시간 조회하여 소수점 없는 정수 형태 포맷인 "HP: 75 / 100" 형태로 화면에 지속적으로 그려줍니다.
						HPText->SetText(FText::FromString(FString::Printf(TEXT("HP: %.0f / 100"), PlayerChar->GetHealth())));
					}
				}
			}
		}
	}
}
