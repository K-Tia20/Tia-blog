#include "BasePawn.h"

#include "Camera/CameraComponent.h"
#include "Components/CapsuleComponent.h"
#include "Components/SkeletalMeshComponent.h"
#include "GameFramework/SpringArmComponent.h"

#include "EnhancedInputComponent.h"
#include "EnhancedInputSubsystems.h"

#include "InputAction.h"
#include "InputMappingContext.h"

#include "Engine/LocalPlayer.h"

ABasePawn::ABasePawn()
{
	PrimaryActorTick.bCanEverTick = true;

	// Capsule
	CapsuleComp =
		CreateDefaultSubobject<UCapsuleComponent>(TEXT("Capsule"));

	RootComponent = CapsuleComp;

	// Mesh
	MeshComp =
		CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("Mesh"));

	MeshComp->SetupAttachment(RootComponent);

	// SpringArm
	SpringArmComp =
		CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArm"));

	SpringArmComp->SetupAttachment(RootComponent);
	SpringArmComp->TargetArmLength = 300.f;

	// Camera
	CameraComp =
		CreateDefaultSubobject<UCameraComponent>(TEXT("Camera"));

	CameraComp->SetupAttachment(SpringArmComp);

	// Physics Off
	CapsuleComp->SetSimulatePhysics(false);
	MeshComp->SetSimulatePhysics(false);
}

void ABasePawn::BeginPlay()
{
	Super::BeginPlay();

	if (APlayerController* PC = Cast<APlayerController>(GetController()))
	{
		if (ULocalPlayer* LocalPlayer = PC->GetLocalPlayer())
		{
			if (UEnhancedInputLocalPlayerSubsystem* Subsystem =
				LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
			{
				if (IMC_Default)
				{
					Subsystem->AddMappingContext(IMC_Default, 0);
				}
			}
		}
	}
}

void ABasePawn::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

	// Move

	FVector MoveDelta =
		(GetActorForwardVector() * MoveInput.X +
		 GetActorRightVector() * MoveInput.Y)
		* MoveSpeed
		* DeltaTime;

	//기존코드
	//AddActorLocalOffset(MoveDelta, true);
	
	//수정코드
	AddActorWorldOffset(MoveDelta, true);
	
	//기존 코드 
	/*
	FRotator RotationDelta;

	RotationDelta.Pitch = LookInput.Y * RotationSpeed * DeltaTime;
	RotationDelta.Yaw = LookInput.X * RotationSpeed * DeltaTime;

	AddActorLocalRotation(RotationDelta);
	*/
	
	//신규 코드
	// Yaw - Pawn 전체 회전
	FRotator YawDelta(0.f, LookInput.X * RotationSpeed * DeltaTime, 0.f);
	AddActorLocalRotation(YawDelta);

	// Pitch - SpringArm만 회전
	FRotator PitchDelta(LookInput.Y * RotationSpeed * DeltaTime, 0.f, 0.f);
	FRotator NewSpringArmRot = SpringArmComp->GetRelativeRotation() + PitchDelta;
	NewSpringArmRot.Pitch = FMath::Clamp(NewSpringArmRot.Pitch, -60.f, 60.f);
	SpringArmComp->SetRelativeRotation(NewSpringArmRot);
	
	//신규 코드
	FVector VerticalDelta = FVector(0.f, 0.f, VerticalInput * MoveSpeed * DeltaTime);
	AddActorWorldOffset(VerticalDelta, true);

	
}

void ABasePawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);

	if (UEnhancedInputComponent* EIC =
		Cast<UEnhancedInputComponent>(PlayerInputComponent))
	{
		if (IA_Move)
		{
			EIC->BindAction(
				IA_Move,
				ETriggerEvent::Triggered,
				this,
				&ABasePawn::Move);
			
			EIC->BindAction(
				IA_Move,
				ETriggerEvent::Completed,
				this,
				&ABasePawn::StopMove);
		}

		if (IA_Look)
		{
			EIC->BindAction(
				IA_Look,
				ETriggerEvent::Triggered,
				this,
				&ABasePawn::Look);
			
			EIC->BindAction(
				IA_Look,
				ETriggerEvent::Completed,
				this,
				&ABasePawn::StopLook);
		}
		
		if (IA_UpDown)
		{
			EIC->BindAction(
				IA_UpDown,
				ETriggerEvent::Triggered,
				this,
				&ABasePawn::Vertical);
			EIC->BindAction(
				IA_UpDown,
				ETriggerEvent::Completed,
				this,
				&ABasePawn::StopVertical);

		}	
	}
}

void ABasePawn::Move(const FInputActionValue& Value)
{
	MoveInput = Value.Get<FVector2D>();
	//UE_LOG(LogTemp, Warning, TEXT("MoveInput X: %f Y: %f"), MoveInput.X, MoveInput.Y);
}

void ABasePawn::Look(const FInputActionValue& Value)
{
	LookInput = Value.Get<FVector2D>();
}

void ABasePawn::StopMove(const FInputActionValue& Value)
{
	MoveInput = FVector2D::ZeroVector;
}

void ABasePawn::StopLook(const FInputActionValue& Value)
{
	LookInput = FVector2D::ZeroVector;
}

void ABasePawn::Vertical(const FInputActionValue& Value)
{
	VerticalInput = Value.Get<float>();
}

void ABasePawn::StopVertical(const FInputActionValue& Value)
{
	VerticalInput = 0.f;
}
