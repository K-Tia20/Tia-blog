#include "WowPawn.h"  
#include "Components/CapsuleComponent.h"  
#include "Components/SkeletalMeshComponent.h"  
#include "GameFramework/SpringArmComponent.h"  
#include "Camera/CameraComponent.h"  
#include "EnhancedInputComponent.h"  
#include "EnhancedInputSubsystems.h"  
  
#include "Engine/LocalPlayer.h"  
  
AWowPawn::AWowPawn()  
{  
    // Tick은 꺼둡니다. (모든 연산은 입력 바인딩 함수 내부에서 직접 처리)  
    PrimaryActorTick.bCanEverTick = false;  
  
    // (1) 루트 충돌체 설정 (Capsule)    
    CapsuleComp = CreateDefaultSubobject<UCapsuleComponent>(TEXT("CapsuleComp"));  
    SetRootComponent(CapsuleComp);  
    CapsuleComp->SetSimulatePhysics(false); // 물리 대신 코드로 제어하기 위해 false
     
    // (2) 스켈레탈 메쉬 컴포넌트 생성 및 부착  
    MeshComp = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("MeshComp"));  
    MeshComp->SetupAttachment(RootComponent);  
    MeshComp->SetSimulatePhysics(false); // 물리 비활성화  
  
    // (3) 스프링 암 생성 및 부착  
    SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComp"));  
    SpringArmComp->SetupAttachment(RootComponent);  
    SpringArmComp->TargetArmLength = 400.0f;  
    SpringArmComp->bUsePawnControlRotation = false; // 기본 컨트롤러 회전 꺼둠  
  
    // (4) 카메라 컴포넌트 생성 및 부착  
    CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComp"));  
    CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);  
    CameraComp->bUsePawnControlRotation = false;  
  
    // 속도 초기값 설정  
    MoveSpeed = 500.0f;  
    RotationSpeed = 45.0f;  
}  
  
void AWowPawn::BeginPlay()  
{  
    Super::BeginPlay();  
  
    // Enhanced Input Subsystem에 IMC 등록  
    if (APlayerController* PC = Cast<APlayerController>(GetController()))  
    {        if (ULocalPlayer* LocalPlayer = PC->GetLocalPlayer())  
        {            if (UEnhancedInputLocalPlayerSubsystem* Subsystem = LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())  
            {                if (InputMappingContext)  
                {                    Subsystem->AddMappingContext(InputMappingContext, 0);  
                }            }        }    }}  
  
void AWowPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)  
{  
    Super::SetupPlayerInputComponent(PlayerInputComponent);  
  
    // Enhanced InputComponent로 캐스팅하여 액션 바인딩  
    if (UEnhancedInputComponent* EnhancedInput = Cast<UEnhancedInputComponent>(PlayerInputComponent))  
    {        if (MoveAction)  
        {            // IA_Move 액션 키를 "키를 누르고 있는 동안" Move() 호출  
            EnhancedInput->BindAction(MoveAction, ETriggerEvent::Triggered, this, &AWowPawn::Move);  
        }  
        if (LookAction)  
        {            // IA_Look 액션 마우스가 "움직일 때" Look() 호출  
            EnhancedInput->BindAction(LookAction, ETriggerEvent::Triggered, this, &AWowPawn::Look);  
        }    }}  
  
void AWowPawn::Move(const FInputActionValue& Value)  
{  
    // 컨트롤러가 있어야 방향 계산 및 이동이 가능  
    if (!Controller) return;  
  
    // Value는 Axis2D로 설정된 IA_Move의 입력값 (WASD)을 담고 있음  
    const FVector2D MoveInput = Value.Get<FVector2D>();  
    const float DeltaTime = GetWorld()->GetDeltaSeconds(); // 프레임 독립성을 위한 델타타임  
  
    // 폰의 Forward/Right 벡터를 기준으로 로컬 오프셋 직접 계산 및 이동  
    if (!FMath::IsNearlyZero(MoveInput.X))  
    {        // 전진/후진: 정면 벡터 방향 * 입력값 * 속도 * 델타타임  
        FVector ForwardOffset = GetActorForwardVector() * MoveInput.X * MoveSpeed * DeltaTime;  
        AddActorLocalOffset(ForwardOffset, true);  
    }  
    if (!FMath::IsNearlyZero(MoveInput.Y))  
    {        // 우측/좌측: 우측 벡터 방향 * 입력값 * 속도 * 델타타임  
        FVector RightOffset = GetActorRightVector() * MoveInput.Y * MoveSpeed * DeltaTime;  
        AddActorLocalOffset(RightOffset, true);  
    }}  
  
void AWowPawn::Look(const FInputActionValue& Value)  
{  
    // 컨트롤러가 있어야 회전이 가능  
    if (!Controller) return;  
  
    // 마우스의 X, Y 움직임을 2D 축으로 가져옴  
    const FVector2D LookInput = Value.Get<FVector2D>();  
    const float DeltaTime = GetWorld()->GetDeltaSeconds(); // 프레임 독립성을 위한 델타타임  
  
    // 엔진 기본 제공 제어 함수(AddControllerYawInput 등) 대신 AddActorLocalRotation 등으로 직접 제어  
    // 1. 좌우 회전 (Yaw) -> Pawn 전체를 Z축 기준으로 회전  
    if (!FMath::IsNearlyZero(LookInput.X))  
    {        float YawRotation = LookInput.X * RotationSpeed * DeltaTime;  
        AddActorLocalRotation(FRotator(0.0f, YawRotation, 0.0f));  
    }  
    // 2. 상하 회전 (Pitch) -> 시야 꺾임을 표현하기 위해 스프링 암 컴포넌트만 회전  
    if (!FMath::IsNearlyZero(LookInput.Y))  
    {        float PitchRotation = LookInput.Y * RotationSpeed * DeltaTime;  
        FRotator NewSpringArmRotation = SpringArmComp->GetRelativeRotation();  
        // 상하 회전 한계선 Clamp 처리 (-60도 ~ 60도)  
        NewSpringArmRotation.Pitch = FMath::Clamp(NewSpringArmRotation.Pitch + PitchRotation, -60.0f, 60.0f);  
        SpringArmComp->SetRelativeRotation(NewSpringArmRotation);  
    }}

### SetupAttachment
## 1. 부모 컴포넌트에만 붙이기
```cpp
MeshComp->SetupAttachment(RootComponent);
```
`MeshComp`를 `RootComponent`의 자식으로 붙인다

## 2. 부모 컴포넌트의 특정 소켓에 붙이기
```cpp
CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);
```
`CameraComp`를 `SpringArmComp`에 붙이되, 스프링 암이 제공하는 특정 소켓 위치에 붙인다

## 3. 캐릭터 Mesh의 특정 소켓에 붙이기
```cpp
WeaponComp->SetupAttachment(GetMesh(), TEXT("WeaponSocket"));
```
`WeaponComp`를 캐릭터 Mesh에 붙이되, `"WeaponSocket"`이라는 소켓 위치에 붙인다

---
### 소켓과 컴포넌트 차이

### 소켓
소켓은 주로 Mesh나 SpringArm 같은 부모 안에 있는 **이름 붙은 위치**

> 캐릭터 손 본에 `WeaponSocket`이라는 지점을 만들어 둔다
> 그러면 무기나 이펙트를 그 위치에 붙일 수 있어요.
> 즉, 소켓은 **붙일 자리**입니다.

## 컴포넌트
컴포넌트는 액터를 구성하는 **실제 기능 단위**

> - `CameraComponent` → 화면을 보는 기능
> - `StaticMeshComponent` → 물체를 보이게 하는 기능
> - `BoxComponent` → 충돌 영역 기능
> - `SpringArmComponent` → 카메라 거리 조절 기능
> 즉, 컴포넌트는 **실제로 동작하는 부품**입니다.

---

### **전방 선언(Forward Declaration)**

**컴파일 시간을 줄이고, 헤더 의존성(파일 간 의존성을 줄이기 위해서)을 낮추기 위해** 사용

이 클래스의 자세한 내용은 지금 몰라도 되지만, **그런 이름의 클래스가 존재한다는 것만 컴파일러에게 알려주는 것**
```cpp
class UCapsuleComponent;
class USkeletalMeshComponent;
```
아래처럼 **포인터로만 선언**하고 있기 때문

```cpp
UCapsuleComponent* CapsuleComp;
USkeletalMeshComponent* MeshComp;
```

- `#include`  
    → “설명서 전체를 가져오기”

- `class UCapsuleComponent;`  
    → “그런 부품이 있다는 이름만 적어두기”

원래는 include가 필요하지만, 헤더에서 포인터로만 들고 있을 때는 컴파일 부담을 줄이기 위해 전방 선언을 쓴다

그래서 **헤더에서는 최대한 가볍게 이름만 알려주고**, 실제 기능을 쓰는 `.cpp`에서 include하는 방식이 자주 사용됩니다.

---
### `struct`와 `class` 차이

**기본 접근 권한** : 둘 다 변수와 함수를 가질 수 있음.
- `struct`는 기본이 `public`
- `class`는 기본이 `private`

즉, `struct`는 보통 **데이터를 담는 용도**로 많이 쓰고,  
`class`는 **데이터와 동작을 함께 관리하는 객체**로 많이 씁니다.

---
## 포인터 `*`
`UInputAction* LookAction`에서 `LookAction`은 **Input Action 에셋을 직접 들고 있는 게 아니라, 그 에셋이 있는 위치를 가리키는 변수**
`LookAction`은 **입력 액션 에셋을 계속 가리켜야 해서 포인터**
## 참조 `&`
`const FInputActionValue& Value`에서 `Value`는 입력 이벤트가 발생했을 때 전달되는 **그 순간의 입력값**을 함수 안에서 잠깐 읽기 위한 이름이에요.
`Value`는 **입력 순간에 전달된 값을 복사 없이 읽기만 해서 참조**

| 구분         | 포인터 `*`   | 참조 `&`      |
| ---------- | --------- | ----------- |
| 느낌         | 주소를 들고 있음 | 기존 값을 빌려 봄  |
| 예시 역할      | 에셋을 연결해둠  | 입력값을 잠깐 읽음  |
| 비유         | 주소 메모지    | 같이 보는 종이    |
| 비어 있을 수 있음 | 가능        | 보통 불가능하게 다룸 |

void AWowPawn::Move(const FInputActionValue& Value)
// 이동 입력 처리 함수
{
    if (!Controller) return;
    // Controller 없으면 종료

    const FVector2D MoveInput =
        Value.Get<FVector2D>();
    // Value
    // 입력값
    //
    // Get<FVector2D>()
    // 2차원 벡터로 변환
    //
    // W,S,A,D 값 저장

    const float DeltaTime =
        GetWorld()->GetDeltaSeconds();
    // 이전 프레임부터 지난 시간
    //
    // 프레임 독립적인 이동을 위해 사용

    if (!FMath::IsNearlyZero(MoveInput.X))
    // X축 입력이 0이 아니면
    {
        FVector ForwardOffset =
            GetActorForwardVector()
            * MoveInput.X
            * MoveSpeed
            * DeltaTime;
        // GetActorForwardVector()
        // 현재 Pawn 정면 방향
        //
        // MoveInput.X
        // 전후 입력값
        //
        // MoveSpeed
        // 이동속도
        //
        // DeltaTime
        // 프레임 보정

        AddActorLocalOffset(
            ForwardOffset,
            true
        );
        // Local 기준으로 이동
        //
        // true
        // 충돌 검사 수행
    }

    if (!FMath::IsNearlyZero(MoveInput.Y))
    // Y축 입력이 0이 아니면
    {
        FVector RightOffset =
            GetActorRightVector()
            * MoveInput.Y
            * MoveSpeed
            * DeltaTime;
        // GetActorRightVector()
        // Pawn의 오른쪽 방향

        AddActorLocalOffset(
            RightOffset,
            true
        );
        // 좌우 이동
    }
}



void AWowPawn::Look(const FInputActionValue& Value)
// 마우스 시점 입력 처리 함수
{
    if (!Controller) return;
    // Controller 없으면 종료

    const FVector2D LookInput =
        Value.Get<FVector2D>();
    // 마우스 이동값 가져오기

    const float DeltaTime =
        GetWorld()->GetDeltaSeconds();
    // 프레임 시간

    if (!FMath::IsNearlyZero(LookInput.X))
    // 좌우 입력 존재
    {
        float YawRotation =
            LookInput.X
            * RotationSpeed
            * DeltaTime;
        // 최종 회전값 계산

        AddActorLocalRotation(
            FRotator(
                0.0f,
                YawRotation,
                0.0f
            )
        );
        // FRotator
        // Pitch, Yaw, Roll
        //
        // Yaw만 회전
        //
        // Pawn 자체 회전
    }

    if (!FMath::IsNearlyZero(LookInput.Y))
    // 상하 입력 존재
    {
        float PitchRotation =
            LookInput.Y
            * RotationSpeed
            * DeltaTime;
        // 상하 회전량 계산

        FRotator NewSpringArmRotation =
            SpringArmComp->GetRelativeRotation();
        // 현재 스프링암 회전값 가져오기

        NewSpringArmRotation.Pitch =
            FMath::Clamp(
                NewSpringArmRotation.Pitch + PitchRotation,
                -60.0f,
                60.0f
            );
        // Clamp
        // 최소~최대 범위 제한
        //
        // Pitch를 -60~60도로 제한

        SpringArmComp->SetRelativeRotation(
            NewSpringArmRotation
        );
        // 최종 회전 적용
    }
}

    // 카메라를 스프링암 끝에 부착

    CameraComp->bUsePawnControlRotation = false;
    // 컨트롤러 회전값 자동 동기화 끔

    MoveSpeed = 500.f;
    // 캐릭터의 이동 속도 계수

    // 캐릭터 및 카메라의 회전 속도 계수 (초당 회전 각도)
}

void AWowPawn::Move(const FInputActionValue& Value)
{
    const FVector2D MoveInput = Value.Get<FVector2D>();
    // FVector2D
    // X(앞뒤), Y(좌우) 입력을 담는 2차원 벡터
    // Value.Get<FVector2D>()
    // 입력 시스템으로부터 전달된 실제 값 추출

    const float DeltaTime = GetWorld()->GetDeltaSeconds();
    // GetDeltaSeconds()
    // 프레임 사이의 시간 간격 (프레임 독립적인 이동을 위해 사용)

    FVector MoveOffset = (GetActorForwardVector() * MoveInput.X + GetActorRightVector() * MoveInput.Y) * MoveSpeed * DeltaTime;
    // GetActorForwardVector() * MoveInput.X
    // 캐릭터가 바라보는 방향으로 앞뒤 이동
    // GetActorRightVector() * MoveInput.Y
    // 캐릭터의 오른쪽 방향을 기준으로 좌우 이동
    // * MoveSpeed * DeltaTime
    // 속도와 시간을 곱해 이번 프레임에 움직일 거리 계산

    AddActorLocalOffset(MoveOffset, true);
    // AddActorLocalOffset
    // 계산된 벡터만큼 캐릭터 위치를 이동
    // MoveOffset
    // 이동할 거리와 방향
    // true (bSweep)
    // 이동 경로에 벽이 있으면 뚫지 못하게 충돌 검사 수행
}

void AWowPawn::Look(const FInputActionValue& Value)
{
    const FVector2D LookInput = Value.Get<FVector2D>();
    // X: 마우스 좌우 이동량, Y: 마우스 상하 이동량

    const float DeltaTime = GetWorld()->GetDeltaSeconds();

    float YawAmount = LookInput.X * RotationSpeed * DeltaTime;
    // Yaw
    // 좌우 회전 (Z축 기준 회전)
    // LookInput.X
    // 마우스 좌우 입력값

    AddActorLocalRotation(FRotator(0.f, YawAmount, 0.f));
    // AddActorLocalRotation
    // 캐릭터 본체 자체를 좌우로 회전 시킴

    float PitchAmount = LookInput.Y * RotationSpeed * DeltaTime;
    // Pitch
    // 상하 회전 (Y축 기준 회전)

    FRotator SpringArmRotation = SpringArmComp->GetRelativeRotation();
    // GetRelativeRotation
    // 스프링암의 현재 회전값(상대 좌표)을 가져옴

    SpringArmRotation.Pitch = FMath::Clamp(SpringArmRotation.Pitch + PitchAmount, -60.f, 60.f);
    // FMath::Clamp
    // 회전 각도를 특정 범위 내로 제한
    // -60.f, 60.f
    // 카메라가 너무 바닥을 보거나 머리 꼭대기 위로 넘어가지 않게 제한

    SpringArmComp->SetRelativeRotation(SpringArmRotation);
    // SetRelativeRotation
    // 계산된 Pitch 값이 적용된 회전으로 스프링암 각도 업데이트
}
